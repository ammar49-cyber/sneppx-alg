184int SNEPPX_profiler_record(SNEPPX_Profiler* prof, const char* name, float elapsed_ms) {
185 if (!prof || !name || !prof->enabled) return -1;
186 for (int i = 0; i < prof->num_entries; i++) {
187 if (strcmp(prof->entries[i].name, name) == 0) {
188 SNEPPX_ProfilerEntry* e = &prof->entries[i];
189 e->num_calls++;
190 e->total_time_ms += elapsed_ms;
191 if (elapsed_ms < e->min_time_ms) e->min_time_ms = elapsed_ms;
192 if (elapsed_ms > e->max_time_ms) e->max_time_ms = elapsed_ms;
193 e->avg_time_ms = e->total_time_ms / e->num_calls;
194 return 0;
195 }
196 }
197 if (prof->num_entries >= SNEPPX_PROFILER_MAX_ENTRIES) return -1;
198 SNEPPX_ProfilerEntry* e = &prof->entries[prof->num_entries++];
199 strncpy(e->name, name, SNEPPX_PROFILER_NAME_MAX - 1);
200 e->name[SNEPPX_PROFILER_NAME_MAX - 1] = '\0';
201 e->num_calls = 1;
202 e->total_time_ms = elapsed_ms;
203 e->min_time_ms = elapsed_ms;
204 e->max_time_ms = elapsed_ms;
205 e->avg_time_ms = elapsed_ms;
206 return 0;
207}
