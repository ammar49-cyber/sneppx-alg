41SNEPPXSimdLevel SNEPPX_simd_detect(void) {
42#if defined(__AVX512F__)
43 return SNEPPX_SIMD_AVX512;
44#elif defined(__AVX2__)
45 return SNEPPX_SIMD_AVX2;
46#elif defined(__SSE4_2__)
47 return SNEPPX_SIMD_SSE42;
48#else
49 /* Runtime CPUID check on x86 */
50#if defined(_MSC_VER)
51 int regs[4];
52 __cpuid(regs, 1);
53 int ecx = regs[2], edx = regs[1];
54 int has_sse42 = (ecx & (1 << 20)) != 0;
55 int has_avx = (ecx & (1 << 28)) != 0;
56 if (has_avx) {
57 __cpuid(regs, 7);
58 int has_avx2 = (regs[1] & (1 << 5)) != 0;
59 if (has_avx2) {
60 int regs7[4];
61 __cpuidex(regs7, 7, 0);
62 int has_avx512f = (regs7[1] & (1 << 16)) != 0;
63 if (has_avx512f) return SNEPPX_SIMD_AVX512;
64 return SNEPPX_SIMD_AVX2;
65 }
66 }
67 if (has_sse42) return SNEPPX_SIMD_SSE42;
68 return SNEPPX_SIMD_SCALAR;
69#elif defined(__GNUC__)
70 unsigned int eax, ebx, ecx, edx;
71 if (__get_cpuid(1, &eax, &ebx, &ecx, &edx)) {
72 if (ecx & (1 << 28)) { /* OSXSAVE + AVX */
73 unsigned int eax7, ebx7, ecx7, edx7;
74 if (__get_cpuid(7, &eax7, &ebx7, &ecx7, &edx7)) {
75 if (ebx7 & (1 << 5)) { /* AVX2 */
76 if (ebx7 & (1 << 16)) return SNEPPX_SIMD_AVX512; /* AVX512F */
77 return SNEPPX_SIMD_AVX2;
78 }
79 }
80 }
81 if (ecx & (1 << 20)) return SNEPPX_SIMD_SSE42; /* SSE4.2 */
82 }
83 return SNEPPX_SIMD_SCALAR;
84#else
85 return SNEPPX_SIMD_SCALAR;
86#endif
87#endif
88}

124void SNEPPX_gemm_scalar(const float* A, const float* B, float* C,
125 int M, int N, int K, float alpha, float beta) {
126 /* Zero / scale C */
127 for (int i = 0; i < M * N; i++) C[i] *= beta;
128
129 const int BM = SNEPPX_GEMM_BLOCK, BN = SNEPPX_GEMM_BLOCK, BK = SNEPPX_GEMM_BLOCK;
130 for (int i0 = 0; i0 < M; i0 += BM) {
131 int imax = (i0 + BM < M) ? i0 + BM : M;
132 for (int j0 = 0; j0 < N; j0 += BN) {
133 int jmax = (j0 + BN < N) ? j0 + BN : N;
134 for (int k0 = 0; k0 < K; k0 += BK) {
135 int kmax = (k0 + BK < K) ? k0 + BK : K;
136 for (int i = i0; i < imax; i++) {
137 const float* a_row = A + (size_t)i * K;
138 float* c_row = C + (size_t)i * N;
139 for (int j = j0; j < jmax; j++) {
140 float acc = 0.0f;
141 for (int k = k0; k < kmax; k++) {
142 acc += a_row[k] * B[(size_t)k * N + j];
143 }
144 c_row[j] += alpha * acc;
145 }
146 }
147 }
148 }
149 }

169void SNEPPX_gemm_avx2(const float* A, const float* B, float* C,
170 int M, int N, int K, float alpha, float beta) {
171 for (int i = 0; i < M * N; i++) C[i] *= beta;
172
173 const int BM = 64, BN = 256, BK = 128; /* tuned for L2 */
174 for (int i0 = 0; i0 < M; i0 += BM) {
175 int imax = (i0 + BM < M) ? i0 + BM : M;
176 for (int j0 = 0; j0 < N; j0 += BN) {
177 int jmax = (j0 + BN < N) ? j0 + BN : N;
178 for (int k0 = 0; k0 < K; k0 += BK) {
179 int kmax = (k0 + BK < K) ? k0 + BK : K;
180 for (int i = i0; i < imax; i++) {
181 const float* a_row = A + (size_t)i * K;
182 float* c_row = C + (size_t)i * N;
183 for (int j = j0; j < jmax; j += 8) {
184 __m256 acc = _mm256_setzero_ps();
185 for (int k = k0; k < kmax; k++) {
186 __m256 a_vec = _mm256_set1_ps(a_row[k]);
187 __m256 b_vec = _mm256_loadu_ps(&B[(size_t)k * N + j]);
188 acc = _mm256_fmadd_ps(a_vec, b_vec, acc);
189 }
190 __m256 c_vec = _mm256_loadu_ps(&c_row[j]);
191 c_vec = _mm256_fmadd_ps(_mm256_set1_ps(alpha), acc, c_vec);
192 _mm256_storeu_ps(&c_row[j], c_vec);
193 }
194 }
195 }
196 }
197 }

234static inline float sneppx_gelu_scalar(float x) {
235 /* tanh approximation: 0.5*x*(1 + tanh(sqrt(2/pi)*(x+0.044715*x^3))) */
236 float c = 0.7978845608f;
237 return 0.5f * x * (1.0f + tanhf(c * (x + 0.044715f * x * x * x)));
238}
