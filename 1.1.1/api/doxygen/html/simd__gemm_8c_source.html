32/* _mm256_exp_ps is an Intel SVML intrinsic not provided by GCC/Clang.
33 * Provide a portable per-lane scalar fallback so the AVX2 SiLU path builds
34 * and runs correctly on non-Intel compilers. */
35#if (defined(__GNUC__) || defined(__clang__)) && defined(__AVX2__)
36static inline __m256 sneppx_mm256_exp_ps(__m256 x) {
37 float v[8];
38 _mm256_storeu_ps(v, x);
39 for (int i = 0; i < 8; i++) v[i] = expf(v[i]);
40 return _mm256_loadu_ps(v);
41}
42#define _mm256_exp_ps(x) sneppx_mm256_exp_ps(x)
43#endif

54SNEPPXSimdLevel SNEPPX_simd_detect(void) {
55#if defined(__AVX512F__)
56 return SNEPPX_SIMD_AVX512;
57#elif defined(__AVX2__)
58 return SNEPPX_SIMD_AVX2;
59#elif defined(__SSE4_2__)
60 return SNEPPX_SIMD_SSE42;
61#else
62 /* Runtime CPUID check on x86 */
63#if defined(_MSC_VER)
64 int regs[4];
65 __cpuid(regs, 1);
66 int ecx = regs[2], edx = regs[1];
67 int has_sse42 = (ecx & (1 << 20)) != 0;
68 int has_avx = (ecx & (1 << 28)) != 0;
69 if (has_avx) {
70 __cpuid(regs, 7);
71 int has_avx2 = (regs[1] & (1 << 5)) != 0;
72 if (has_avx2) {
73 int regs7[4];
74 __cpuidex(regs7, 7, 0);
75 int has_avx512f = (regs7[1] & (1 << 16)) != 0;
76 if (has_avx512f) return SNEPPX_SIMD_AVX512;
77 return SNEPPX_SIMD_AVX2;
78 }
79 }
80 if (has_sse42) return SNEPPX_SIMD_SSE42;
81 return SNEPPX_SIMD_SCALAR;
82#elif defined(__GNUC__)
83 unsigned int eax, ebx, ecx, edx;
84 if (__get_cpuid(1, &eax, &ebx, &ecx, &edx)) {
85 if (ecx & (1 << 28)) { /* OSXSAVE + AVX */
86 unsigned int eax7, ebx7, ecx7, edx7;
87 if (__get_cpuid(7, &eax7, &ebx7, &ecx7, &edx7)) {
88 if (ebx7 & (1 << 5)) { /* AVX2 */
89 if (ebx7 & (1 << 16)) return SNEPPX_SIMD_AVX512; /* AVX512F */
90 return SNEPPX_SIMD_AVX2;
91 }
92 }
93 }
94 if (ecx & (1 << 20)) return SNEPPX_SIMD_SSE42; /* SSE4.2 */
95 }
96 return SNEPPX_SIMD_SCALAR;
97#else
98 return SNEPPX_SIMD_SCALAR;
99#endif
100#endif
101}

137void SNEPPX_gemm_scalar(const float* A, const float* B, float* C,
138 int M, int N, int K, float alpha, float beta) {
139 /* Zero / scale C */
140 for (int i = 0; i < M * N; i++) C[i] *= beta;
141
142 const int BM = SNEPPX_GEMM_BLOCK, BN = SNEPPX_GEMM_BLOCK, BK = SNEPPX_GEMM_BLOCK;
143 for (int i0 = 0; i0 < M; i0 += BM) {
144 int imax = (i0 + BM < M) ? i0 + BM : M;
145 for (int j0 = 0; j0 < N; j0 += BN) {
146 int jmax = (j0 + BN < N) ? j0 + BN : N;
147 for (int k0 = 0; k0 < K; k0 += BK) {
148 int kmax = (k0 + BK < K) ? k0 + BK : K;
149 for (int i = i0; i < imax; i++) {
150 const float* a_row = A + (size_t)i * K;
151 float* c_row = C + (size_t)i * N;
152 for (int j = j0; j < jmax; j++) {
153 float acc = 0.0f;
154 for (int k = k0; k < kmax; k++) {
155 acc += a_row[k] * B[(size_t)k * N + j];
156 }
157 c_row[j] += alpha * acc;
158 }
159 }
160 }
161 }
162 }

182void SNEPPX_gemm_avx2(const float* A, const float* B, float* C,
183 int M, int N, int K, float alpha, float beta) {
184 for (int i = 0; i < M * N; i++) C[i] *= beta;
185
186 const int BM = 64, BN = 256, BK = 128; /* tuned for L2 */
187 for (int i0 = 0; i0 < M; i0 += BM) {
188 int imax = (i0 + BM < M) ? i0 + BM : M;
189 for (int j0 = 0; j0 < N; j0 += BN) {
190 int jmax = (j0 + BN < N) ? j0 + BN : N;
191 for (int k0 = 0; k0 < K; k0 += BK) {
192 int kmax = (k0 + BK < K) ? k0 + BK : K;
193 for (int i = i0; i < imax; i++) {
194 const float* a_row = A + (size_t)i * K;
195 float* c_row = C + (size_t)i * N;
196 for (int j = j0; j < jmax; j += 8) {
197 __m256 acc = _mm256_setzero_ps();
198 for (int k = k0; k < kmax; k++) {
199 __m256 a_vec = _mm256_set1_ps(a_row[k]);
200 __m256 b_vec = _mm256_loadu_ps(&B[(size_t)k * N + j]);
201 acc = _mm256_fmadd_ps(a_vec, b_vec, acc);
202 }
203 __m256 c_vec = _mm256_loadu_ps(&c_row[j]);
204 c_vec = _mm256_fmadd_ps(_mm256_set1_ps(alpha), acc, c_vec);
205 _mm256_storeu_ps(&c_row[j], c_vec);
206 }
207 }
208 }
209 }
210 }

247static inline float sneppx_gelu_scalar(float x) {
248 /* tanh approximation: 0.5*x*(1 + tanh(sqrt(2/pi)*(x+0.044715*x^3))) */
249 float c = 0.7978845608f;
250 return 0.5f * x * (1.0f + tanhf(c * (x + 0.044715f * x * x * x)));
251}
