36static void grad_accum(SNEPPXTensor** grad, SNEPPXTensor* new_grad) {
37 if (!new_grad) return;
38 if (*grad) {
39 float* gd = (float*)(*grad)->data;
40 float* ng = (float*)new_grad->data;
41 size_t n = (*grad)->size < new_grad->size ? (*grad)->size : new_grad->size;
42 for (size_t i = 0; i < n; i++) gd[i] += ng[i];
43 SNEPPX_tensor_destroy(new_grad);
44 } else {
45 *grad = new_grad;
46 }
47}

55static void reduce_grad_to_shape(SNEPPXTensor** grad_ptr, const SNEPPXTensor* target) {
56 SNEPPXTensor* g = *grad_ptr;
57 if (!g) return;
58 if (g->ndim == target->ndim && g->size == target->size) return;
59 size_t max_ndim = g->ndim > target->ndim ? g->ndim : target->ndim;
60 size_t* g_shape = (size_t*)calloc(max_ndim, sizeof(size_t));
61 size_t* t_shape = (size_t*)calloc(max_ndim, sizeof(size_t));
62 if (!g_shape || !t_shape) { SNEPPX_free(g_shape, max_ndim * sizeof(size_t)); SNEPPX_free(t_shape, max_ndim * sizeof(size_t)); return; }
63 for (size_t i = 0; i < g->ndim; i++) g_shape[max_ndim - 1 - i] = g->shape[g->ndim - 1 - i];
64 for (size_t i = 0; i < target->ndim; i++) t_shape[max_ndim - 1 - i] = target->shape[target->ndim - 1 - i];
65 int broadcast = 0;
66 for (size_t i = 0; i < max_ndim; i++) { if (g_shape[i] != t_shape[i]) { broadcast = 1; break; } }
67 if (!broadcast) { free(g_shape); free(t_shape); return; }
68 size_t flat_size = 1;
69 for (size_t i = 0; i < target->ndim; i++) flat_size *= target->shape[i];
70 float* reduced = (float*)calloc(flat_size, sizeof(float));
71 if (!reduced) { free(g_shape); free(t_shape); return; }
72 size_t* strides_g = (size_t*)calloc(max_ndim, sizeof(size_t));
73 size_t* strides_t = (size_t*)calloc(target->ndim, sizeof(size_t));
74 if (!strides_g || !strides_t) { free(g_shape); free(t_shape); free(reduced); free(strides_g); free(strides_t); return; }
75 strides_t[target->ndim - 1] = 1;
76 for (size_t i = target->ndim; i > 1; i--) strides_t[i - 2] = strides_t[i - 1] * target->shape[i - 1];
77 strides_g[max_ndim - 1] = 1;
78 for (size_t i = max_ndim; i > 1; i--) strides_g[i - 2] = strides_g[i - 1] * g_shape[i - 1];
79 for (size_t i = 0; i < flat_size; i++) {
80 size_t g_idx = 0; size_t tmp = i;
81 for (size_t d = 0; d < target->ndim; d++) {
82 size_t t_dim_idx = tmp / strides_t[d];
83 tmp %= strides_t[d];
84 size_t g_dim = max_ndim - target->ndim + d;
85 size_t g_dim_idx = g_shape[g_dim] == 1 ? 0 : t_dim_idx;
86 g_idx += g_dim_idx * strides_g[g_dim];
87 }
88 reduced[i] = ((float*)g->data)[g_idx];
89 }
90 memcpy(g->data, reduced, flat_size * sizeof(float));
91 g->size = flat_size;
92 size_t new_ndim = target->ndim;
93 size_t* new_shape = (size_t*)SNEPPX_malloc(new_ndim * sizeof(size_t), 64);
94 if (new_shape) { memcpy(new_shape, target->shape, new_ndim * sizeof(size_t)); }
95 SNEPPX_free(g->shape, g->ndim * sizeof(size_t));
96 g->shape = new_shape;
97 g->ndim = new_ndim;
98 free(g_shape); free(t_shape); free(reduced); free(strides_g); free(strides_t);
99}

195static void backward_conv1d(void* ctx, SNEPPXTensor* grad_output) {
196 Conv1DCtx* c = (Conv1DCtx*)ctx;
197 float* go = (float*)grad_output->data;
198 size_t N = c->input->data->shape[0], C = c->input->data->shape[1], L = c->input->data->shape[2];
199 size_t K = c->kernel->data->shape[0];
200 size_t KH = c->kernel->data->shape[2];
201 size_t OH = (L + 2 * c->padding - KH) / c->stride + 1;
202
203 if (c->input->requires_grad) {
204 SNEPPXTensor* gi = SNEPPX_tensor_zeros(c->input->data->shape, c->input->data->ndim, SNEPPX_FLOAT32);
205 float* gid = (float*)gi->data;
206 float* kd = (float*)c->kernel->data->data;
207 for (size_t n = 0; n < N; n++)
208 for (size_t k = 0; k < K; k++)
209 for (size_t oh = 0; oh < OH; oh++) {
210 float grad_val = go[n * K * OH + k * OH + oh];
211 for (size_t kh = 0; kh < KH; kh++) {
212 size_t ih = oh * c->stride + kh - c->padding;
213 if (ih < L)
214 for (size_t cc = 0; cc < C; cc++)
215 gid[n * C * L + cc * L + ih] += kd[k * C * KH + cc * KH + kh] * grad_val;
216 }
217 }
218 grad_accum(&c->input->grad, gi);
219 }
220 if (c->kernel->requires_grad) {
221 SNEPPXTensor* gk = SNEPPX_tensor_zeros(c->kernel->data->shape, c->kernel->data->ndim, SNEPPX_FLOAT32);
222 float* gkd = (float*)gk->data;
223 float* id = (float*)c->input->data->data;
224 for (size_t n = 0; n < N; n++)
225 for (size_t k = 0; k < K; k++)
226 for (size_t oh = 0; oh < OH; oh++) {
227 float grad_val = go[n * K * OH + k * OH + oh];
228 for (size_t kh = 0; kh < KH; kh++) {
229 size_t ih = oh * c->stride + kh - c->padding;
230 if (ih < L)
231 for (size_t cc = 0; cc < C; cc++)
232 gkd[k * C * KH + cc * KH + kh] += id[n * C * L + cc * L + ih] * grad_val;
233 }
234 }
235 grad_accum(&c->kernel->grad, gk);
236 }

280 CumsumCtx* c = (CumsumCtx*)ctx;
281 if (!c->a->requires_grad) return;
282 float* go = (float*)grad_output->data;
283 size_t outer = 1, inner = 1;
284 for (size_t i = 0; i < c->dim; i++) outer *= c->a->data->shape[i];
285 size_t dim_size = c->a->data->shape[c->dim];
286 for (size_t i = c->dim + 1; i < c->a->data->ndim; i++) inner *= c->a->data->shape[i];

293 float cum = 0.0f;

298 }
299 }

1445static void backward_layer_norm(void* ctx, SNEPPXTensor* grad_output) {
1446 LayerNormCtx* c = (LayerNormCtx*)ctx;
1447 if (!c->a->requires_grad && !(c->gamma && c->gamma->requires_grad) && !(c->beta && c->beta->requires_grad)) return;
1448 size_t d = c->a->data->shape[c->a->data->ndim - 1];
1449 size_t n = c->a->data->size / d;
1450 float* xd = (float*)c->a->data->data;
1451 float* gd = c->gamma ? (float*)c->gamma->data->data : NULL;
1452 float* go = (float*)grad_output->data;
1453 float eps = c->eps;
1454 if (c->a->requires_grad) {
1455 SNEPPXTensor* gx = SNEPPX_tensor_zeros(c->a->data->shape, c->a->data->ndim, SNEPPX_FLOAT32);
1456 if (gx) {
1457 float* gxd = (float*)gx->data;
1458 for (size_t i = 0; i < n; i++) {
1459 double sum = 0.0, sum2 = 0.0;
1460 for (size_t j = 0; j < d; j++) { float v = xd[i * d + j]; sum += v; sum2 += (double)v * v; }
1461 double mean = sum / (double)d;
1462 double var = sum2 / (double)d - mean * mean;
1463 double inv_std = 1.0 / sqrt(var + (double)eps);
1464 for (size_t j = 0; j < d; j++) {
1465 double x_hat = (xd[i * d + j] - mean) * inv_std;
1466 double g = gd ? gd[j] : 1.0;
1467 gxd[i * d + j] = (float)(g * go[i * d + j]);
1468 }
1469 double dx_norm_sum = 0.0, dx_norm_x_hat_sum = 0.0;
1470 for (size_t j = 0; j < d; j++) {
1471 dx_norm_sum += gxd[i * d + j];
1472 dx_norm_x_hat_sum += gxd[i * d + j] * (xd[i * d + j] - mean) * inv_std;
1473 }
1474 for (size_t j = 0; j < d; j++) {
1475 double x_hat = (xd[i * d + j] - mean) * inv_std;
1476 double grad = gxd[i * d + j] - dx_norm_sum / (double)d - x_hat * dx_norm_x_hat_sum / (double)d;
1477 gxd[i * d + j] = (float)(grad * inv_std);
1478 }
1479 }
1480 grad_accum(&c->a->grad, gx);
1481 }
1482 }
1483 if (c->gamma && c->gamma->requires_grad) {
1484 SNEPPXTensor* gg = SNEPPX_tensor_zeros(c->gamma->data->shape, c->gamma->data->ndim, SNEPPX_FLOAT32);
1485 if (gg) {
1486 float* ggd = (float*)gg->data;
1487 for (size_t i = 0; i < n; i++) {
1488 double sum = 0.0, sum2 = 0.0;
1489 for (size_t j = 0; j < d; j++) { float v = xd[i * d + j]; sum += v; sum2 += (double)v * v; }
1490 double mean = sum / (double)d;
1491 double var = sum2 / (double)d - mean * mean;
1492 double inv_std = 1.0 / sqrt(var + (double)eps);
1493 for (size_t j = 0; j < d; j++) {
1494 double x_hat = (xd[i * d + j] - mean) * inv_std;
1495 ggd[j] += (float)(x_hat * go[i * d + j]);
1496 }
1497 }
1498 grad_accum(&c->gamma->grad, gg);
1499 }
1500 }
1501 if (c->beta && c->beta->requires_grad) {
1502 SNEPPXTensor* gb = SNEPPX_tensor_zeros(c->beta->data->shape, c->beta->data->ndim, SNEPPX_FLOAT32);
1503 if (gb) {
1504 float* gbd = (float*)gb->data;
1505 for (size_t i = 0; i < n; i++)
1506 for (size_t j = 0; j < d; j++)
1507 gbd[j] += go[i * d + j];
1508 grad_accum(&c->beta->grad, gb);
1509 }
1510 }
1511}

1560 size_t N = c->input->data->shape[0], C = c->input->data->shape[1];
1561 size_t H = c->input->data->shape[2], W = c->input->data->shape[3];
1562 size_t K = c->kernel->data->shape[0];
1563 size_t KH = c->kernel->data->shape[2], KW = c->kernel->data->shape[3];
1564 size_t OH = (H + 2 * c->pad_h - KH) / c->stride_h + 1;
1565 size_t OW = (W + 2 * c->pad_w - KW) / c->stride_w + 1;
1566
1567 if (c->input->requires_grad) {
1568 SNEPPXTensor* gi = SNEPPX_tensor_zeros(c->input->data->shape, c->input->data->ndim, SNEPPX_FLOAT32);
1569 float* gid = (float*)gi->data;
1570 float* kd = (float*)c->kernel->data->data;
1571 for (size_t n = 0; n < N; n++)
1572 for (size_t k = 0; k < K; k++)
1573 for (size_t oh = 0; oh < OH; oh++)
1574 for (size_t ow = 0; ow < OW; ow++) {
1575 float grad_val = go[n * K * OH * OW + k * OH * OW + oh * OW + ow];
1576 for (size_t kh = 0; kh < KH; kh++) {
1577 size_t ih = oh * c->stride_h + kh - c->pad_h;
1578 if (ih >= H) continue;
1579 for (size_t kw = 0; kw < KW; kw++) {
1580 size_t iw = ow * c->stride_w + kw - c->pad_w;
1581 if (iw >= W) continue;
1582 for (size_t cc = 0; cc < C; cc++)
1583 gid[n * C * H * W + cc * H * W + ih * W + iw] +=
1584 kd[k * C * KH * KW + cc * KH * KW + kh * KW + kw] * grad_val;
1585 }
1586 }
1587 }
1588 grad_accum(&c->input->grad, gi);
1589 }

1592 float* gkd = (float*)gk->data;
1593 float* id = (float*)c->input->data->data;
1594 for (size_t n = 0; n < N; n++)
1595 for (size_t k = 0; k < K; k++)
1596 for (size_t oh = 0; oh < OH; oh++)
1597 for (size_t ow = 0; ow < OW; ow++) {
1598 float grad_val = go[n * K * OH * OW + k * OH * OW + oh * OW + ow];
1599 for (size_t kh = 0; kh < KH; kh++) {
1600 size_t ih = oh * c->stride_h + kh - c->pad_h;
1601 if (ih >= H) continue;
1602 for (size_t kw = 0; kw < KW; kw++) {
1603 size_t iw = ow * c->stride_w + kw - c->pad_w;
1604 if (iw >= W) continue;
1605 for (size_t cc = 0; cc < C; cc++)
1606 gkd[k * C * KH * KW + cc * KH * KW + kh * KW + kw] +=

2815 gsum += (double)go[idx] * gamma_val;
2816 gxsum += (double)go[idx] * gamma_val * x_hat;
2817 }
2818 for (size_t i = 0; i < per_chn; i++) {

2821 gxd[idx] = (float)(gamma_val * inv_std * (go[idx] - (gsum * inv_N) - x_hat * (gxsum * inv_N)));
2822 }
2823 }

2825 }
2826 }

2836 }

2838 double var = sum_x2 * (double)inv_N - mean * mean;
2839 double inv_std = 1.0 / sqrt(var + (double)eps);
2840 double accum = 0.0;
2841 for (size_t i = 0; i < per_chn; i++) {
2842 size_t idx = ch * per_chn + i;
2843 double x_hat = (xd[idx] - mean) * inv_std;
2844 accum += (double)go[idx] * x_hat;
2845 }
2846 ggd[ch] = (float)accum;
2847 }
2848 grad_accum(&c->gamma->grad, gg);
2849 }
2850 }

2857 for (size_t i = 0; i < per_chn; i++)
2858 accum += (double)go[ch * per_chn + i];
2859 gbd[ch] = (float)accum;
2860 }
2861 grad_accum(&c->beta->grad, gb);
2862 }
2863 }
