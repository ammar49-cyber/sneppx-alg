34static size_t compute_offset_flat(const SNEPPXTensor* tensor, size_t flat_idx) {
35 size_t offset = 0;
36 size_t remaining = flat_idx;
37 for (size_t i = 0; i < tensor->ndim; i++) {
38 size_t idx = remaining / (tensor->size / (tensor->strides[i] * tensor->shape[i]));
39 if (tensor->size > 0 && tensor->strides[i] > 0) {
40 size_t dim_stride = tensor->strides[i];
41 size_t dim_size = tensor->shape[i];
42 idx = (flat_idx / dim_stride) % dim_size;
43 }
44 offset += idx * tensor->strides[i];
45 if (tensor->strides[i] > 0) {
46 remaining = flat_idx % tensor->strides[i];
47 } else {
48 remaining = 0;
49 }
50 }
51 return offset;
52}

217SNEPPXTensor* SNEPPX_tensor_create(const size_t* shape, size_t ndim, SNEPPXDtype dtype) {
218 if (ndim > 0 && !shape) return NULL;
219 SNEPPXTensor* tensor = (SNEPPXTensor*)aligned_alloc_wrapper(sizeof(SNEPPXTensor), 64);
220 if (!tensor) return NULL;
221
222 tensor->ndim = ndim;
223 tensor->dtype = dtype;
224 tensor->item_size = SNEPPX_tensor_dtype_size(dtype);
225 tensor->device = SNEPPX_DEVICE_CPU;
226 tensor->device_id = 0;
227 tensor->layout = SNEPPX_LAYOUT_ROW_MAJOR;
228 tensor->owns_data = 1;
229 tensor->backend_handle = NULL;
230 tensor->storage = NULL;
231 tensor->offset = 0;
232
233 size_t safe_ndim = ndim > 0 ? ndim : 1;
234 tensor->shape = (size_t*)aligned_alloc_wrapper(safe_ndim * sizeof(size_t), 64);
235 tensor->strides = (size_t*)aligned_alloc_wrapper(safe_ndim * sizeof(size_t), 64);
236 if (!tensor->shape || !tensor->strides) {
237 SNEPPX_free(tensor->shape, safe_ndim * sizeof(size_t));
238 SNEPPX_free(tensor->strides, safe_ndim * sizeof(size_t));
239 SNEPPX_free(tensor, sizeof(SNEPPXTensor));
240 return NULL;
241 }
242
243 size_t total = 1;
244 for (size_t i = 0; i < ndim; i++) {
245 tensor->shape[i] = shape[i];
246 total *= shape[i];
247 }
248 if (ndim == 0) total = 1;
249 tensor->size = total;
250
251 size_t stride = 1;

253 tensor->strides[i - 1] = stride;
254 stride *= shape[i - 1];
255 }

258 tensor->storage = SNEPPX_storage_create(num_bytes);
259 if (!tensor->storage) {

261 SNEPPX_free(tensor->strides, ndim * sizeof(size_t));
262 SNEPPX_free(tensor, sizeof(SNEPPXTensor));

863SNEPPXTensor* SNEPPX_tensor_tile(const SNEPPXTensor* src, const size_t* reps, size_t reps_ndim) {
864 if (!src || !reps) return NULL;
865 size_t out_ndim = (src->ndim > reps_ndim) ? src->ndim : reps_ndim;
866 size_t out_shape[16];
867 size_t offset = out_ndim - src->ndim;
868 for (size_t i = 0; i < out_ndim; i++) {
869 size_t s = (i < offset) ? 1 : src->shape[i - offset];
870 size_t r = (i < reps_ndim) ? reps[i] : 1;
871 out_shape[i] = s * r;
872 }
873 SNEPPXTensor* result = SNEPPX_tensor_empty(out_shape, out_ndim, src->dtype);
874 if (!result) return NULL;
875 unsigned char* src_data = (unsigned char*)src->data;
876 unsigned char* dst_data = (unsigned char*)result->data;
877 size_t is = src->item_size;
878 size_t* indices = (size_t*)aligned_alloc_wrapper(out_ndim * sizeof(size_t), 64);
879 if (!indices) { SNEPPX_tensor_destroy(result); return NULL; }
880 for (size_t flat = 0; flat < result->size; flat++) {
881 size_t tmp = flat;
882 for (size_t i = out_ndim; i > 0; i--) {
883 indices[i - 1] = (result->strides[i - 1] > 0) ? (tmp / result->strides[i - 1]) % result->shape[i - 1] : 0;
884 if (result->strides[i - 1] > 0) tmp %= result->strides[i - 1];
885 }
886 size_t src_flat = 0;
887 for (size_t i = 0; i < src->ndim; i++) {
888 size_t si = indices[offset + i] % src->shape[i];
889 src_flat += si * src->strides[i];
890 }
891 memcpy(dst_data + flat * is, src_data + src_flat * is, is);
892 }
893 SNEPPX_free(indices, out_ndim * sizeof(size_t));
894 return result;
895}

1760 if (a1 && b1) { if (ta) SNEPPX_tensor_destroy(ta); if (tb) SNEPPX_tensor_destroy(tb); return NULL; }
1761 size_t a_m = a1 ? 1 : a->shape[a->ndim - 2];

1768 size_t b_lead_ndim = b1 ? 0 : b->ndim - 2;

1770 for (size_t i = 0; i < a_lead_ndim; i++) a_lead[i] = a->shape[i];
1771 for (size_t i = 0; i < b_lead_ndim; i++) b_lead[i] = b->shape[i];
1772 size_t out_lead_ndim = a_lead_ndim > b_lead_ndim ? a_lead_ndim : b_lead_ndim;
1773 size_t out_lead[8];

1781 }

1783 size_t out_shape[8];
1784 for (size_t i = 0; i < out_lead_ndim; i++) out_shape[i] = out_lead[i];
1785 if (!a1) out_shape[out_lead_ndim] = a_m;
1786 if (!b1) out_shape[out_lead_ndim + (a1 ? 0 : 1)] = b_n;

1795

1800 long aoff = 0, boff = 0;

1803 long bi = (long)i - (long)(out_lead_ndim - b_lead_ndim);

1805 size_t bc = bi >= 0 ? (b_lead[bi] == 1 ? 0 : coords[i]) : 0;
1806 if (ai >= 0) { size_t pre = 1; for (long k = 0; k < ai; k++) pre *= a_lead[k]; aoff += (long)(ac * pre); }
1807 if (bi >= 0) { size_t pre = 1; for (long k = 0; k < bi; k++) pre *= b_lead[k]; boff += (long)(bc * pre); }
1808 }

1810 size_t Bbase = (size_t)boff * (b1 ? b_k : b_k * b_n);

2652SNEPPXTensor* SNEPPX_tensor_contiguous(const SNEPPXTensor* src) {
2653 if (!src) return NULL;
2654 if (SNEPPX_tensor_is_contiguous(src)) {
2655 SNEPPX_storage_retain(src->storage);
2656 return SNEPPX_tensor_as_strided(src, src->offset, src->shape, src->ndim, src->strides);
2657 }
2658 SNEPPXTensor* result = SNEPPX_tensor_create(src->shape, src->ndim, src->dtype);
2659 if (!result) return NULL;
2660 /* Strided copy: iterate over all indices */
2661 unsigned char* dst = (unsigned char*)result->data;
2662 unsigned char* src_data = (unsigned char*)src->data;
2663 size_t is = src->item_size;
2664 size_t* indices = (size_t*)aligned_alloc_wrapper(src->ndim * sizeof(size_t), 64);
2665 if (!indices) { SNEPPX_tensor_destroy(result); return NULL; }
2666 memset(indices, 0, src->ndim * sizeof(size_t));
2667 for (size_t flat = 0; flat < src->size; flat++) {
2668 size_t tmp = flat;
2669 for (size_t i = src->ndim; i > 0; i--) {
2670 indices[i - 1] = (src->strides[i - 1] > 0) ? (tmp / src->strides[i - 1]) % src->shape[i - 1] : 0;
2671 if (src->strides[i - 1] > 0) tmp %= src->strides[i - 1];
2672 }
2673 size_t offset = 0;
2674 for (size_t i = 0; i < src->ndim; i++)
2675 offset += indices[i] * src->strides[i];
2676 memcpy(dst + flat * is, src_data + offset * is, is);
2677 }
2678 SNEPPX_free(indices, src->ndim * sizeof(size_t));
2679 return result;
2680}
