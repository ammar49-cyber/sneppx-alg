94int SNEPPX_update_verifier_check(SNEPPXUpdateVerifier* uv, const SNEPPXSignedUpdate* update) {
95 if (!uv || !update) return 0;
96 if (!uv->verification_enabled) return 1;
97
98 if (uv->rollback_protection_enabled) {
99 uint32_t target[3] = {update->version_major, update->version_minor, update->version_patch};
100 if (SNEPPX_update_verifier_rollback_check(uv, target) != 0) return 0;
101 if (target[0] < uv->min_allowed_version[0]) return 0;
102 if (target[0] == uv->min_allowed_version[0] && target[1] < uv->min_allowed_version[1]) return 0;
103 if (target[0] == uv->min_allowed_version[0] && target[1] == uv->min_allowed_version[1] && target[2] < uv->min_allowed_version[2]) return 0;
104 }
105
106 SNEPPXBlake3State ctx;
107 uint8_t computed_hash[SNEPPX_UPDATE_HASH_LEN];
108 SNEPPX_blake3_init(&ctx);
109 SNEPPX_blake3_update(&ctx, (const uint8_t*)&update->version_major, sizeof(uint32_t) * 3);
110 SNEPPX_blake3_update(&ctx, (const uint8_t*)&update->chunk_count, sizeof(int));
111 SNEPPX_blake3_update(&ctx, (const uint8_t*)&update->timestamp, sizeof(uint64_t));
112 SNEPPX_blake3_finish(&ctx, computed_hash);
113
114 int sig_ok = SNEPPX_ct_equal(computed_hash, update->update_hash, SNEPPX_UPDATE_HASH_LEN) ? 1 : 0;
115 return sig_ok;
116}

237int SNEPPX_update_verifier_sign_update(SNEPPXSignedUpdate* update, const uint8_t* signing_key, size_t key_len) {
238 if (!update || !signing_key) return -1;
239 SNEPPXBlake3State ctx;
240 uint8_t hash[SNEPPX_UPDATE_HASH_LEN];
241 SNEPPX_blake3_init(&ctx);
242 SNEPPX_blake3_update(&ctx, (const uint8_t*)&update->version_major, sizeof(uint32_t) * 3);
243 SNEPPX_blake3_update(&ctx, (const uint8_t*)&update->chunk_count, sizeof(int));
244 SNEPPX_blake3_update(&ctx, (const uint8_t*)&update->timestamp, sizeof(uint64_t));
245 SNEPPX_blake3_update(&ctx, signing_key, key_len);
246 SNEPPX_blake3_finish(&ctx, hash);
247 memcpy(update->update_hash, hash, SNEPPX_UPDATE_HASH_LEN);
248 memset(update->signature, 0, SNEPPX_UPDATE_SIG_LEN);
249 for (size_t i = 0; i < SNEPPX_UPDATE_SIG_LEN && i < SNEPPX_UPDATE_HASH_LEN; i++)
250 update->signature[i] = hash[i] ^ 0xAA;
251 return 0;
252}

262int SNEPPX_update_verifier_verify_signature(const SNEPPXSignedUpdate* update, const uint8_t* public_key, size_t key_len) {
263 (void)public_key; (void)key_len;
264 if (!update) return -1;
265 uint8_t expected_hash[SNEPPX_UPDATE_HASH_LEN];
266 for (size_t i = 0; i < SNEPPX_UPDATE_HASH_LEN && i < SNEPPX_UPDATE_SIG_LEN; i++)
267 expected_hash[i] = update->signature[i] ^ 0xAA;
268 for (size_t i = SNEPPX_UPDATE_HASH_LEN; i < SNEPPX_UPDATE_SIG_LEN; i++)
269 expected_hash[i] = 0;
270 return (memcmp(expected_hash, update->update_hash, SNEPPX_UPDATE_HASH_LEN) == 0) ? 0 : -1;
271}
