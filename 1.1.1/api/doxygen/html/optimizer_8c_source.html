142static void step_sgd(SNEPPXOptimizer* opt, SNEPPXTensor** params, SNEPPXTensor** grads, size_t n) {
143 for (size_t i = 0; i < n; i++) {
144 if (!params[i] || !grads[i]) continue;
145 float* pd = (float*)params[i]->data;
146 float* gd = (float*)grads[i]->data;
147 float lr = opt->learning_rate, wd = opt->weight_decay, clip = opt->grad_clip;
148 float mu = opt->momentum, dp = opt->dampening;
149 float* md = opt->momentum_buffers[i] ? (float*)opt->momentum_buffers[i]->data : NULL;
150 size_t sz = params[i]->size;
151
152 if (md && mu > 0) {
153 for (size_t j = 0; j < sz; j++) {
154 float g = clip_grad(gd[j], clip);
155 g += wd * pd[j];
156 md[j] = mu * md[j] - lr * g;
157 pd[j] += md[j];
158 }
159 } else {
160 for (size_t j = 0; j < sz; j++) {
161 float g = clip_grad(gd[j], clip);
162 g += wd * pd[j];
163 pd[j] -= lr * g;
164 }
165 }
166 }

169static void step_adam(SNEPPXOptimizer* opt, SNEPPXTensor** params, SNEPPXTensor** grads, size_t n, int decoupled_wd) {
170 float lr = opt->learning_rate, b1 = opt->beta1, b2 = opt->beta2;
171 float eps = opt->epsilon, wd = opt->weight_decay, clip = opt->grad_clip;
172 float bias1 = 1.0f, bias2 = 1.0f;
173 for (size_t t = 0; t < opt->step_count; t++) { bias1 *= b1; bias2 *= b2; }
174 bias1 = 1.0f - bias1;
175 bias2 = 1.0f - bias2;
176
177 for (size_t i = 0; i < n; i++) {
178 if (!params[i] || !grads[i]) continue;
179 float* pd = (float*)params[i]->data;
180 float* gd = (float*)grads[i]->data;
181 float* m1 = (float*)opt->momentum_buffers[i]->data;
182 float* m2 = (float*)opt->state_buf2[i]->data;
183 size_t sz = params[i]->size;
184
185 for (size_t j = 0; j < sz; j++) {
186 float g = clip_grad(gd[j], clip);
187 m1[j] = b1 * m1[j] + (1.0f - b1) * g;
188 m2[j] = b2 * m2[j] + (1.0f - b2) * g * g;
189 float m1_hat = m1[j] / bias1;
190 float m2_hat = m2[j] / bias2;
191 float step = lr * m1_hat / (sqrtf(m2_hat) + eps);
192 if (decoupled_wd) step += lr * wd * pd[j];
193 pd[j] -= step;
194 if (!decoupled_wd) pd[j] -= lr * wd * pd[j];
195 }
196 }

199static void step_adamax(SNEPPXOptimizer* opt, SNEPPXTensor** params, SNEPPXTensor** grads, size_t n) {
200 float lr = opt->learning_rate, b1 = opt->beta1, b2 = opt->beta2;
201 float eps = opt->epsilon, wd = opt->weight_decay, clip = opt->grad_clip;
202 float bias1 = 1.0f;
203 for (size_t t = 0; t < opt->step_count; t++) bias1 *= b1;
204 bias1 = 1.0f - bias1;
205
206 for (size_t i = 0; i < n; i++) {
207 if (!params[i] || !grads[i]) continue;
208 float* pd = (float*)params[i]->data;
209 float* gd = (float*)grads[i]->data;
210 float* m1 = (float*)opt->momentum_buffers[i]->data;
211 float* m2 = (float*)opt->state_buf2[i]->data;
212 size_t sz = params[i]->size;
213
214 for (size_t j = 0; j < sz; j++) {
215 float g = clip_grad(gd[j], clip);
216 m1[j] = b1 * m1[j] + (1.0f - b1) * g;
217 m2[j] = fmaxf(b2 * m2[j], fabsf(g));
218 pd[j] -= lr * m1[j] / (bias1 * (m2[j] + eps)) + lr * wd * pd[j];
219 }
220 }

223static void step_rmsprop(SNEPPXOptimizer* opt, SNEPPXTensor** params, SNEPPXTensor** grads, size_t n) {
224 float lr = opt->learning_rate, alpha = opt->beta1, eps = opt->epsilon;
225 float wd = opt->weight_decay, clip = opt->grad_clip, mu = opt->momentum;
226
227 for (size_t i = 0; i < n; i++) {
228 if (!params[i] || !grads[i]) continue;
229 float* pd = (float*)params[i]->data;
230 float* gd = (float*)grads[i]->data;
231 float* v = (float*)opt->state_buf2[i]->data;
232 float* m = opt->momentum_buffers[i] ? (float*)opt->momentum_buffers[i]->data : NULL;
233 size_t sz = params[i]->size;
234
235 for (size_t j = 0; j < sz; j++) {
236 float g = clip_grad(gd[j], clip);
237 g += wd * pd[j];
238 v[j] = alpha * v[j] + (1.0f - alpha) * g * g;
239 float step = lr * g / (sqrtf(v[j]) + eps);
240 if (m && mu > 0) {
241 m[j] = mu * m[j] + step;
242 pd[j] -= m[j];
243 } else {
244 pd[j] -= step;
245 }
246 }
247 }

269static void step_adadelta(SNEPPXOptimizer* opt, SNEPPXTensor** params, SNEPPXTensor** grads, size_t n) {
270 float rho = opt->rho, eps = opt->epsilon, wd = opt->weight_decay, clip = opt->grad_clip;
271
272 for (size_t i = 0; i < n; i++) {
273 if (!params[i] || !grads[i]) continue;
274 float* pd = (float*)params[i]->data;
275 float* gd = (float*)grads[i]->data;
276 float* v = (float*)opt->state_buf2[i]->data;
277 float* d = (float*)opt->state_buf3[i]->data;
278 size_t sz = params[i]->size;
279
280 for (size_t j = 0; j < sz; j++) {
281 float g = clip_grad(gd[j], clip);
282 g += wd * pd[j];
283 v[j] = rho * v[j] + (1.0f - rho) * g * g;
284 float delta = g * sqrtf(d[j] + eps) / sqrtf(v[j] + eps);
285 d[j] = rho * d[j] + (1.0f - rho) * delta * delta;
286 pd[j] -= delta;
287 }
288 }

298void SNEPPX_optimizer_step(SNEPPXOptimizer* opt, SNEPPXTensor** params, SNEPPXTensor** grads, size_t num_params) {
299 if (!opt || !params || !grads || num_params == 0) return;
300 ensure_bufs(opt, params, num_params);
301 if (opt->num_params == 0) return;
302 opt->step_count++;
303
304 switch (opt->type) {
305 case SNEPPX_OPTIMIZER_SGD: step_sgd(opt, params, grads, num_params); break;
306 case SNEPPX_OPTIMIZER_ADAM: step_adam(opt, params, grads, num_params, 0); break;
307 case SNEPPX_OPTIMIZER_ADAMW: step_adam(opt, params, grads, num_params, 1); break;
308 case SNEPPX_OPTIMIZER_ADAMAX: step_adamax(opt, params, grads, num_params); break;
309 case SNEPPX_OPTIMIZER_RMSPROP: step_rmsprop(opt, params, grads, num_params); break;
310 case SNEPPX_OPTIMIZER_ADAGRAD: step_adagrad(opt, params, grads, num_params); break;
311 case SNEPPX_OPTIMIZER_ADADELTA: step_adadelta(opt, params, grads, num_params); break;
312 }

494void SNEPPX_lr_scheduler_step(SNEPPXLRScheduler* sched, float current_loss) {
495 if (!sched || !sched->lr_ptr) return;
496 sched->last_epoch++;
497
498 switch (sched->type) {
499 case SNEPPX_LR_STEP:
500 if (sched->step_size > 0 && sched->last_epoch % sched->step_size == 0)
501 *sched->lr_ptr *= sched->gamma;
502 break;
503 case SNEPPX_LR_EXPONENTIAL:
504 *sched->lr_ptr *= sched->gamma;
505 break;
506 case SNEPPX_LR_COSINE: {
507 float progress = (float)sched->last_epoch / (float)(sched->total_steps > 0 ? sched->total_steps : 1);
508 if (progress > 1.0f) progress = 1.0f;
509 *sched->lr_ptr = sched->min_lr + 0.5f * (sched->max_lr - sched->min_lr) * (1.0f + cosf((float)3.14159265f * progress));
510 break;
511 }
512 case SNEPPX_LR_REDUCE_ON_PLATEAU: {
513 int improved = sched->mode_min ? (current_loss < sched->best_loss) : (current_loss > sched->best_loss);
514 if (improved) {
515 sched->best_loss = current_loss;
516 sched->wait_count = 0;
517 } else {
518 sched->wait_count++;
519 if (sched->wait_count >= (int)sched->patience && sched->patience > 0) {
520 *sched->lr_ptr *= sched->factor;
521 sched->wait_count = 0;
522 }
523 }
524 break;
525 }
526 }
