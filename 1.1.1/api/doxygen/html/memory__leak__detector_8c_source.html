73int SNEPPX_leak_track_alloc(void *ptr, size_t size, const char *file, int line, const char *func) {
74 if (!ptr || !tracking_enabled) return -1;
75 if (alloc_count >= LEAK_MAX_ALLOCATIONS) return -1;
76 allocations[alloc_count].ptr = ptr;
77 allocations[alloc_count].size = size;
78 allocations[alloc_count].file = file;
79 allocations[alloc_count].line = line;
80 allocations[alloc_count].func = func;
81 allocations[alloc_count].timestamp = (uint64_t)clock();
82 allocations[alloc_count].freed = 0;
83 allocations[alloc_count].bt_depth = 0;
84 alloc_count++;
85 total_allocated += size;
86 current_usage += size;
87 if (current_usage > peak_usage) peak_usage = current_usage;
88 return 0;
89}

96int SNEPPX_leak_track_free(void *ptr) {
97 if (!ptr || !tracking_enabled) return -1;
98 for (int i = 0; i < alloc_count; i++) {
99 if (allocations[i].ptr == ptr && !allocations[i].freed) {
100 allocations[i].freed = 1;
101 current_usage -= allocations[i].size;
102 total_freed += allocations[i].size;
103 return 0;
104 }
105 }
106 return 1;
107}

116int SNEPPX_leak_check(leak_report_t *reports, int max_reports) {
