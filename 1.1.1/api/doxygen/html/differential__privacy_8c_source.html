58double SNEPPX_dp_gaussian_mech(double value, double sensitivity, double epsilon, double delta) {

185int SNEPPX_dp_dpsgd_step(double *params, const double *gradients, int n, double lr, double epsilon, double delta, double clip_norm) {
186 double norm = 0;
187 for (int i = 0; i < n; i++) norm += gradients[i] * gradients[i];
188 norm = sqrt(norm);
189 double scale = (norm > clip_norm) ? clip_norm / norm : 1.0;
190 double sigma = clip_norm * sqrt(2 * log(1.25 / delta)) / epsilon;
191 for (int i = 0; i < n; i++) {
192 double u1 = (double)rand() / RAND_MAX;
193 double u2 = (double)rand() / RAND_MAX;
194 double noise = sqrt(-2 * log(u1)) * cos(2 * 3.141592653589793 * u2) * sigma;
195 params[i] -= lr * (gradients[i] * scale + noise);
196 }
197 return 0;
198}
