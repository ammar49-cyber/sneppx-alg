274 if (!fingerprint) return -1;

315int SNEPPX_identity_verify_cert_chain(const uint8_t* cert_chain, int chain_len) {
316 if (!cert_chain || chain_len <= 0) return 0;
317 for (int i = 0; i < chain_len; i++) {
318 int offset = i * SNEPPX_CERT_FINGERPRINT_LEN;
319 uint8_t hash[SNEPPX_CERT_FINGERPRINT_LEN];
320 for (int j = 0; j < SNEPPX_CERT_FINGERPRINT_LEN; j++)
321 hash[j] = cert_chain[offset + j] ^ (uint8_t)(i + 1);
322 int valid = 0;
323 for (int j = 0; j < SNEPPX_CERT_FINGERPRINT_LEN; j++)
324 if (hash[j]) valid = 1;
325 if (!valid) return 0;
326 }
327 return 1;
