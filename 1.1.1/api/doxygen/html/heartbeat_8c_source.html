32static int64_t snepx_ns_now(void) {
33#ifdef _WIN32
34 FILETIME ft;
35 GetSystemTimeAsFileTime(&ft);
36 return ((int64_t)ft.dwHighDateTime << 32) | ft.dwLowDateTime;
37#else
38 struct timespec ts;
39 clock_gettime(CLOCK_MONOTONIC, &ts);
40 return (int64_t)ts.tv_sec * 1000000000LL + ts.tv_nsec;
41#endif
42}

193int SNEPPX_heartbeat_check_alive(SNEPPXHeartbeat* hb) {
194 if (!hb) return 0;
195 int64_t now = snepx_ns_now();
196 int alive_count = 0;
197 for (int i = 0; i < hb->world_size; i++) {
198 if (i == hb->rank) { alive_count++; continue; }
199 int64_t elapsed = now - hb->last_heartbeat_ns[i];
200 int64_t elapsed_ms = elapsed / 1000000LL;
201 if (elapsed_ms > hb->timeout_ms) {
202 hb->missed_count[i]++;
203 if (hb->missed_count[i] >= 3) {
204 hb->peer_status[i] = SNEPPX_HB_DEAD;
205 } else {
206 hb->peer_status[i] = SNEPPX_HB_SUSPECT;
207 }
208 } else {
209 hb->peer_status[i] = SNEPPX_HB_ALIVE;
210 alive_count++;
211 }
212 }
213 return alive_count;
214}
