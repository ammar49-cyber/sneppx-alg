68int SNEPPX_key_vault_generate_key(SNEPPXKeyVault* vault, uint8_t* key_id, uint64_t ttl_seconds) {
69 if (!vault || vault->is_locked || !key_id || vault->key_count >= SNEPPX_VAULT_MAX_KEYS) return -1;
70 SNEPPXVaultKey* k = &vault->keys[vault->key_count];
71 SNEPPX_random_bytes(k->id, SNEPPX_VAULT_ID_LEN);
72 SNEPPX_random_bytes(k->key_data, SNEPPX_VAULT_KEY_LEN);
73 k->created_at = (uint64_t)time(NULL);
74 k->expires_at = (ttl_seconds > 0) ? k->created_at + ttl_seconds : 0;
75 k->is_active = 1;
76 k->access_count = 0;
77 if (key_id) memcpy(key_id, k->id, SNEPPX_VAULT_ID_LEN);
78 return vault->key_count++;

167int SNEPPX_key_vault_unlock(SNEPPXKeyVault* vault, const uint8_t* master_key) {
168 if (!vault) return -1;
169 if (master_key_set && master_key) {
170 uint32_t check = 0;
171 for (int i = 0; i < 32 && i < SNEPPX_VAULT_KEY_LEN; i++)
172 check |= (uint8_t)(master_key[i] ^ master_key_hash[i]);
173 if (check != 0) return -1;
174 }
175 if (master_key && !master_key_set) {
176 for (int i = 0; i < 32 && i < SNEPPX_VAULT_KEY_LEN; i++)
177 master_key_hash[i] = master_key[i];
178 master_key_set = 1;
179 }
180 vault->is_locked = 0;
181 return 0;
182}

455static int vault_compute_checksum(const uint8_t* data, size_t len, uint8_t* sum_out) {
456 if (!data || !sum_out || len == 0) return -1;
457 uint32_t crc = 0xFFFFFFFF;
458 for (size_t i = 0; i < len; i++) {
459 crc ^= data[i];
460 for (int b = 0; b < 8; b++) crc = (crc >> 1) ^ (crc & 1 ? 0xEDB88320 : 0);
461 }
462 crc ^= 0xFFFFFFFF;
463 for (int i = 0; i < 4; i++) sum_out[i] = (uint8_t)(crc >> (i * 8));
464 return 0;
465}
