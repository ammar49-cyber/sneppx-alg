85static void snepx_ckpt_io_thread_fn(void* arg) {
86 struct SNEPPXCkptAsync* ckpt = (struct SNEPPXCkptAsync*)arg;
87 while (ckpt->io_running) {

103static int snepx_ckpt_write_file(const char* path, const void* data,
104 size_t data_size, int step) {
105 SNEPPXCheckpointHeader hdr;
106 memset(&hdr, 0, sizeof(hdr));
107 hdr.magic_lo = SNEPPX_CKPT_MAGIC;
108 hdr.magic_hi = SNEPPX_CKPT_MAGIC_HI;
109 hdr.version = SNEPPX_CKPT_VERSION;
110 hdr.num_tensors = 1;
111 hdr.metadata_offset = 0;
112 hdr.metadata_size = 0;
113
114 void* handle = NULL;
115 if (SNEPPX_ckpt_write_open(path, &hdr, &handle) != 0) {
116 /* Fallback: raw write */
117 FILE* f = fopen(path, "wb");
118 if (!f) return -1;
119 fwrite(&step, sizeof(int), 1, f);
120 fwrite(&data_size, sizeof(size_t), 1, f);
121 fwrite(data, 1, data_size, f);
122 fclose(f);
123 return 0;
124 }
125
126 SNEPPXTensorRecord rec;
127 memset(&rec, 0, sizeof(rec));
128 rec.shape[0] = (uint64_t)data_size;
129 rec.ndim = 1;
130 rec.dtype = 0; /* float32 */
131 rec.data_offset = 0;
132 rec.data_size = (uint64_t)data_size;
133
134 SNEPPX_ckpt_write_tensor(handle, data, &rec);
135 SNEPPX_ckpt_write_close(handle);
136 return 0;
137}

155int SNEPPX_checkpoint_init(SNEPPX_CheckpointCoord** cp,
156 const char* dir, int world_size, int rank,
157 int save_interval, int keep_last,
158 int async_enabled) {
159 if (!cp || !dir) return -1;
160 struct SNEPPXCkptAsync* ckpt = (struct SNEPPXCkptAsync*)calloc(1, sizeof(struct SNEPPXCkptAsync));
161 if (!ckpt) return -1;
162 strncpy(ckpt->checkpoint_dir, dir, 511);
163 ckpt->world_size = world_size;
164 ckpt->rank = rank;
165 ckpt->save_interval_steps = save_interval > 0 ? save_interval : 100;
166 ckpt->keep_last_n = keep_last > 0 ? keep_last : 5;
167 ckpt->current_step = 0;
168 ckpt->write_idx = 0;
169 ckpt->save_idx = -1;
170 ckpt->io_pending = 0;
171 ckpt->io_running = async_enabled;
172
173 cudaStreamCreateWithFlags(&ckpt->copy_stream, cudaStreamNonBlocking);
174 cudaStreamCreateWithFlags(&ckpt->save_stream, cudaStreamNonBlocking);
175
176 for (int i = 0; i < SNEPPX_CKPT_NUM_BUFFERS; i++) {
177 ckpt->buffers[i].host_buffer = NULL;
178 ckpt->buffers[i].host_capacity = 0;
179 ckpt->buffers[i].in_use = 0;
180 cudaEventCreateWithFlags(&ckpt->buffers[i].copy_event,
181 cudaEventDisableTiming);
182 }
183
184 if (async_enabled) {
185#ifdef _WIN32
186 void* arg = malloc(2 * sizeof(void*));
187 ((void**)arg)[0] = snepx_ckpt_io_thread_fn;
188 ((void**)arg)[1] = ckpt;
189 ckpt->io_thread = CreateThread(NULL, 0, snepx_thread_wrapper, arg, 0, NULL);
190#else
191 pthread_create(&ckpt->io_thread, NULL,
192 (void* (*)(void*))snepx_ckpt_io_thread_fn, ckpt);
193#endif
194 }
195
196 SNEPPX_CheckpointCoord* coord = (SNEPPX_CheckpointCoord*)calloc(1, sizeof(SNEPPX_CheckpointCoord));
197 if (!coord) { free(ckpt); return -1; }
198 coord->_async_state = ckpt;
199 coord->world_size = world_size;
200 coord->rank = rank;
201 coord->save_interval_steps = save_interval;
202 coord->keep_last_n = keep_last;
203 coord->async_save = async_enabled;
204 coord->current_step = 0;
205 coord->save_in_progress = 0;
206 *cp = coord;
207 return 0;
208}

245int sneppx_checkpoint_save(SNEPPX_CheckpointCoord* cp,
246 const void* model_state, size_t state_size,
247 int current_step, cudaStream_t stream) {
248 if (!cp || !model_state) return -1;
249 struct SNEPPXCkptAsync* ckpt = (struct SNEPPXCkptAsync*)cp->_async_state;
250 if (!ckpt) return -1;
251
252 ckpt->current_step = current_step;
253 if (current_step % ckpt->save_interval_steps != 0) return 0;
254
255 /* Wait for previous save to finish */
256 if (ckpt->io_pending) {
257 cudaStreamSynchronize(ckpt->save_stream);
258 ckpt->buffers[ckpt->save_idx].in_use = 0;
259 ckpt->io_pending = 0;
260 }
261
262 int buf_idx = ckpt->write_idx;
263 SNEPPXCkptBuffer* buf = &ckpt->buffers[buf_idx];
264
265 /* Allocate host buffer if needed */
266 if (!buf->host_buffer || buf->host_capacity < state_size) {
267 char* new_buf = (char*)realloc(buf->host_buffer, state_size);
268 if (!new_buf) return -1;
269 buf->host_buffer = new_buf;
270 buf->host_capacity = state_size;
271 }
272
273 /* Async D2D copy via temporary device buffer, then D2H */
274 void* d_tmp = NULL;
275 cudaError_t err = cudaMalloc(&d_tmp, state_size);
276 if (err != cudaSuccess) return -1;
277 cudaMemcpyAsync(d_tmp, model_state, state_size,
278 cudaMemcpyDeviceToDevice, ckpt->copy_stream);
279 cudaMemcpyAsync(buf->host_buffer, d_tmp, state_size,
280 cudaMemcpyDeviceToHost, ckpt->copy_stream);
281 cudaEventRecord(buf->copy_event, ckpt->copy_stream);
282 cudaStreamWaitEvent(ckpt->save_stream, buf->copy_event, 0);
283 cudaFree(d_tmp);
284
285 /* Build path */
286 char path[512];
287 sneppx_build_checkpoint_path(cp, current_step, path, sizeof(path));
288
289 /* Sync save stream, then write to disk */
290 cudaStreamSynchronize(ckpt->save_stream);
291 snepx_ckpt_write_file(path, buf->host_buffer, state_size, current_step);
292
293 strncpy(ckpt->last_checkpoint_path, path, 511);
294 ckpt->write_idx = (ckpt->write_idx + 1) % SNEPPX_CKPT_NUM_BUFFERS;
295 buf->in_use = 0;
296
297 cp->current_step = current_step;
298 if (cp->rank == 0)
299 printf("[SNEPPX Checkpoint] Saved to %s (step %d)\n", path, current_step);
300 return 0;
301}

303int sneppx_checkpoint_load(SNEPPX_CheckpointCoord* cp,
304 void* model_state, size_t state_size,
305 int* loaded_step) {
306 if (!cp || !model_state) return -1;
307 struct SNEPPXCkptAsync* ckpt = (struct SNEPPXCkptAsync*)cp->_async_state;
308 if (!ckpt) return -1;
309
310 if (ckpt->io_pending) {
311 cudaStreamSynchronize(ckpt->save_stream);
312 ckpt->buffers[ckpt->save_idx].in_use = 0;
313 ckpt->io_pending = 0;
314 }
315
316 char path[512];
317 if (ckpt->last_checkpoint_path[0]) {
318 strncpy(path, ckpt->last_checkpoint_path, 511);
319 } else {
320 sneppx_build_checkpoint_path(cp, ckpt->current_step, path, sizeof(path));
321 }
322
323 int step = ckpt->current_step;
324
325 /* Check if it's the new format */
326 if (SNEPPX_ckpt_validate(path) == 0) {
327 void* handle = NULL;
328 SNEPPXCheckpointHeader header;
329 if (SNEPPX_ckpt_read_open(path, &header, &handle) != 0) {
330 return -1;
331 }
332 size_t offset = 0;
333 for (uint32_t i = 0; i < header.num_tensors; i++) {
334 SNEPPXTensorRecord rec;
335 if (SNEPPX_ckpt_read_tensor(handle, i, (uint8_t*)model_state + offset, &rec) != 0) {
336 SNEPPX_ckpt_read_close(handle);
337 return -1;
338 }
339 offset += (size_t)rec.data_size;
340 }
341 SNEPPX_ckpt_read_close(handle);
342 } else {
343 /* Fallback: raw binary */
344 FILE* f = fopen(path, "rb");
345 if (!f) return -1;
346 int file_step;
347 size_t saved_size;
348 if (fread(&file_step, sizeof(int), 1, f) != 1) { fclose(f); return -1; }
349 if (fread(&saved_size, sizeof(size_t), 1, f) != 1) { fclose(f); return -1; }
350 if (saved_size > state_size) { fclose(f); return -1; }
351 char* host_buf = (char*)malloc(saved_size);
352 if (!host_buf) { fclose(f); return -1; }
353 fread(host_buf, 1, saved_size, f);
354 fclose(f);
355 cudaMemcpy(model_state, host_buf, saved_size, cudaMemcpyHostToDevice);
356 free(host_buf);
357 step = file_step;
358 }
359
360 if (loaded_step) *loaded_step = step;
361 ckpt->current_step = step;
362 cp->current_step = step;
363 if (cp->rank == 0) printf("[SNEPPX Checkpoint] Loaded from %s (step %d)\n", path, step);
364 return 0;
365}
