96int SNEPPX_transport_new_session(SNEPPXTransportSecurity* ts, const uint8_t* psk, size_t psk_len) {
97 if (!ts || !ts->enabled || ts->session_count >= SNEPPX_TLS_MAX_SESSIONS) return -1;
98 SNEPPXTLSSession* s = &ts->sessions[ts->session_count];
99 s->session_id = ts->session_count;
100 s->is_active = 1;
101 s->creation_time = (uint64_t)time(NULL);
102 s->last_used = s->creation_time;
103 if (psk && psk_len >= SNEPPX_TLS_KEY_LEN)

107 session_rekey_time[ts->session_count] = s->creation_time;
108 session_timeouts[ts->session_count] = 0;

145int SNEPPX_transport_encrypt(SNEPPXTransportSecurity* ts, int session_id,
146 const uint8_t* plaintext, size_t len,
147 uint8_t* ciphertext, uint8_t nonce[SNEPPX_TLS_NONCE_LEN]) {
148 if (!ts || !plaintext || !ciphertext || !nonce) return -1;
149 if (session_id < 0 || session_id >= ts->session_count) return -1;
150 SNEPPXTLSSession* s = &ts->sessions[session_id];
151 if (!s->is_active) return -1;
152 if (session_timeouts[session_id] > 0) {
153 uint64_t now = (uint64_t)time(NULL);
154 if (now - s->last_used > session_timeouts[session_id]) return -1;
155 }
156 SNEPPX_random_bytes(nonce, SNEPPX_TLS_NONCE_LEN);
157 if (use_aes_gcm) {
158 for (size_t i = 0; i < len; i++)
159 ciphertext[i] = plaintext[i] ^ s->session_key[i % SNEPPX_TLS_KEY_LEN];
160 } else {
161 SNEPPXChaCha20State state;
162 SNEPPX_chacha20_init(&state, s->session_key, nonce, 0);
163 memcpy(ciphertext, plaintext, len);
164 SNEPPX_chacha20_encrypt(&state, ciphertext, len);
165 }
166 s->last_used = (uint64_t)time(NULL);
167 session_rekey_counter[session_id]++;
168 g_total_encrypted += (uint64_t)len;
169 if (session_rekey_counter[session_id] >= 1000) {
170 uint8_t new_key[SNEPPX_TLS_KEY_LEN];
171 SNEPPX_random_bytes(new_key, SNEPPX_TLS_KEY_LEN);
172 memcpy(s->session_key, new_key, SNEPPX_TLS_KEY_LEN);
173 session_rekey_counter[session_id] = 0;
174 session_rekey_time[session_id] = (uint64_t)time(NULL);
175 }
176 return 0;
177}

225int SNEPPX_transport_noise_handshake(SNEPPXTransportSecurity* ts,
226 const uint8_t* prologue, size_t prologue_len,
227 uint8_t* handshake_msg, size_t* msg_len) {
228 (void)prologue; (void)prologue_len;
229 if (!ts || !handshake_msg || !msg_len) return -1;
230 if (*msg_len < 48) return -1;
231 SNEPPX_random_bytes(handshake_msg, 48);
232 if (ts->use_noise_protocol) {
233 uint8_t eph_key[32], eph_pub[32];
234 SNEPPX_random_bytes(eph_key, 32);
235 for (int i = 0; i < 32; i++)
236 eph_pub[i] = eph_key[i] ^ 0xAA;
237 for (int i = 0; i < 32; i++)
238 handshake_msg[i] ^= eph_pub[i];
239 }
240 *msg_len = 48;
241 return 0;
242}

270int SNEPPX_transport_cleanup_expired(SNEPPXTransportSecurity* ts, uint64_t max_age_seconds) {
271 if (!ts) return 0;
272 int cleaned = 0;
273 uint64_t now = (uint64_t)time(NULL);
274 for (int i = 0; i < ts->session_count; i++) {
275 if (ts->sessions[i].is_active) {
276 uint64_t age = now - ts->sessions[i].last_used;
277 if (age > max_age_seconds || (session_timeouts[i] > 0 && age > session_timeouts[i])) {
278 memset(ts->sessions[i].session_key, 0, SNEPPX_TLS_KEY_LEN);
279 ts->sessions[i].is_active = 0;
280 cleaned++;
281 }
282 }
283 }
284 g_active_sessions -= cleaned;
285 if (g_active_sessions < 0) g_active_sessions = 0;
286 return cleaned;

367 const uint8_t* plaintext, size_t len,
368 const uint8_t* aad, size_t aad_len,
369 uint8_t* ciphertext, uint8_t nonce[SNEPPX_TLS_NONCE_LEN]) {
370 (void)aad; (void)aad_len;
371 if (!ts || !plaintext || !ciphertext || !nonce) return -1;
372 if (session_id < 0 || session_id >= ts->session_count) return -1;
373 SNEPPXTLSSession* s = &ts->sessions[session_id];
