52static int find_concept_idx(SNEPPXSemanticGraph* graph, const char* key) {

61static int find_or_create_concept(SNEPPXSemanticGraph* graph, const char* key) {
62 int idx = find_concept_idx(graph, key);
63 if (idx >= 0) return idx;
64 if (graph->num_concepts >= graph->max_concepts) {
65 uint64_t oldest = graph->access_counter;
66 int lru_idx = -1;
67 for (size_t i = 0; i < graph->num_concepts; i++) {
68 if (graph->concepts[i].valid && graph->concepts[i].last_access < oldest) {
69 oldest = graph->concepts[i].last_access;
70 lru_idx = (int)i;
71 }
72 }
73 if (lru_idx >= 0) {
74 if (graph->concepts[lru_idx].embedding) {
75 SNEPPX_tensor_destroy(graph->concepts[lru_idx].embedding);
76 graph->concepts[lru_idx].embedding = NULL;
77 }
78 for (size_t e = 0; e < graph->num_edges; e++) {
79 if (graph->edges[e].valid &&
80 (graph->edges[e].from_idx == (size_t)lru_idx ||
81 graph->edges[e].to_idx == (size_t)lru_idx)) {
82 graph->edges[e].valid = 0;
83 }
84 }
85 memset(graph->concepts[lru_idx].key, 0, SNEPPX_MAX_CONCEPT_KEY_LEN);
86 graph->concepts[lru_idx].valid = 0;
87 idx = lru_idx;
88 } else {
89 return -1;
90 }
91 } else {
92 idx = (int)graph->num_concepts;
93 graph->num_concepts++;
94 }
95 return idx;
96}

98SNEPPXSemanticGraph* sneppx_semantic_create(const SNEPPXSemanticConfig* cfg, unsigned int* seed_ptr) {
99 SNEPPXSemanticGraph* graph = SNEPPX_malloc(sizeof(SNEPPXSemanticGraph), 64);
100 if (!graph) return NULL;
101 memset(graph, 0, sizeof(SNEPPXSemanticGraph));
102 graph->max_concepts = cfg->max_concepts > 0 ? cfg->max_concepts : 100;
103 graph->max_edges = cfg->max_edges > 0 ? cfg->max_edges : 500;
104 graph->embedding_dim = cfg->embedding_dim > 0 ? cfg->embedding_dim : 64;
105 graph->seed_ptr = seed_ptr;
106 graph->concepts = SNEPPX_malloc(graph->max_concepts * sizeof(SNEPPXSemanticConcept), 64);
107 if (!graph->concepts) {
108 SNEPPX_free(graph, sizeof(SNEPPXSemanticGraph));
109 return NULL;
110 }
111 memset(graph->concepts, 0, graph->max_concepts * sizeof(SNEPPXSemanticConcept));
112 graph->edges = SNEPPX_malloc(graph->max_edges * sizeof(SNEPPXSemanticEdge), 64);
113 if (!graph->edges) {
114 SNEPPX_free(graph->concepts, graph->max_concepts * sizeof(SNEPPXSemanticConcept));
115 SNEPPX_free(graph, sizeof(SNEPPXSemanticGraph));
116 return NULL;
117 }
118 memset(graph->edges, 0, graph->max_edges * sizeof(SNEPPXSemanticEdge));
119 return graph;
120}

132int sneppx_semantic_store(SNEPPXSemanticGraph* graph, const char* key,
133 const SNEPPXTensor* embedding) {
134 if (!graph || !key || !embedding) return -1;
135 int idx = find_or_create_concept(graph, key);
136 if (idx < 0) return -1;
137 SNEPPXSemanticConcept* conc = &graph->concepts[idx];
138 strncpy(conc->key, key, SNEPPX_MAX_CONCEPT_KEY_LEN - 1);
139 conc->key[SNEPPX_MAX_CONCEPT_KEY_LEN - 1] = '\0';
140 if (conc->embedding) SNEPPX_tensor_destroy(conc->embedding);
141 conc->embedding = SNEPPX_tensor_copy(embedding);
142 if (!conc->embedding) return -1;
143 conc->ref_count = 0;
144 conc->last_access = ++graph->access_counter;
145 conc->valid = 1;
146 return 0;

149int sneppx_semantic_retrieve(SNEPPXSemanticGraph* graph, const SNEPPXTensor* query,
150 size_t top_k, SNEPPXTensor** output, char*** out_keys) {
151 if (!graph || !query || !output) return -1;
152 if (graph->num_concepts == 0) { *output = NULL; return 0; }
153 size_t n = graph->num_concepts;
154 float* scores = SNEPPX_malloc(n * sizeof(float), 64);
155 size_t* indices = SNEPPX_malloc(n * sizeof(size_t), 64);
156 if (!scores || !indices) {
157 if (scores) SNEPPX_free(scores, n * sizeof(float));
158 if (indices) SNEPPX_free(indices, n * sizeof(size_t));
159 return -1;
160 }
161 size_t valid_count = 0;
162 float query_norm = 0.0f;
163 float* qd = (float*)query->data;
164 size_t qlen = query->size;
165 for (size_t i = 0; i < qlen; i++) query_norm += qd[i] * qd[i];
166 query_norm = sqrtf(query_norm + 1e-10f);
167 for (size_t i = 0; i < graph->num_concepts; i++) {
168 if (!graph->concepts[i].valid || !graph->concepts[i].embedding) continue;
169 float* ed = (float*)graph->concepts[i].embedding->data;
170 size_t elen = graph->concepts[i].embedding->size;
171 size_t min_dim = qlen < elen ? qlen : elen;
172 float dot = 0.0f, norm = 0.0f;
173 for (size_t j = 0; j < min_dim; j++) dot += qd[j] * ed[j];
174 for (size_t j = 0; j < elen; j++) norm += ed[j] * ed[j];
175 norm = sqrtf(norm + 1e-10f);
176 scores[valid_count] = dot / (query_norm * norm + 1e-10f);
177 indices[valid_count] = i;
178 graph->concepts[i].last_access = ++graph->access_counter;
179 graph->concepts[i].ref_count++;
180 valid_count++;
181 }
182 if (valid_count == 0) { *output = NULL; SNEPPX_free(scores, n * sizeof(float)); SNEPPX_free(indices, n * sizeof(size_t)); return 0; }
183 for (size_t i = 0; i < valid_count - 1; i++) {
184 for (size_t j = 0; j < valid_count - 1 - i; j++) {
185 if (scores[j] < scores[j + 1]) {
186 float ts = scores[j]; scores[j] = scores[j + 1]; scores[j + 1] = ts;
187 size_t ti = indices[j]; indices[j] = indices[j + 1]; indices[j + 1] = ti;
188 }
189 }
190 }
191 size_t actual_k = top_k < valid_count ? top_k : valid_count;
192 size_t shape[2] = {actual_k, graph->embedding_dim};
193 SNEPPXTensor* result = SNEPPX_tensor_zeros(shape, 2, SNEPPX_FLOAT32);
194 if (!result) { SNEPPX_free(scores, n * sizeof(float)); SNEPPX_free(indices, n * sizeof(size_t)); return -1; }
195 float* rd = (float*)result->data;
196 if (out_keys) {
197 *out_keys = SNEPPX_malloc(actual_k * sizeof(char*), 64);
198 if (!*out_keys) {
199 SNEPPX_tensor_destroy(result);
200 SNEPPX_free(scores, n * sizeof(float));
201 SNEPPX_free(indices, n * sizeof(size_t));
202 return -1;
203 }
204 }
205 for (size_t i = 0; i < actual_k; i++) {
206 SNEPPXSemanticConcept* c = &graph->concepts[indices[i]];
207 if (c->embedding) {
208 memcpy(rd + i * graph->embedding_dim, c->embedding->data,
209 (graph->embedding_dim < c->embedding->size ? graph->embedding_dim : c->embedding->size) * sizeof(float));
210 }
211 if (out_keys) {
212 (*out_keys)[i] = SNEPPX_malloc(SNEPPX_MAX_CONCEPT_KEY_LEN, 64);
213 if ((*out_keys)[i]) {
214 strncpy((*out_keys)[i], c->key, SNEPPX_MAX_CONCEPT_KEY_LEN - 1);
215 (*out_keys)[i][SNEPPX_MAX_CONCEPT_KEY_LEN - 1] = '\0';
216 }
217 }
218 }
219 *output = result;
220 SNEPPX_free(scores, n * sizeof(float));
221 SNEPPX_free(indices, n * sizeof(size_t));
222 return (int)actual_k;
223}

225int sneppx_semantic_relate(SNEPPXSemanticGraph* graph, const char* from_key,
226 const char* to_key, float weight) {
227 if (!graph || !from_key || !to_key) return -1;
228 int fi = find_concept_idx(graph, from_key);
229 int ti = find_concept_idx(graph, to_key);
230 if (fi < 0 || ti < 0) return -1;
231 if (graph->num_edges >= graph->max_edges) return -1;
232 SNEPPXSemanticEdge* edge = &graph->edges[graph->num_edges++];
233 edge->from_idx = (size_t)fi;
234 edge->to_idx = (size_t)ti;
235 edge->weight = weight;
236 edge->valid = 1;
237 return 0;
238}
