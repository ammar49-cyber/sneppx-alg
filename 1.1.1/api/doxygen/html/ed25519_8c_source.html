106static void fe_mul(field* r, const field* a, const field* b) {
107 uint128 t[9] = {0};
108 for (int i = 0; i < 5; i++)
109 for (int j = 0; j < 5; j++) {
110 int k = i + j;
111 if (k < 9) t[k] = add128(t[k], mul64_64(a->v[i], b->v[j]));

113 /* Fold high terms (×19) back into lower 5 */
114 for (int i = 8; i >= 5; i--) {
115 uint64_t low51 = t[i].lo & 0x7ffffffffffffULL;
116 uint128 high = rshift128(t[i], 51);
117 t[i-5] = add128(t[i-5], mul64_64(19, low51));

249/* Decode Edwards point from 32 bytes: Y in bytes, X parity in top bit of byte[31] */
250static int point_from_bytes(point* p, const uint8_t b[32]) {
251 fe_from_bytes(&p->Y, b);
252 memset(&p->Z, 0, sizeof(field));
253 p->Z.v[0] = 1;
254 field u, v, v3, vxx, check;
255 fe_sq(&u, &p->Y);
256 fe_sub(&u, &u, &p->Z); /* u = Y^2 - 1 */
257 fe_sq(&v, &p->Y); fe_mul(&v, &v, &D); fe_add(&v, &v, &p->Z); /* v = d*Y^2 + 1 */
258 fe_sq(&v3, &v); fe_mul(&v3, &v3, &v); /* v3 = v^3 */
259 fe_sq(&p->X, &v3); fe_mul(&p->X, &p->X, &v);
260 fe_mul(&p->X, &p->X, &u); /* X = u * v^7 */
261 fe_pow22523(&p->X, &p->X); /* X = (u * v^7)^((p-5)/8) */
262 fe_mul(&p->X, &p->X, &v3); fe_mul(&p->X, &p->X, &u); /* X = u * v^3 * (u*v^7)^((p-5)/8) */
263 fe_sq(&vxx, &p->X); fe_mul(&vxx, &vxx, &v);
264 fe_sub(&check, &vxx, &u);
265 uint8_t ck[32]; fe_to_bytes(ck, &check);
266 if (!SNEPPX_ct_is_zero(ck, 32)) {
267 fe_add(&check, &vxx, &u); fe_to_bytes(ck, &check);
268 if (!SNEPPX_ct_is_zero(ck, 32)) { return -1; }
269 fe_mul(&p->X, &p->X, &SQRTM1);
270 }
271 if ((p->X.v[0] & 1) != (b[31] >> 7)) fe_neg(&p->X, &p->X);
272 fe_mul(&p->T, &p->X, &p->Y);
273 return 0;
274}

321static void point_double(point* r, const point* p) {
322 field a, b, c, d, e, f, g, h, t;
323 fe_sub(&a, &p->Y, &p->X); fe_sq(&a, &a); /* A = (Y-X)^2 */
324 fe_add(&b, &p->Y, &p->X); fe_sq(&b, &b); /* B = (Y+X)^2 */
325 fe_sq(&c, &p->T); fe_mul(&c, &c, &D2); /* C = 2*d*T^2 */
326 fe_sq(&d, &p->Z); fe_add(&d, &d, &d); /* D = 2*Z^2 */
327 fe_sub(&e, &b, &a); /* E = B-A = 4*X*Y */
328 fe_sub(&f, &d, &c); /* F = D-C = 2*(Z^2 - d*T^2) */
329 fe_add(&g, &d, &c); /* G = D+C = 2*(Z^2 + d*T^2) */
330 fe_add(&h, &b, &a); /* H = B+A = 2*(Y^2+X^2) */
331 fe_mul(&r->X, &e, &f);
332 fe_mul(&r->Y, &g, &h);
333 fe_mul(&r->T, &e, &h);
334 fe_mul(&r->Z, &f, &g);
335 /* Identity guard: if X == 0 and Y == Z, force output to (0,1,1,0) */
336 { uint8_t xb[32], yzb[32]; fe_to_bytes(xb, &p->X);
337 fe_sub(&t, &p->Y, &p->Z); fe_to_bytes(yzb, &t);
338 uint64_t idm = (uint64_t)(-(int)(SNEPPX_ct_is_zero(xb, 32) && SNEPPX_ct_is_zero(yzb, 32)));
339 for (int i = 0; i < 5; i++) {
340 r->X.v[i] &= ~idm; r->Y.v[i] = (r->Y.v[i] & ~idm) | (idm & 1ULL);
341 r->Z.v[i] = (r->Z.v[i] & ~idm) | (idm & 1ULL); r->T.v[i] &= ~idm;
342 }
343 }
344}

405/* Reduce a 64-byte value mod L to 32 bytes by repeatedly subtracting L << (8*k) */
406static void sc_reduce64(uint8_t r[32], const uint8_t a[64]) {
407 uint8_t t[64]; memcpy(t, a, 64);
408 /* Find highest non-zero byte in t */
409 int hi = 63;
410 while (hi >= 0 && t[hi] == 0) hi--;
411 if (hi < 0) { memset(r, 0, 32); return; }
412 /* Subtract L << (8*k) for k from (hi-31) down to 0 */
413 for (int k = hi - 31; k >= 0; k--) {
414 while (1) {
415 /* Check if t >= L << (8*k) by comparing bytes from hi down to k */
416 int ge = 0;
417 for (int i = hi; i >= k + 32; i--) { if (t[i]) { ge = 1; break; } }
418 if (!ge) {
419 /* Compare the overlapping 32-byte window */
420 int cmp = 0;
421 for (int i = k + 31; i >= k && cmp == 0; i--) {
422 uint8_t tv = t[i];
423 uint8_t lv = L_BYTES[i - k];
424 if (tv > lv) { cmp = 1; } else if (tv < lv) { cmp = -1; }
425 }
426 if (cmp < 0) break; /* t < L << (8*k) */
427 }
428 if (sub_sc_shifted(t, k)) break; /* borrow = overshot */
429 }
430 }
431 /* Final subtraction of L (shift 0) up to 4 times */
432 for (int iter = 0; iter < 4; iter++) {
433 int cmp = 0;
434 for (int i = 31; i >= 0 && cmp == 0; i--) {
435 if (t[i] > L_BYTES[i]) cmp = 1; else if (t[i] < L_BYTES[i]) cmp = -1;
436 }
437 if (cmp < 0) break;
438 sub_sc_shifted(t, 0);
439 }
440 memcpy(r, t, 32);
441}

443/* Multiply two 32-byte scalars -> 64-byte product (schoolbook, 16-bit limbs) */
444static void sc_mul256(uint8_t p[64], const uint8_t a[32], const uint8_t b[32]) {
445 uint32_t t[64] = {0};
446 for (int i = 0; i < 32; i++)
447 for (int j = 0; j < 32; j++)
448 t[i + j] += (uint32_t)a[i] * (uint32_t)b[j];
449 uint64_t carry = 0;
450 for (int i = 0; i < 64; i++) {
451 carry += t[i];
452 p[i] = (uint8_t)(carry & 0xFF);
453 carry >>= 8;
454 }
455}

501int SNEPPX_ed25519_sign(const SNEPPXEd25519Keypair* kp, const uint8_t* message, size_t msg_len, SNEPPXEd25519Signature* sig) {
502 if (!kp || !message || !sig) return -1;
503 init_base_point();
504 uint8_t r_seed[64], r_scalar[32], h_scalar[32], hram[64], product[64], sum[64];
505 /* r = SHA-512(nonce_seed || message) */
506 SNEPPXSHA512Context ctx;
507 SNEPPX_sha512_init(&ctx);
508 SNEPPX_sha512_update(&ctx, kp->private_key + 32, 32);
509 SNEPPX_sha512_update(&ctx, message, msg_len);
510 SNEPPX_sha512_finish(&ctx, r_seed);
511 sc_reduce64(r_scalar, r_seed);
512 /* R = r_scalar * B */
513 point R; point_scalar_mult(&R, r_scalar, 32, &B);
514 point_normalize(&R);
515 fe_to_bytes(sig->data, &R.Y);
516 sig->data[31] |= (uint8_t)((R.X.v[0] & 1) << 7);
517 /* h = SHA-512(R || A || message) */
518 SNEPPX_sha512_init(&ctx);
519 SNEPPX_sha512_update(&ctx, sig->data, 32);
520 SNEPPX_sha512_update(&ctx, kp->public_key, 32);
521 SNEPPX_sha512_update(&ctx, message, msg_len);
522 SNEPPX_sha512_finish(&ctx, hram);
523 /* Reduce full 64-byte challenge hash mod L */
524 sc_reduce64(h_scalar, hram);
525 /* S = (r + h*a) mod L */
526 /* a is the clamped private key (kp->private_key[0:32]) */
527 sc_mul256(product, h_scalar, kp->private_key);
528 /* product + r_scalar as 64-byte sum */
529 uint32_t carry = 0;
530 for (int i = 0; i < 32; i++) {
531 carry = (uint32_t)product[i] + (uint32_t)r_scalar[i] + carry;
532 sum[i] = (uint8_t)(carry & 0xFF);
533 carry >>= 8;
534 }
535 for (int i = 32; i < 64; i++) { carry += product[i]; sum[i] = (uint8_t)(carry & 0xFF); carry >>= 8; }
536 /* Reduce sum mod L to get S */
537 sc_reduce64(sig->data + 32, sum);
538 return 0;
539}

552static int ed25519_verify_core(const uint8_t* public_key, const uint8_t* message, size_t msg_len,
553 const SNEPPXEd25519Signature* sig, int legacy) {
554 if (!public_key || !message || !sig) return -1;
555 init_base_point();
556 uint8_t hram[64];
557 SNEPPXSHA512Context ctx;
558 SNEPPX_sha512_init(&ctx);
559 SNEPPX_sha512_update(&ctx, sig->data, 32);
560 SNEPPX_sha512_update(&ctx, public_key, 32);
561 SNEPPX_sha512_update(&ctx, message, msg_len);
562 if (legacy)
563 SNEPPX_sha512_legacy_finish(&ctx, hram);
564 else
565 SNEPPX_sha512_finish(&ctx, hram);
566 point A; int ret_A = point_from_bytes(&A, public_key);
567 if (ret_A != 0) return -1;
568 point R;
569 int ret_R = point_from_bytes(&R, sig->data);
570 if (ret_R != 0) return -1;
571
572 /* Verify: [S]B = R + [h]A */
573 uint8_t h_scalar[32];
574 sc_reduce64(h_scalar, hram);
575 point S_point;
576 point_scalar_mult(&S_point, sig->data + 32, 32, &B);
577
578 point hA;
579 point_scalar_mult(&hA, h_scalar, 32, &A);
580
581 point_add(&hA, &R, &hA);
582
583 point_normalize(&S_point);
584 point_normalize(&hA);
585
586 /* Compare in affine coordinates (canonical bytes) */
587 uint8_t bx[32], cx[32], by[32], cy[32];
588 fe_to_bytes(bx, &S_point.X); fe_to_bytes(cx, &hA.X);
589 fe_to_bytes(by, &S_point.Y); fe_to_bytes(cy, &hA.Y);
590 return SNEPPX_ct_equal(bx, cx, 32) && SNEPPX_ct_equal(by, cy, 32);
591}

603 return ed25519_verify_core(public_key, message, msg_len, sig, 0);

623int SNEPPX_ed25519_verify_compat(const uint8_t* sig, const uint8_t* m, size_t mlen, const uint8_t* pk, int allow_legacy) {
624 if (!sig || !m || !pk) return -1;
625 SNEPPXEd25519Signature sig_struct;
626 memcpy(sig_struct.data, sig, SNEPPX_ED25519_SIGNATURE_LEN);
627
628 /* RFC-correct verify first */
629 int ret = ed25519_verify_core(pk, m, mlen, &sig_struct, 0);
630 if (ret == 1) return 1;
631 /* On hard error (-1), do not retry with legacy */
632 if (ret == -1 || !allow_legacy) return ret;
633 /* Legacy fallback: re-derive challenge with old H1 digest */
634 return ed25519_verify_core(pk, m, mlen, &sig_struct, 1);
635}
