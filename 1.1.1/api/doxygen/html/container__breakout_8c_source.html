100int SNEPPX_breakout_detect_ns_change(int pid, const char *comm) {

156int SNEPPX_breakout_detect_capability(int pid, uint64_t cap_effective, const char *comm) {
157 int idx = find_process(pid);
158 if (idx < 0) idx = add_process(pid, comm);
159 if (idx < 0) return 0;
160 uint64_t dangerous_caps = 0;

162 dangerous_caps |= (1ULL << 5);
163 dangerous_caps |= (1ULL << 6);
164 dangerous_caps |= (1ULL << 12);
165 dangerous_caps |= (1ULL << 17);
166 dangerous_caps |= (1ULL << 18);
167 dangerous_caps |= (1ULL << 21);
168 dangerous_caps |= (1ULL << 22);

210int SNEPPX_breakout_detect_syscall(int pid, const char *syscall_name, const char *comm) {
211 if (!syscall_name) return 0;
212 for (int i = 0; suspicious_syscalls[i]; i++) {
213 if (strcmp(syscall_name, suspicious_syscalls[i]) == 0) {
214 int idx = find_process(pid);
215 if (idx < 0) idx = add_process(pid, comm);
216 if (idx >= 0) {
217 processes[idx].syscall_count++;
218 processes[idx].anomaly_score += 0.25;
219 processes[idx].suspicious = processes[idx].anomaly_score > 1.0;
220 if (processes[idx].suspicious) total_breakout_attempts++;
221 }
