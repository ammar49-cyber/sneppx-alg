29int sneppx_flex_attn_forward(const float* q, const float* k, const float* v,
30 float* output, const SNEPPX_FlexAttnMask* mask,
31 int batch, int heads, int dim, float scale) {
32 if (!q || !k || !v || !output || !mask) return -1;
33 int seq = mask->seq_len;
34 int bs = mask->block_size;
35 int nb = (seq + bs - 1) / bs;
36 cublasHandle_t h; cublasCreate(&h);
37 float one = 1.0f, zero = 0.0f;
38 float* scores; cudaMalloc(&scores, bs * bs * sizeof(float));
39 for (int b = 0; b < batch; b++) {
40 for (int hd = 0; hd < heads; hd++) {
41 float* out_head = output + ((b * heads + hd) * seq) * dim;
42 for (int qi = 0; qi < nb; qi++) {
43 for (int kj = 0; kj < nb; kj++) {
44 if (mask->block_mask[qi * nb + kj]) {
45 int q_off = (b * heads + hd) * seq * dim + qi * bs * dim;
46 int k_off = (b * heads + hd) * seq * dim + kj * bs * dim;
47 cublasSgemm(h, CUBLAS_OP_T, CUBLAS_OP_N, bs, bs, dim,
48 &scale, k + k_off, dim, q + q_off, dim,
49 &zero, scores, bs);
50 // Apply mask modulation function
51 if (mask->modulate_fn) {
52 for (int i = 0; i < bs && qi * bs + i < seq; i++) {
53 int q_idx = qi * bs + i;
54 for (int j = 0; j < bs && kj * bs + j < seq; j++) {
55 int kv_idx = kj * bs + j;
56 float mod = mask->modulate_fn(q_idx, kv_idx, hd, mask->mod_ctx);
57 scores[i * bs + j] += mod;
58 }
59 }
60 }
61 // Softmax and V accumulation
62 float* vs = scores;
63 for (int i = 0; i < bs; i++) {
64 float maxv = -1e30f, sum = 0.0f;
65 for (int j = 0; j < bs; j++) {
66 if (vs[i * bs + j] > maxv) maxv = vs[i * bs + j];
67 }
68 for (int j = 0; j < bs; j++) {
69 vs[i * bs + j] = expf(vs[i * bs + j] - maxv);
70 sum += vs[i * bs + j];
71 }
72 if (sum > 0) for (int j = 0; j < bs; j++) vs[i * bs + j] /= sum;
73 }
74 int v_off = (b * heads + hd) * seq * dim + kj * bs * dim;
75 cublasSgemm(h, CUBLAS_OP_N, CUBLAS_OP_N, dim, bs, bs,
76 &one, v + v_off, dim, scores, bs,
77 &one, out_head + qi * bs * dim, dim);
78 }
79 }
80 }
81 }
82 }
83 cublasDestroy(h);
84 cudaFree(scores);
85 return 0;
86}

92int sneppx_multimodal_cross_attn(const float* text_q, const float* vision_k,
93 const float* vision_v, float* output,
94 int text_len, int vision_len, int heads, int dim) {
95 if (!text_q || !vision_k || !vision_v || !output) return -1;
96 cublasHandle_t h; cublasCreate(&h);
97 float scale = 1.0f / sqrtf(dim / heads);
98 float one = 1.0f, zero = 0.0f;
99 int head_dim = dim / heads;
100 float* scores; cudaMalloc(&scores, heads * text_len * vision_len * sizeof(float));
101 for (int hd = 0; hd < heads; hd++) {
102 cublasSgemm(h, CUBLAS_OP_T, CUBLAS_OP_N, vision_len, text_len, head_dim,
103 &scale, vision_k + hd * vision_len * head_dim, head_dim,
104 text_q + hd * text_len * head_dim, head_dim,
105 &zero, scores + hd * text_len * vision_len, vision_len);
106 }
107 for (int b = 0; b < heads; b++) {
108 for (int t = 0; t < text_len; t++) {
109 float maxv = -1e30f, sum = 0.0f;
110 for (int v = 0; v < vision_len; v++) {
111 float s = scores[(b * text_len + t) * vision_len + v];
112 if (s > maxv) maxv = s;
113 }
114 for (int v = 0; v < vision_len; v++) {
115 float s = scores[(b * text_len + t) * vision_len + v];
116 s = expf(s - maxv); sum += s;
117 scores[(b * text_len + t) * vision_len + v] = s;
118 }
119 if (sum > 0) for (int v = 0; v < vision_len; v++)
120 scores[(b * text_len + t) * vision_len + v] /= sum;
121 }
122 }
123 for (int hd = 0; hd < heads; hd++) {
124 cublasSgemm(h, CUBLAS_OP_N, CUBLAS_OP_N, head_dim, text_len, vision_len,
125 &one, vision_v + hd * vision_len * head_dim, head_dim,
126 scores + hd * text_len * vision_len, vision_len,
127 &zero, output + hd * text_len * head_dim, head_dim);
128 }
129 cublasDestroy(h);
130 cudaFree(scores);
131 return 0;
132}

163// ============================================================================
164// Gated Activations: SwiGLU = SiLU(x) * gate, GeGLU = GELU(x) * gate, ReGLU = ReLU(x) * gate
165// ============================================================================
166
167int sneppx_gated_activation_forward(const float* x, const float* gate,
168 float* output, SNEPPX_GatedActType act,
169 int numel) {
170 if (!x || !gate || !output) return -1;
171 for (int i = 0; i < numel; i++) {
172 float xv = x[i], gv = gate[i];
173 float act_val;
174 switch (act) {
175 case SNEPPX_GATED_SWIGLU:
176 act_val = xv / (1.0f + expf(-xv)); // SiLU
177 break;
178 case SNEPPX_GATED_GEGLU: {
179 float c = 0.7978845608028654f;
180 float x3 = xv * xv * xv;
181 act_val = 0.5f * xv * (1.0f + tanhf(c * (xv + 0.044715f * x3)));
182 break;
183 }
184 case SNEPPX_GATED_REGLU:
185 act_val = fmaxf(0.0f, xv);
186 break;
187 default:
188 act_val = xv;
189 }
190 output[i] = act_val * gv;
191 }
192 return 0;
193}

199int sneppx_yarn_rope_forward(const float* x, float* output,
200 const float* cos_cache, const float* sin_cache,
201 int batch, int seq, int heads, int dim,
202 float scale, float alpha, float beta) {
203 if (!x || !output || !cos_cache || !sin_cache) return -1;
204 int half_dim = dim / 2;
205 for (int b = 0; b < batch; b++) {
206 for (int s = 0; s < seq; s++) {
207 for (int h = 0; h < heads; h++) {
208 int base = ((b * heads + h) * seq + s) * dim;
209 for (int d = 0; d < half_dim; d++) {
210 float x0 = x[base + 2 * d];
211 float x1 = x[base + 2 * d + 1];
212 float cos_v = cos_cache[s * half_dim + d];
213 float sin_v = sin_cache[s * half_dim + d];
214 // YaRN: interpolation + NTK-aware scaling
215 float t = (float)s / (float)seq;
216 float ram = (1.0f - t) * alpha + t * beta;
217 float cos_scaled = cosf(ram * acosf(cos_v));
218 float sin_scaled = sinf(ram * asinf(sin_v));
219 output[base + 2 * d] = x0 * cos_scaled - x1 * sin_scaled;
220 output[base + 2 * d + 1] = x0 * sin_scaled + x1 * cos_scaled;
221 }
222 }
223 }
224 }
225 return 0;
226}
