77int SNEPPX_act_backward(const float* x, const float* grad_out, float* grad_in, SNEPPXActivationType act, size_t n) {
78 if (!x || !grad_out || !grad_in || n == 0) return -1;
79 for (size_t i = 0; i < n; i++) {
80 float xv = x[i], g = grad_out[i];
81 switch (act) {
82 case SNEPPX_ACT_RELU:
83 grad_in[i] = xv > 0.0f ? g : 0.0f;
84 break;
85 case SNEPPX_ACT_LEAKY_RELU:
86 grad_in[i] = xv > 0.0f ? g : g * 0.01f;
87 break;
88 case SNEPPX_ACT_GELU_TANH: {
89 float c = 0.7978845608028654f;
90 float x3 = xv * xv * xv;
91 float t = tanhf(c * (xv + 0.044715f * x3));
92 float sech2 = 1.0f - t * t;
93 float dg = 0.5f * (1.0f + t) + 0.5f * xv * c * (1.0f + 3.0f * 0.044715f * xv * xv) * sech2;
94 grad_in[i] = g * dg;
95 break;
96 }
97 case SNEPPX_ACT_SILU: {
98 float s = 1.0f / (1.0f + expf(-xv));
99 grad_in[i] = g * s * (1.0f + xv * (1.0f - s));
100 break;
101 }
102 case SNEPPX_ACT_SIGMOID: {
103 float s = 1.0f / (1.0f + expf(-xv));
104 grad_in[i] = g * s * (1.0f - s);
105 break;
106 }
107 case SNEPPX_ACT_TANH: {
108 float t = tanhf(xv);
109 grad_in[i] = g * (1.0f - t * t);
110 break;
111 }
112 default:
113 grad_in[i] = g;
114 }
115 }
116 return 0;
117}

127int SNEPPX_act_softmax_stable(const float* x, float* output, size_t n) {
128 if (!x || !output || n == 0) return -1;
129 float maxv = -FLT_MAX;
130 for (size_t i = 0; i < n; i++) if (x[i] > maxv) maxv = x[i];
131 float sum = 0.0f;
132 for (size_t i = 0; i < n; i++) { float e = expf(x[i] - maxv); output[i] = e; sum += e; }
133 if (sum > 0.0f) for (size_t i = 0; i < n; i++) output[i] /= sum;
134 return 0;
135}
