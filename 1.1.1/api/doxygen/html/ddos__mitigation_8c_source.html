93int SNEPPX_ddos_detect_syn_flood(uint32_t src_ip, uint32_t dst_ip) {
94 int idx = find_source(src_ip);
95 if (idx < 0) idx = add_source(src_ip);
96 if (idx < 0) return 0;
97 sources[idx].syn_count++;
98 sources[idx].total_count++;
99 time_t now = time(NULL);
100 if (now - sources[idx].window_start > DDOS_WINDOW_SEC) {
101 sources[idx].syn_count = 0;
102 sources[idx].total_count = 0;
103 sources[idx].window_start = now;
104 }
105 if (sources[idx].blocked && now < sources[idx].block_until) return 1;
106 if (sources[idx].blocked && now >= sources[idx].block_until) {
107 sources[idx].blocked = 0;
108 sources[idx].syn_count = 0;
109 }
110 if (sources[idx].syn_count > active_config.syn_flood_threshold) {
111 sources[idx].blocked = 1;
112 sources[idx].block_until = now + active_config.block_duration;
113 total_blocks_issued++;
114 total_sources_blocked++;
115 return 1;
116 }
117 double rate = (double)sources[idx].total_count / (double)(now - sources[idx].window_start + 1);
118 sources[idx].request_rate = rate;
119 if (rate > active_config.conn_rate_threshold && active_config.enable_adaptive) {
120 sources[idx].blocked = 1;
121 sources[idx].block_until = now + active_config.block_duration;
122 total_blocks_issued++;
123 total_sources_blocked++;
124 return 1;
125 }
126 return 0;
127}

137int SNEPPX_ddos_detect_app_layer(uint32_t src_ip, const uint8_t *payload, size_t len) {
138 if (!active_config.enable_app_layer) return 0;
139 int idx = find_source(src_ip);
140 if (idx < 0) idx = add_source(src_ip);
141 if (idx < 0) return 0;
142 time_t now = time(NULL);
143 if (sources[idx].blocked && now < sources[idx].block_until) return 1;
144 if (len > 0) {
145 int slow_attack = 0;
146 for (size_t i = 0; i < len && i < 64; i++) {
147 if (payload[i] == 0) slow_attack++;
148 }
149 if (slow_attack > (int)(len > 64 ? 64 : len) / 2) {
150 sources[idx].blocked = 1;
151 sources[idx].block_until = now + active_config.block_duration;
152 total_blocks_issued++;
153 return 1;
154 }
155 }
156 return 0;
157}

167uint32_t SNEPPX_ddos_generate_syn_cookie(uint32_t src_ip, uint32_t dst_ip, uint16_t src_port) {
168 if (!active_config.enable_syn_cookies) return 0;
169 time_t now = time(NULL);
170 uint32_t cookie = src_ip ^ dst_ip ^ ((uint32_t)src_port << 16);
171 cookie ^= (uint32_t)(now / 60) * 2654435761u;
172 cookie ^= (uint32_t)(now % 60) * 2246822519u;
173 return cookie & 0x00FFFFFF;
174}
