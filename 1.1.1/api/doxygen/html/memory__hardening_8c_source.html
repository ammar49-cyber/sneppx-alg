649int SNEPPX_mem_scrub(void* ptr, size_t size) {
650 if (!ptr || size == 0) return -1;
651 uint8_t* p = (uint8_t*)ptr;
652 size_t half = size / 2;
653 size_t quarter = size / 4;
654 size_t three_quarter = half + quarter;
655 size_t pattern_idx = (size_t)g_scrub_phase % 4;
656 volatile uint8_t sink = 0;
657 for (size_t pass = 0; pass < SNEPPX_SCRUB_PASSES; pass++) {
658 uint8_t current_pat = g_scrub_patterns[(pattern_idx + pass) % 4];
659 for (size_t i = 0; i < size; i++) {
660 p[i] ^= current_pat;
661 p[i] = (uint8_t)((p[i] << 3) | (p[i] >> 5));
662 p[i] ^= (uint8_t)(pass * 0x33);
663 sink ^= p[i];
664 }
665 if (pass == 0) {
666 for (size_t i = 0; i < size; i++) p[i] = 0xFD;
667 }
668 if (pass == 1) {
669 for (size_t i = 0; i < half; i++) p[i] = (uint8_t)(p[i] ^ 0xDF);
670 for (size_t i = half; i < size; i++) p[i] = (uint8_t)(p[i] ^ 0xBA);
671 }
672 if (pass == 2) {
673 for (size_t i = 0; i < quarter; i++) p[i] = (uint8_t)(p[i] * 0xAB);
674 for (size_t i = quarter; i < three_quarter; i++) p[i] = (uint8_t)(p[i] ^ 0xCD);
675 for (size_t i = three_quarter; i < size; i++) p[i] = (uint8_t)(p[i] * 0xEF);
676 }
677 if (pass == 3) {
678 for (size_t i = 0; i < size; i++) p[i] = 0x00;
679 }
680 }
681 (void)sink;
682 g_scrub_phase = (g_scrub_phase + 1) % 1024;
683 return 0;
684}

866int SNEPPX_mem_pool_init(SNEPPXMemPool* pool, size_t block_size, int blocks) {
867 if (!pool || block_size == 0 || blocks <= 0 || blocks > SNEPPX_MEM_POOL_MAX_BLOCKS) return -1;
868 if (block_size < sizeof(uint64_t)) block_size = sizeof(uint64_t);
869 pool->block_size = block_size;
870 pool->total_blocks = blocks;
871 pool->free_blocks = blocks;
872 size_t bitmap_size = (size_t)((blocks + SNEPPX_BITMAP_BITS_PER_BYTE - 1) / SNEPPX_BITMAP_BITS_PER_BYTE);
873 pool->bitmap = (uint8_t*)calloc(bitmap_size, 1);
874 if (!pool->bitmap) return -1;
875 pool->memory = (uint8_t*)calloc((size_t)blocks, block_size);
876 if (!pool->memory) {
877 free(pool->bitmap);
878 pool->bitmap = NULL;
879 return -1;
880 }
881 memset(pool->bitmap, 0, bitmap_size);
882 memset(pool->memory, 0xAB, (size_t)blocks * block_size);
883 return 0;
884}
885
891void* SNEPPX_mem_pool_alloc(SNEPPXMemPool* pool) {
892 if (!pool || !pool->bitmap || !pool->memory) return NULL;
893 if (pool->free_blocks <= 0) return NULL;
894 for (int i = 0; i < pool->total_blocks; i++) {
895 int byte_idx = i / SNEPPX_BITMAP_BITS_PER_BYTE;
896 int bit_idx = i % SNEPPX_BITMAP_BITS_PER_BYTE;
897 if (!(pool->bitmap[byte_idx] & (uint8_t)(1 << bit_idx))) {
898 pool->bitmap[byte_idx] |= (uint8_t)(1 << bit_idx);
899 pool->free_blocks--;
900 void* block = (uint8_t*)pool->memory + (size_t)i * pool->block_size;
901 memset(block, 0, pool->block_size);
902 return block;
903 }
904 }
905 return NULL;
906}
907
915int SNEPPX_mem_pool_free(SNEPPXMemPool* pool, void* ptr) {
916 if (!pool || !pool->bitmap || !pool->memory || !ptr) return -1;
917 uintptr_t mem_start = (uintptr_t)pool->memory;
918 uintptr_t mem_end = mem_start + (uintptr_t)pool->total_blocks * pool->block_size;
919 uintptr_t p = (uintptr_t)ptr;
920 if (p < mem_start || p >= mem_end) return -1;
921 uintptr_t offset = p - mem_start;
922 if (offset % pool->block_size != 0) return -1;
923 int idx = (int)(offset / pool->block_size);
924 if (idx < 0 || idx >= pool->total_blocks) return -1;
925 int byte_idx = idx / SNEPPX_BITMAP_BITS_PER_BYTE;
926 int bit_idx = idx % SNEPPX_BITMAP_BITS_PER_BYTE;
927 if (!(pool->bitmap[byte_idx] & (uint8_t)(1 << bit_idx))) return -1;
928 memset(ptr, SNEPPX_POOL_SCRUB_BYTE, pool->block_size);
929 pool->bitmap[byte_idx] &= (uint8_t)~(1 << bit_idx);
930 pool->free_blocks++;
931 return 0;
932}

1029int SNEPPX_shadow_stack_validate(SNEPPXShadowStack* ss) {
1030 if (!ss) return -1;
1031 if (ss->overflow_detected) return -1;
1032 if (ss->sp < -1 || ss->sp >= SNEPPX_SHADOW_STACK_DEPTH) return -1;
1033 if (ss->sp < 0) return 0;
1034 uint64_t check = 0;
1035 for (int i = 0; i <= ss->sp; i++) {
1036 if (ss->stack[i] == 0 || ss->stack[i] == (uintptr_t)-1) return -1;
1037 if (i > 0) {
1038 if (ss->stack[i] == ss->stack[i - 1] && ss->stack[i] == 0) return -1;
1039 }
1040 check ^= (uint64_t)ss->stack[i];
1041 check = (check << 7) | (check >> 57);
1042 }
1043 uint64_t expected = 0;
1044 for (int i = 0; i <= ss->sp; i++) {
1045 expected ^= (uint64_t)ss->stack[i];
1046 expected = (expected << 7) | (expected >> 57);
1047 }
1048 if (check != expected) return -1;
1049 for (int i = ss->sp + 1; i < SNEPPX_SHADOW_STACK_DEPTH; i++) {
1050 if (ss->stack[i] != 0) return -1;
1051 }
1052 return 0;
1053}
