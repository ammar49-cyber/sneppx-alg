89 int SNEPPX_bn_mul_word(SNEPPXBigNum* r, SNEPPX_BN_WORD w) {
90 if (!r) return -1;
91 if (w==0||(r->used==1&&r->words[0]==0)) { SNEPPX_bn_zero(r); return 0; }
92 if (w==1) return 0;
93 uint64_t carry=0;
94 for (int i=0;i<r->used;i++) {
95 uint128_t prod=(uint128_t)r->words[i]*w+carry;
96 r->words[i]=(uint64_t)prod;
97 carry=(uint64_t)(prod>>64);
98 }
99 if (carry) {
100 if (r->used>=SNEPPX_BN_MAX_WORDS) return -1;
101 r->words[r->used++]=carry;
102 }
103 return 0;
104 }

108 if (SNEPPX_bn_is_zero(a)||SNEPPX_bn_is_zero(b)) { SNEPPX_bn_zero(r); return 0; }

148 int SNEPPX_bn_div(SNEPPXBigNum* q, SNEPPXBigNum* rem, const SNEPPXBigNum* a, const SNEPPXBigNum* b) {
149 if (!q||!rem||!a||!b) return -1;
150 if (SNEPPX_bn_is_zero(b)) return -1;
151 if (SNEPPX_bn_cmp(a,b)<0) { SNEPPX_bn_zero(q); SNEPPX_bn_copy(rem,a); return 0; }
152 if (b->used==1) {
153 SNEPPXBigNum tmp; SNEPPX_bn_init(&tmp); SNEPPX_bn_copy(&tmp,a);
154 uint64_t rv=SNEPPX_bn_div_word(&tmp,b->words[0]);
155 SNEPPX_bn_copy(q,&tmp); SNEPPX_bn_set_word(rem,rv);
156 return 0;
157 }
158 int n=b->used, m=a->used-n;
159 if (m<0) { SNEPPX_bn_zero(q); SNEPPX_bn_copy(rem,a); return 0; }
160 unsigned shift=clz64(b->words[n-1]);
161 SNEPPXBigNum u,v; SNEPPX_bn_init(&u); SNEPPX_bn_init(&v);
162 SNEPPX_bn_copy(&v,b);
163 if (shift) {
164 uint64_t carry=0;
165 for (int i=0;i<a->used;i++) {
166 uint128_t w=((uint128_t)a->words[i]<<shift)|carry;
167 u.words[i]=(uint64_t)w; carry=(uint64_t)(w>>64);
168 }
169 if (carry) u.words[a->used]=carry;
170 u.used=a->used+(carry?1:0);
171 carry=0;
172 for (int i=0;i<n;i++) {
173 uint128_t w=((uint128_t)v.words[i]<<shift)|carry;
174 v.words[i]=(uint64_t)w; carry=(uint64_t)(w>>64);
175 }
176 } else {
177 SNEPPX_bn_copy(&u,a);
178 }
179 SNEPPXBigNum qu; SNEPPX_bn_init(&qu); qu.used=m+1;
180 for (int j=m;j>=0;j--) {
181 uint64_t ujn=(j+n<u.used)?u.words[j+n]:0;
182 uint64_t ujn1=(j+n-1<u.used)?u.words[j+n-1]:0;
183 uint64_t ujn2=(j+n-2<u.used)?u.words[j+n-2]:0;
184 uint128_t num=((uint128_t)ujn<<64)|ujn1;
185 uint64_t qhat=(uint64_t)(num/v.words[n-1]);
186 uint64_t rhat=(uint64_t)(num%v.words[n-1]);
187 if (qhat==0) { qu.words[j]=0; continue; }
188 if (qhat>UINT64_MAX) qhat=UINT64_MAX;
189 while (qhat&&(uint128_t)qhat*v.words[n-2]>(((uint128_t)rhat<<64)|ujn2)) {
190 qhat--; rhat+=v.words[n-1];
191 if (rhat<v.words[n-1]) break;
192 }
193 int64_t borrow=0;
194 for (int i=0;i<n;i++) {
195 uint128_t p=(uint128_t)qhat*v.words[i];
196 uint64_t pl=(uint64_t)p;
197 uint64_t ph=(uint64_t)(p>>64);
198 uint64_t diff=u.words[j+i]-pl-borrow;
199 int64_t nb=(borrow&&u.words[j+i]==0&&pl==0)?1:((u.words[j+i]-borrow<pl)?1:0);
200 if (diff>u.words[j+i]&&!borrow) nb=1;
201 u.words[j+i]=diff;
202 borrow=ph+nb;
203 }
204 uint64_t ujn_adj=(j+n<u.used)?u.words[j+n]:0;
205 int64_t fin=(int64_t)ujn_adj-borrow;
206 if (fin<0) {
207 qhat--;
208 uint64_t carry=0;
209 for (int i=0;i<n;i++) {
210 uint128_t s=(uint128_t)u.words[j+i]+v.words[i]+carry;
211 u.words[j+i]=(uint64_t)s; carry=(uint64_t)(s>>64);
212 }
213 u.words[j+n]+=carry;
214 }
215 qu.words[j]=qhat;
216 }
217 while (qu.used>0&&!qu.words[qu.used-1]) qu.used--;
218 if (qu.used==0) qu.used=1;
219 SNEPPX_bn_copy(q,&qu); q->sign=0;
220 uint64_t carry=0;
221 for (int i=u.used-1;i>=0;i--) {
222 uint128_t w=((uint128_t)carry<<64)|u.words[i];
223 u.words[i]=(uint64_t)(w>>shift); carry=(uint64_t)(w&((((uint128_t)1)<<shift)-1));
224 }
225 u.used=n; while (u.used>0&&!u.words[u.used-1]) u.used--;
226 if (u.used==0) u.used=1;
227 SNEPPX_bn_copy(rem,&u); rem->sign=0;
228 return 0;
229 }

301int SNEPPX_bn_from_hex(SNEPPXBigNum* bn, const char* hex) {
302 if (!bn||!hex) return -1;
303 size_t len=strlen(hex);
304 SNEPPX_bn_zero(bn); bn->used=1;
305 for (size_t i=0;i<len;i++) {
306 char c=hex[i];
307 int v;
308 if (c>='0'&&c<='9') v=c-'0';
309 else if (c>='a'&&c<='f') v=10+c-'a';
310 else if (c>='A'&&c<='F') v=10+c-'A';

312 if (SNEPPX_bn_mul_word(bn,16)!=0) return -1;
313 uint64_t ov=bn->words[0];
314 bn->words[0]+=v;
315 if (bn->words[0]<ov) {
316 int idx=1;
317 while (idx<bn->used) { bn->words[idx]++; if (bn->words[idx]) break; idx++; }
318 if (idx>=bn->used) {
319 if (bn->used>=SNEPPX_BN_MAX_WORDS) return -1;
320 bn->words[bn->used++]=1;
321 }
322 }
323 }

413int SNEPPX_bn_add(SNEPPXBigNum* r, const SNEPPXBigNum* a, const SNEPPXBigNum* b) {
414 if (!r||!a||!b) return -1;
415 int max=a->used>b->used?a->used:b->used;
416 uint64_t carry=0;
417 for (int i=0;i<max;i++) {
418 uint64_t av=i<a->used?a->words[i]:0;
419 uint64_t bv=i<b->used?b->words[i]:0;
420 uint64_t s=av+carry;
421 uint64_t c=(s<av)?1:0;
422 s+=bv;
423 if (s<bv) c=1;
424 r->words[i]=s;
425 carry=c;
426 }
427 if (carry) {
428 if (max>=SNEPPX_BN_MAX_WORDS) return -1;
429 r->words[max]=1;
430 max++;
431 }
432 r->used=max;

447 if (cmp==0) { SNEPPX_bn_zero(r); return 0; }
448 const SNEPPXBigNum* bg=(cmp>0)?a:b;
449 const SNEPPXBigNum* sm=(cmp>0)?b:a;
450 uint64_t borrow=0;
451 for (int i=0;i<bg->used;i++) {
452 uint64_t av=bg->words[i];

454 uint64_t t=av-borrow;

458 r->words[i]=d;

463 if (r->used==0) { r->used=1; r->words[0]=0; }
464 r->sign=0;

569int SNEPPX_bn_is_prime(const SNEPPXBigNum* bn) {
570 if (!bn) return -1;
571 if (SNEPPX_bn_cmp_word(bn,2)<0) return 0;
572 if (bn->used==1&&bn->words[0]%2==0) return SNEPPX_bn_cmp_word(bn,2)==0;
573 SNEPPXBigNum n,d,a,x,one,tmp; SNEPPX_bn_init(&n); SNEPPX_bn_init(&d);
574 SNEPPX_bn_init(&a); SNEPPX_bn_init(&x); SNEPPX_bn_init(&one); SNEPPX_bn_init(&tmp);
575 SNEPPX_bn_copy(&n,bn); SNEPPX_bn_set_word(&one,1);
576 SNEPPX_bn_sub(&d,&n,&one);
577 int s=0;
578 while (!SNEPPX_bn_is_zero(&d)&&!(d.words[0]&1)) {
579 SNEPPX_bn_div_word(&d,2); s++;
580 }
581 static const SNEPPX_BN_WORD bases[]={2,3,5,7,11,13,17};
582 int nb=sizeof(bases)/sizeof(bases[0]);
583 for (int i=0;i<nb;i++) {
584 if (SNEPPX_bn_cmp_word(&n,bases[i])<=0) continue;
