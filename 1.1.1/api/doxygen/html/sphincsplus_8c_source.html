8/*
9 * SNEPPX - SPHINCS+ Stateless Hash-Based Signatures
10 *
11 * WHAT
12 * Deterministic stateless hash-based signature scheme in the SPHINCS+
13 * family (FIPS 205 style). Hypertree of Merkle trees over WOTS+ one-time
14 * signatures, message digest committed via a FORS forest.
15 *
16 * CONCEPT
17 * All private randomness is derived deterministically via a PRF
18 * H(seed || domain-tag || counters) keyed by the secret seed, so
19 * signatures are reproducible and verifiable within the scheme.
20 *
21 * ROLE
22 * Layer S0 post-quantum crypto for long-lived signatures (firmware
23 * manifests, release signing).
24 *
25 * REFERENCES
26 * FIPS 205 (SLH-DSA / SPHINCS+), NIST PQC Round 3 finalist.
27 *
28 * STRUCTURE
29 * Keys: pk = pub_seed(16) || top-root(16); sk = sk_seed(16) || sk_prf(16) || pub_seed(16)
30 * Hash: H = first 16 bytes of SHA-256
31 * PRF: sk_x = H(seed || tag || a(8 LE) || b(8 LE))
32 * WOTS: w=16, len=67 (64 message nibbles + 3 checksum), chain x_{j+1} = H(x_j || j)
33 * FORS: 30 trees x 32 leaves (5-bit indices), roots -> H -> commit
34 * Trees: D=10 layers x height-6 Merkle trees over H(WOTS pk) leaves
35 * Sig: R(16) | FORS sig(2880) | FORS pk(16) | HT(11680) = 14592 bytes
36 */
37

180static void spx_fors_sign(uint8_t *sig, uint8_t *pk, const uint8_t *md, const uint8_t *sk_seed) {

182 for (int t = 0; t < SPX_FORS_TREES; t++) {
183 uint32_t idx = spx_fors_index(md, t);
184 uint8_t tree[SPX_FORS_INDICES * SPX_N];
185 for (int i = 0; i < SPX_FORS_INDICES; i++)
186 spx_prf(tree + (size_t)i * SPX_N, sk_seed, SPX_TAG_FORS, (uint64_t)t, (uint64_t)i);
187 uint8_t *tsig = sig + (size_t)t * ((SPX_FORS_HEIGHT + 1) * SPX_N);
188 memcpy(tsig, tree + (size_t)idx * SPX_N, SPX_N);
189 uint32_t pos = idx;

191 uint32_t sibling = pos ^ 1;
192 memcpy(tsig + (size_t)(h + 1) * SPX_N, tree + (size_t)sibling * SPX_N, SPX_N);

194 for (int j = 0; j < pairs; j++)

196 pos >>= 1;
197 }
198 memcpy(roots + (size_t)t * SPX_N, tree, SPX_N);
199 }

227static void spx_ht_sign(uint8_t *sig, const uint8_t *sk_seed, uint32_t idx0, const uint8_t *fors_pk) {
228 size_t pos = 0;
229 uint8_t ctx[SPX_N];
230 memcpy(ctx, fors_pk, SPX_N);
231 for (int d = 0; d < SPX_D; d++) {
232 uint64_t leaf = (d == 0) ? (uint64_t)idx0 : 0;
233 spx_wots_sign(sig + pos, ctx, sk_seed, d, leaf);
234 pos += SPX_WOTS_SIG_BYTES;
235 for (int h = 0; h < SPX_TREE_HEIGHT; h++) {
236 uint64_t sibling = leaf ^ UINT64_C(1) << h;
237 uint64_t offset = (sibling >> h) << h;
238 spx_subtree_root(sig + pos, sk_seed, d, h, (int)offset);
239 pos += SPX_N;
240 }
241 if (d + 1 < SPX_D)
242 spx_treehash(ctx, sk_seed, d);
243 }
