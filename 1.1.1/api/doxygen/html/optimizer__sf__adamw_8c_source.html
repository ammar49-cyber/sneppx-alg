59int SNEPPX_sf_adamw_step(SNEPPXSFRAdamWState* state, float* params, const float* grads) {
60 if (!state || !params || !grads) return -1;
61 if (!state->z || !state->m || !state->v) return -1;
62 size_t sz = state->size;
63 float b1 = state->beta1, b2 = state->beta2;
64 float eps = state->eps, lr = state->lr, wd = state->weight_decay;
65 float* z = state->z, *m = state->m, *v = state->v;
66
67 state->t++;
68
69 for (size_t i = 0; i < sz; i++) {
70 float g = grads[i];
71 m[i] = b1 * m[i] + (1.0f - b1) * g;
72 v[i] = b2 * v[i] + (1.0f - b2) * g * g;
73 z[i] = (1.0f - b1) * params[i] + b1 * z[i];
74 float step = m[i] / (sqrtf(v[i]) + eps);
75 params[i] = z[i] - lr * step - lr * wd * z[i];
76 }
