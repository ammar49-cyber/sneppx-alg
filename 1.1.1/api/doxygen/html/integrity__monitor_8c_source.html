294static uint32_t crc32_c(const void* data, size_t len) {
295 const unsigned char* buf = (const unsigned char*)data;
296 uint32_t crc = 0xFFFFFFFF;
297 for (size_t i = 0; i < len; i++) {
298 crc ^= buf[i];
299 for (int j = 0; j < 8; j++)
300 crc = (crc >> 1) ^ (0xEDB88320 & -(crc & 1));
301 }
302 return ~crc;

423int SNEPPX_monitor_timing_check(uint64_t elapsed_us) {
424 if (g_timing_samples < 5) {
425 g_timing_samples++;
426 return 0;
427 }
428 if (g_sliding_window_count < SNEPPX_MONITOR_SLIDING_WINDOW_SIZE) {
429 g_sliding_window[g_sliding_window_count++] = (double)elapsed_us;
430 } else {
431 g_sliding_window[g_sliding_window_index] = (double)elapsed_us;
432 g_sliding_window_index = (g_sliding_window_index + 1) % SNEPPX_MONITOR_SLIDING_WINDOW_SIZE;
433 }
434 double sum = 0;
435 int n = g_sliding_window_count > 0 ? g_sliding_window_count : 1;
436 for (int i = 0; i < g_sliding_window_count; i++) sum += g_sliding_window[i];
437 double avg = sum / n;
438 double variance = 0;
439 for (int i = 0; i < g_sliding_window_count; i++) {
440 double diff = g_sliding_window[i] - avg;
441 variance += diff * diff;
442 }
443 double effective_stddev = (n > 1) ? sqrt(variance / (n - 1)) : g_timing_stddev;
444 if (effective_stddev < 0.001) effective_stddev = g_timing_stddev;
445 double z = (fabs((double)elapsed_us - g_timing_baseline)) / effective_stddev;
446 if (z > 3.0) {
447 char desc[256];
448 snprintf(desc, sizeof(desc), "Timing anomaly: z-score %.2f (elapsed %llu us, baseline %.0f us)",
449 z, (unsigned long long)elapsed_us, g_timing_baseline);
450 fire_event(SNEPPX_MONITOR_EVENT_CANARY_TRIGGERED, desc, 0, 0);
451 return 1;
452 }
453 return 0;
454}

694 if (elapsed_heartbeat > g_heartbeat_interval_ms) {

711

713 if (!g_regions[i].active) continue;

716 (now_us - g_region_freq[i].last_check_time) : 0;
717 if (interval > 0) {
718 if (g_region_freq[i].min_interval_us == 0 || interval < g_region_freq[i].min_interval_us)
719 g_region_freq[i].min_interval_us = interval;
720 if (interval > g_region_freq[i].max_interval_us)
721 g_region_freq[i].max_interval_us = interval;
722 }

724 if (interval > 0 && interval < 100 && g_region_freq[i].check_count > 5) {
725 char desc[256];

729 violations++;
730 }
731 if (g_region_freq[i].check_count > 50 && g_last_verify_time > 0) {
732 uint64_t total_span = now_us - g_last_verify_time;
733 if (total_span > 0) {
734 double checks_per_sec = (double)g_region_freq[i].check_count / ((double)total_span / 1000000.0);
735 if (checks_per_sec < 0.01 && g_region_freq[i].check_count > 0) {
736 char desc[256];
737 snprintf(desc, sizeof(desc), "Evasion suspect: region '%s' checked only %.2f/sec (%d checks)",
738 g_regions[i].name, checks_per_sec, g_region_freq[i].check_count);
739 fire_event(SNEPPX_MONITOR_EVENT_FREQ_ANOMALY, desc, 0, 0);
740 violations++;
741 }
742 }
743 }
744 uint32_t current = crc32_c(g_regions[i].addr, g_regions[i].size);
745 if (current != g_regions[i].baseline_crc) {
746 char desc[256];
747 snprintf(desc, sizeof(desc), "Region '%s' modified: CRC mismatch (expected 0x%08X, got 0x%08X)",
748 g_regions[i].name, g_regions[i].baseline_crc, current);
749 fire_event(SNEPPX_MONITOR_EVENT_TEXT_MODIFIED, desc,
750 (uint64_t)(uintptr_t)g_regions[i].addr, g_regions[i].size);
751 violations++;
752 }
753 }
754
755 if (!violations) {
756 int canary_ok = SNEPPX_monitor_check_canary();
757 if (!canary_ok) {
758 fire_event(SNEPPX_MONITOR_EVENT_CANARY_TRIGGERED, "Stack canary corrupted", 0, 0);
759 violations++;
760 }
761 }
762
763 violations += SNEPPX_monitor_freq_analyze();
764 violations += SNEPPX_monitor_api_hook_check();
765
766 if (g_auto_learn_enabled) {
767 uint64_t now = (uint64_t)time(NULL);
768 if (now - g_auto_learn_start < g_auto_learn_duration) {
769 for (int i = 0; i < g_region_count; i++) {
770 if (g_regions[i].active) {
771 uint32_t current = crc32_c(g_regions[i].addr, g_regions[i].size);
772 if (current != g_regions[i].baseline_crc) {
773 g_regions[i].baseline_crc = current;
774 }
775 }
776 }
777 memcpy(g_event_type_baselines, g_event_type_counts, sizeof(g_event_type_baselines));
778 } else {
779 g_auto_learn_enabled = 0;
780 }
781 }
782
783 return violations;
784}

944 }
