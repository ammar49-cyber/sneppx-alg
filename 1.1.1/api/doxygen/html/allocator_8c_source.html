76void* SNEPPX_malloc(size_t size, size_t alignment) {
77 void* ptr = NULL;
78#ifdef _WIN32
79 ptr = _aligned_malloc(size, alignment);
80#elif defined(__linux__) || defined(__APPLE__)
81 size_t align = alignment;
82 if (align < sizeof(void*)) align = sizeof(void*);
83 if ((align & (align - 1)) != 0) {
84 size_t p = 1;
85 while (p < align) p <<= 1;
86 align = p;
87 }
88 if (posix_memalign(&ptr, align, size) != 0) {
89 return NULL;
90 }
91#else
92 ptr = malloc(size);
93#endif
94 if (ptr) {
95 memset(ptr, 0, size);
96 }
97 return ptr;
98}

157/* ==================================================================
158 * Pool Allocator
159 * ==================================================================
160 *
161 * Architecture
162 * ────────────
163 * 18 size classes (16, 32, 48, 64, 96, 128, 192, 256, 384, 512,
164 * 768, 1024, 1536, 2048, 3072, 4096, 6144, 8192)
165 *
166 * Each pool has a lock-free Treiber stack of free blocks. Blocks
167 * are carved from 1 MiB chunks obtained via VirtualAlloc / mmap.
168 * Every thread keeps a small TLS cache (max 32 entries per class)
169 * to avoid global contention on the fast path.
170 */

178 768, 1024, 1536, 2048, 3072, 4096, 6144, 8192
179};

188 if (size == 0 || size > SNEPPX_POOL_MAX_SIZE) return -1;
189 if (size <= 16) return 0;
190 if (size <= 32) return 1;
191 if (size <= 48) return 2;
192 if (size <= 64) return 3;

194 if (size <= 128) return 5;
195 if (size <= 192) return 6;

246/* ------------------------------------------------------------------
247 * Lock-free stack operations (Treiber)
248 * ------------------------------------------------------------------ */

250static inline void lf_push(SNEPPXLockFreeStack* stack, SNEPPXMemNode* node) {
251 SNEPPXMemNode* old;
252 do {
253 old = (SNEPPXMemNode*)SNEPPX_ATOMIC_LOAD_PTR(&stack->head);
254 node->next = old;
255 } while (!SNEPPX_CAS_PTR(&stack->head, old, node));
256}
257
258static inline SNEPPXMemNode* lf_pop(SNEPPXLockFreeStack* stack) {
259 SNEPPXMemNode* old = (SNEPPXMemNode*)SNEPPX_ATOMIC_LOAD_PTR(&stack->head);
260 while (old) {
261 SNEPPXMemNode* next = old->next;
262 if (SNEPPX_CAS_PTR(&stack->head, old, next))
263 return old;
264 old = (SNEPPXMemNode*)SNEPPX_ATOMIC_LOAD_PTR(&stack->head);
265 }
266 return NULL;
267}

313int SNEPPX_mem_pool_init(void) {

330/* ------------------------------------------------------------------
331 * Allocate one chunk and push its blocks onto the pool stack
332 * ------------------------------------------------------------------ */
333
334static int pool_grow(SNEPPXMemPool* pool) {
335 size_t block_size = pool->block_size;
336 size_t nblocks = pool->blocks_per_chunk;
337 size_t chunk_size = block_size * nblocks;
338 uint8_t* base = (uint8_t*)os_alloc(chunk_size);
339
340 if (!base) return -1;
341
342 /* Record chunk for cleanup */
343 ChunkHeader* hdr = (ChunkHeader*)os_alloc(sizeof(ChunkHeader));
344 if (!hdr) { os_free(base, chunk_size); return -1; }
345 hdr->base = base;
346 hdr->size = chunk_size;
347 hdr->next = g_chunk_list;
348 g_chunk_list = hdr;
349
350 /* Carve into blocks and push onto the lock-free stack */
351 for (size_t b = 0; b < nblocks; b++) {
352 SNEPPXMemNode* node = (SNEPPXMemNode*)(base + b * block_size);
353 lf_push(&pool->stack, node);
354 }
355 pool->chunk_count++;
356 return 0;
357}

381void SNEPPX_tls_cache_destroy(void) {
382 if (!g_tls_cache) return;
383 /* Flush all entries back to their respective pools */
384 for (int i = 0; i < SNEPPX_NUM_SIZE_CLASSES; i++) {
385 SNEPPXTlsEntry* e = &g_tls_cache->entries[i];
386 while (e->free_list) {
387 SNEPPXMemNode* node = (SNEPPXMemNode*)e->free_list;
388 e->free_list = (void*)node->next;
389 e->count--;
390 lf_push(&g_pools[i].stack, node);
391 }
392 }
393 free(g_tls_cache);
394 g_tls_cache = NULL;
395 SNEPPX_ATOMIC_ADD_INT(&g_active_tls_caches, -1);
396}

398/* ------------------------------------------------------------------
399 * SNEPPX_pool_alloc – fast path: TLS → global stack → grow
400 * ------------------------------------------------------------------ */
401
407void* SNEPPX_pool_alloc(size_t size) {
408 if (size > SNEPPX_POOL_MAX_SIZE)
409 return SNEPPX_malloc(size, 16);
410
411 int idx = SNEPPX_pool_size_class(size);
412 if (idx < 0) return SNEPPX_malloc(size, 16);
413
414 /* 1. lazily init TLS cache */
415 if (!g_tls_cache) SNEPPX_tls_cache_init();
416
417 /* 2. try TLS cache first */
418 if (g_tls_cache) {
419 SNEPPXTlsEntry* e = &g_tls_cache->entries[idx];
420 if (e->free_list) {
421 SNEPPXMemNode* node = (SNEPPXMemNode*)e->free_list;
422 e->free_list = (void*)node->next;
423 e->count--;
424 g_tls_cache->hits++;
425 memset(node, 0, g_size_classes[idx]);
426 return node;
427 }
428 }
429
430 /* 3. try global lock-free stack */
431 SNEPPXMemNode* node = lf_pop(&g_pools[idx].stack);
432 if (!node) {
433 /* 4. grow the pool */
434 if (pool_grow(&g_pools[idx]) != 0)
435 return NULL;
436 node = lf_pop(&g_pools[idx].stack);
437 if (!node) return NULL;
438 }
439 g_pools[idx].alloc_count++;
440 SNEPPX_ATOMIC_ADD_SIZE(&g_total_pool_allocated, g_size_classes[idx]);
441 memset(node, 0, g_size_classes[idx]);
442 return node;
443}

445/* ------------------------------------------------------------------
446 * SNEPPX_pool_free – return to TLS (or fall back to global)
447 * ------------------------------------------------------------------ */
448
454void SNEPPX_pool_free(void* ptr, size_t size) {
455 if (!ptr) return;
456
457 /* Large allocations were handled by SNEPPX_malloc */
458 if (size > SNEPPX_POOL_MAX_SIZE) {
459 SNEPPX_free(ptr, size);
460 return;
461 }
462
463 int idx = SNEPPX_pool_size_class(size);
464 if (idx < 0) {
465 SNEPPX_free(ptr, size);
466 return;
467 }
468
469 if (!g_tls_cache) SNEPPX_tls_cache_init();
470
471 if (g_tls_cache) {
472 SNEPPXTlsEntry* e = &g_tls_cache->entries[idx];
473 if (e->count < e->max) {
474 ((SNEPPXMemNode*)ptr)->next = (SNEPPXMemNode*)e->free_list;
475 e->free_list = ptr;
476 e->count++;
477 return;
478 }
479 }
480
481 /* TLS full or unavailable → return to global stack */
482 lf_push(&g_pools[idx].stack, (SNEPPXMemNode*)ptr);
483 g_pools[idx].free_count++;
484 SNEPPX_ATOMIC_ADD_SIZE(&g_total_pool_freed, g_size_classes[idx]);
485}

487/* ------------------------------------------------------------------
488 * SNEPPX_mem_pool_destroy – tear down everything
489 * ------------------------------------------------------------------ */
490
494void SNEPPX_mem_pool_destroy(void) {
495 /* Destroy the calling thread's TLS cache first (flushes entries to pool) */
496 if (g_tls_cache) SNEPPX_tls_cache_destroy();
497
498 /* Free all chunks */
499 ChunkHeader* h = g_chunk_list;
500 while (h) {
501 ChunkHeader* next = h->next;
502 os_free(h->base, h->size);
503 os_free(h, sizeof(ChunkHeader));
504 h = next;
505 }
506 g_chunk_list = NULL;
507
508 /* Reset pools */
509 for (int i = 0; i < SNEPPX_NUM_SIZE_CLASSES; i++) {
510 g_pools[i].stack.head = NULL;
511 g_pools[i].alloc_count = 0;
512 g_pools[i].free_count = 0;
513 g_pools[i].chunk_count = 0;
514 }
515
516 g_pool_initialized = 0;
517 g_total_pool_allocated = 0;
518 g_total_pool_freed = 0;
519}
