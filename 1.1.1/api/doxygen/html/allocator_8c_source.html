150/* ==================================================================
151 * Pool Allocator
152 * ==================================================================
153 *
154 * Architecture
155 * ────────────
156 * 18 size classes (16, 32, 48, 64, 96, 128, 192, 256, 384, 512,
157 * 768, 1024, 1536, 2048, 3072, 4096, 6144, 8192)
158 *
159 * Each pool has a lock-free Treiber stack of free blocks. Blocks
160 * are carved from 1 MiB chunks obtained via VirtualAlloc / mmap.
161 * Every thread keeps a small TLS cache (max 32 entries per class)
162 * to avoid global contention on the fast path.

239/* ------------------------------------------------------------------
240 * Lock-free stack operations (Treiber)
241 * ------------------------------------------------------------------ */
242
243static inline void lf_push(SNEPPXLockFreeStack* stack, SNEPPXMemNode* node) {
244 SNEPPXMemNode* old;
245 do {
246 old = (SNEPPXMemNode*)SNEPPX_ATOMIC_LOAD_PTR(&stack->head);
247 node->next = old;
248 } while (!SNEPPX_CAS_PTR(&stack->head, old, node));
249}
250
251static inline SNEPPXMemNode* lf_pop(SNEPPXLockFreeStack* stack) {
252 SNEPPXMemNode* old = (SNEPPXMemNode*)SNEPPX_ATOMIC_LOAD_PTR(&stack->head);
253 while (old) {
254 SNEPPXMemNode* next = old->next;
255 if (SNEPPX_CAS_PTR(&stack->head, old, next))
256 return old;
257 old = (SNEPPXMemNode*)SNEPPX_ATOMIC_LOAD_PTR(&stack->head);
258 }
259 return NULL;
260}

306int SNEPPX_mem_pool_init(void) {

323/* ------------------------------------------------------------------
324 * Allocate one chunk and push its blocks onto the pool stack
325 * ------------------------------------------------------------------ */
326
327static int pool_grow(SNEPPXMemPool* pool) {
328 size_t block_size = pool->block_size;
329 size_t nblocks = pool->blocks_per_chunk;
330 size_t chunk_size = block_size * nblocks;
331 uint8_t* base = (uint8_t*)os_alloc(chunk_size);
332
333 if (!base) return -1;
334
335 /* Record chunk for cleanup */
336 ChunkHeader* hdr = (ChunkHeader*)os_alloc(sizeof(ChunkHeader));
337 if (!hdr) { os_free(base, chunk_size); return -1; }
338 hdr->base = base;
339 hdr->size = chunk_size;
340 hdr->next = g_chunk_list;
341 g_chunk_list = hdr;
342
343 /* Carve into blocks and push onto the lock-free stack */
344 for (size_t b = 0; b < nblocks; b++) {
345 SNEPPXMemNode* node = (SNEPPXMemNode*)(base + b * block_size);
346 lf_push(&pool->stack, node);
347 }
348 pool->chunk_count++;
349 return 0;
350}

374void SNEPPX_tls_cache_destroy(void) {
375 if (!g_tls_cache) return;
376 /* Flush all entries back to their respective pools */
377 for (int i = 0; i < SNEPPX_NUM_SIZE_CLASSES; i++) {
378 SNEPPXTlsEntry* e = &g_tls_cache->entries[i];
379 while (e->free_list) {
380 SNEPPXMemNode* node = (SNEPPXMemNode*)e->free_list;
381 e->free_list = (void*)node->next;
382 e->count--;
383 lf_push(&g_pools[i].stack, node);
384 }
385 }
386 free(g_tls_cache);
387 g_tls_cache = NULL;
388 SNEPPX_ATOMIC_ADD_INT(&g_active_tls_caches, -1);
389}

400void* SNEPPX_pool_alloc(size_t size) {
401 if (size > SNEPPX_POOL_MAX_SIZE)
402 return SNEPPX_malloc(size, 16);
403
404 int idx = SNEPPX_pool_size_class(size);
405 if (idx < 0) return SNEPPX_malloc(size, 16);
406
407 /* 1. lazily init TLS cache */
408 if (!g_tls_cache) SNEPPX_tls_cache_init();
409
410 /* 2. try TLS cache first */
411 if (g_tls_cache) {
412 SNEPPXTlsEntry* e = &g_tls_cache->entries[idx];
413 if (e->free_list) {
414 SNEPPXMemNode* node = (SNEPPXMemNode*)e->free_list;
415 e->free_list = (void*)node->next;
416 e->count--;
417 g_tls_cache->hits++;
418 memset(node, 0, g_size_classes[idx]);
419 return node;
420 }
421 }
422
423 /* 3. try global lock-free stack */
424 SNEPPXMemNode* node = lf_pop(&g_pools[idx].stack);
425 if (!node) {
426 /* 4. grow the pool */
427 if (pool_grow(&g_pools[idx]) != 0)
428 return NULL;
429 node = lf_pop(&g_pools[idx].stack);
430 if (!node) return NULL;
431 }
432 g_pools[idx].alloc_count++;
433 SNEPPX_ATOMIC_ADD_SIZE(&g_total_pool_allocated, g_size_classes[idx]);
434 memset(node, 0, g_size_classes[idx]);
435 return node;
436}

438/* ------------------------------------------------------------------
439 * SNEPPX_pool_free – return to TLS (or fall back to global)
440 * ------------------------------------------------------------------ */
441
447void SNEPPX_pool_free(void* ptr, size_t size) {
448 if (!ptr) return;
449
450 /* Large allocations were handled by SNEPPX_malloc */
451 if (size > SNEPPX_POOL_MAX_SIZE) {
452 SNEPPX_free(ptr, size);
453 return;
454 }
455
456 int idx = SNEPPX_pool_size_class(size);
457 if (idx < 0) {
458 SNEPPX_free(ptr, size);
459 return;
460 }
461
462 if (!g_tls_cache) SNEPPX_tls_cache_init();
463
464 if (g_tls_cache) {
465 SNEPPXTlsEntry* e = &g_tls_cache->entries[idx];
466 if (e->count < e->max) {
467 ((SNEPPXMemNode*)ptr)->next = (SNEPPXMemNode*)e->free_list;
468 e->free_list = ptr;
469 e->count++;
470 return;
471 }
472 }
473
474 /* TLS full or unavailable → return to global stack */
475 lf_push(&g_pools[idx].stack, (SNEPPXMemNode*)ptr);
476 g_pools[idx].free_count++;
477 SNEPPX_ATOMIC_ADD_SIZE(&g_total_pool_freed, g_size_classes[idx]);
478}

480/* ------------------------------------------------------------------
481 * SNEPPX_mem_pool_destroy – tear down everything
482 * ------------------------------------------------------------------ */
483
487void SNEPPX_mem_pool_destroy(void) {
488 /* Destroy the calling thread's TLS cache first (flushes entries to pool) */
489 if (g_tls_cache) SNEPPX_tls_cache_destroy();
490
491 /* Free all chunks */
492 ChunkHeader* h = g_chunk_list;
493 while (h) {
494 ChunkHeader* next = h->next;
495 os_free(h->base, h->size);
496 os_free(h, sizeof(ChunkHeader));
497 h = next;
498 }
499 g_chunk_list = NULL;
500
501 /* Reset pools */
502 for (int i = 0; i < SNEPPX_NUM_SIZE_CLASSES; i++) {
503 g_pools[i].stack.head = NULL;
504 g_pools[i].alloc_count = 0;
505 g_pools[i].free_count = 0;
506 g_pools[i].chunk_count = 0;
507 }
508
509 g_pool_initialized = 0;
510 g_total_pool_allocated = 0;
511 g_total_pool_freed = 0;
512}
