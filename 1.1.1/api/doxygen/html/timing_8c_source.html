106void SNEPPX_timing_random_delay(uint32_t min_ns, uint32_t max_ns) {
107 if (max_ns <= min_ns) return;
108 uint32_t range = max_ns - min_ns;

110 SNEPPX_random_bytes((uint8_t*)&r, sizeof(r));
