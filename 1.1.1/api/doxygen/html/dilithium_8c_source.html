285static int32_t dilithium_decompose_scalar(int32_t *a0, int32_t a, int32_t gamma2) {
286 int32_t alpha = 2 * gamma2;
287 int32_t r = a % DILITHIUM_Q;
288 if (r < 0) r += DILITHIUM_Q; /* r mod+ q, in [0, q-1] */
289 int32_t r0 = r % alpha; /* central remainder mod+- alpha */
290 if (r0 > alpha / 2) r0 -= alpha;
291 int32_t r1 = (r - r0) / alpha; /* HighBits (r - r0 is a multiple of alpha) */
292 if (r - r0 == DILITHIUM_Q - 1) { /* FIPS 204 Decompose special case */
293 r1 = 0;
294 r0 = r0 - 1;
295 }
296 *a0 = r0;
297 return r1;
298}
