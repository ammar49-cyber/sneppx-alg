74static int find_matching_skill(SNEPPXProceduralCache* cache, const SNEPPXTensor* state) {
75 float* sd = (float*)state->data;
76 size_t slen = state->size;
77 float s_norm = 0.0f;
78 for (size_t i = 0; i < slen; i++) s_norm += sd[i] * sd[i];
79 s_norm = sqrtf(s_norm + 1e-10f);
80 int best_idx = -1;
81 float best_sim = -1.0f;
82 for (size_t i = 0; i < cache->num_skills; i++) {
83 if (!cache->skills[i].valid || !cache->skills[i].state_pattern) continue;
84 float* pd = (float*)cache->skills[i].state_pattern->data;
85 size_t plen = cache->skills[i].state_pattern->size;
86 size_t min_dim = slen < plen ? slen : plen;
87 float dot = 0.0f, p_norm = 0.0f;
88 for (size_t j = 0; j < min_dim; j++) dot += sd[j] * pd[j];
89 for (size_t j = 0; j < plen; j++) p_norm += pd[j] * pd[j];
90 p_norm = sqrtf(p_norm + 1e-10f);
91 float sim = dot / (s_norm * p_norm + 1e-10f);
92 if (sim > best_sim) { best_sim = sim; best_idx = (int)i; }
93 }
94 if (best_sim > 0.85f) return best_idx;
95 return -1;
96}

127int sneppx_procedural_learn(SNEPPXProceduralCache* cache, const SNEPPXTensor* state,
128 const SNEPPXTensor* skill) {
129 if (!cache || !state || !skill) return -1;
130 int idx = find_matching_skill(cache, state);
131 if (idx >= 0) {
132 SNEPPXProceduralSkill* sk = &cache->skills[idx];
133 if (sk->skill_output) SNEPPX_tensor_destroy(sk->skill_output);
134 sk->skill_output = SNEPPX_tensor_copy(skill);
135 if (!sk->skill_output) return -1;
136 sk->frequency++;
137 sk->last_used = ++cache->access_counter;
138 return 0;
139 }
140 idx = evict_lru(cache);
141 if (idx < 0) return -1;
142 SNEPPXProceduralSkill* sk = &cache->skills[idx];
143 sk->state_pattern = SNEPPX_tensor_copy(state);
144 if (!sk->state_pattern) return -1;
145 sk->skill_output = SNEPPX_tensor_copy(skill);
146 if (!sk->skill_output) {
147 SNEPPX_tensor_destroy(sk->state_pattern);
148 sk->state_pattern = NULL;
149 return -1;
150 }
151 sk->frequency = 1;
152 sk->last_used = ++cache->access_counter;
153 sk->sequence_length = 1;
154 sk->valid = 1;
155 return 0;
