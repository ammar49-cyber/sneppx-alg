180void* SNEPPX_secure_malloc(SNEPPXSecurePool* pool, size_t size, size_t alignment) {
181 if (!pool || !size) return NULL;
182 if (alignment < 16) alignment = 16;
183 size_t header = pool->use_canaries ? SNEPPX_CANARY_SIZE : 0;
184 size_t aligned_offset = (pool->used + alignment - 1) & ~(alignment - 1);
185 size_t needed = aligned_offset + header + size + header;
186 if (needed > pool->capacity) return NULL;
187 uint8_t* ptr = pool->base + aligned_offset + header;
188 if (pool->use_canaries) {
189 SNEPPXCanary canary;
190 SNEPPX_canary_generate(&canary);
191 SNEPPX_canary_write(&canary, ptr - SNEPPX_CANARY_SIZE);
192 SNEPPX_canary_write(&canary, ptr + size);
193 }
194 pool->used = needed;
195 if (pool->used > pool->peak) pool->peak = pool->used;
196 pool->alloc_count++;
197 return ptr;
198}

206void SNEPPX_secure_pool_free(SNEPPXSecurePool* pool, void* ptr, size_t size) {
207 if (!pool || !ptr) return;
208 if (pool->use_canaries) {
209 uint8_t* mem = (uint8_t*)ptr;
210 SNEPPXCanary* lead = (SNEPPXCanary*)(mem - SNEPPX_CANARY_SIZE);
211 SNEPPXCanary* trail = (SNEPPXCanary*)(mem + size);
212 if (!SNEPPX_canary_verify(lead, lead->value)) {
213 fprintf(stderr, "HEAP CORRUPTION: leading canary mismatch at %p\n", ptr);
214 abort();
215 }
216 if (!SNEPPX_canary_verify(trail, trail->value)) {
217 fprintf(stderr, "HEAP CORRUPTION: trailing canary mismatch at %p\n", ptr);
218 abort();
219 }
220 }
221 SNEPPX_secure_zero(ptr, size);
222}
