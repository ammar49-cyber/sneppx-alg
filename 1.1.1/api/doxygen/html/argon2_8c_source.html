21static void blake2b_round(uint64_t v[16], const uint64_t m[16]) {
22 #define B2B_G(a,b,c,d,x,y) do { \
23 v[a] += v[b] + x; v[d] ^= v[a]; v[d] = (v[d] >> 32) | (v[d] << 32); \
24 v[c] += v[d]; v[b] ^= v[c]; v[b] = (v[b] >> 24) | (v[b] << 40); \
25 v[a] += v[b] + y; v[d] ^= v[a]; v[d] = (v[d] >> 16) | (v[d] << 48); \
26 v[c] += v[d]; v[b] ^= v[c]; v[b] = (v[b] >> 63) | (v[b] << 1); \
27 } while(0)
28 B2B_G(0,4,8,12, m[0], m[1]);
29 B2B_G(1,5,9,13, m[2], m[3]);
30 B2B_G(2,6,10,14, m[4], m[5]);
31 B2B_G(3,7,11,15, m[6], m[7]);
32 B2B_G(0,5,10,15, m[8], m[9]);
33 B2B_G(1,6,11,12, m[10], m[11]);
34 B2B_G(2,7,8,13, m[12], m[13]);
35 B2B_G(3,4,9,14, m[14], m[15]);
36 #undef B2B_G
37}

39static void argon2_G(uint64_t out[128], const uint64_t in1[128], const uint64_t in2[128]) {
40 uint64_t z[128], v[16], m[16];
41 for (int i = 0; i < 128; i++) z[i] = in1[i] ^ in2[i];
42 memcpy(out, z, 128 * 8);
43 for (int r = 0; r < 8; r++) {
44 for (int slice = 0; slice < 8; slice++) {
45 int s = slice * 16;
46 memcpy(v, out + s, 16 * 8);
47 memcpy(m, z + s, 16 * 8);
48 blake2b_round(v, m);
49 memcpy(out + s, v, 16 * 8);
50 }
51 }
52 for (int i = 0; i < 128; i++) out[i] ^= z[i];
53}

55static void fill_block_with_xor(const uint64_t prev[128], const uint64_t ref[128], uint64_t next[128], int with_xor) {
56 uint64_t tmp[128];
57 argon2_G(tmp, prev, ref);
58 if (with_xor) { for (int i = 0; i < 128; i++) next[i] ^= tmp[i]; }
59 else { memcpy(next, tmp, 128 * 8); }
60}

73int SNEPPX_argon2id(const uint8_t* password, size_t password_len, const uint8_t* salt, size_t salt_len,
74 const SNEPPXArgon2Config* config, uint8_t* hash) {
75 if (!password || !salt || !config || !hash) return -1;
76 size_t m = config->memory_kb;
77 size_t t = config->iterations;
78 size_t p = config->parallelism;
79 size_t hlen = config->hash_len;
80 if (m < 2 || t < 1 || p < 1 || m % p != 0) return -1;
81
82 size_t segments = m;
83 uint64_t* memory = (uint64_t*)calloc(segments, 128 * sizeof(uint64_t));
84 if (!memory) return -1;
85
86 uint64_t input[16];
87 memset(input, 0, sizeof(input));
88 input[0] = 0x0100 | (uint64_t)p; input[1] = (uint64_t)t;
89 input[2] = (uint64_t)m; input[3] = (uint64_t)password_len;
90 input[4] = (uint64_t)salt_len; input[5] = 0;
91 if (password_len > 0) memcpy(((uint8_t*)input) + 64, password, password_len < 64 ? password_len : 64);
92 if (salt_len > 0) memcpy(((uint8_t*)input) + 64 + password_len, salt, salt_len < 64 ? salt_len : 32);
93
94 uint64_t h0[8], v[16], msg[16];
95 memcpy(v, input, 128);
96 memcpy(v + 8, input, 64);
97 memcpy(msg, input, 128);
98 blake2b_round(v, msg);
99 memcpy(h0, v, 64);
100
101 size_t blk = 0;
102 memcpy(memory, h0, 64);
103 memset(memory + 8, 0, 120 * 8);
104 for (size_t i = 1; i < m; i++) {
105 argon2_G(memory + i * 128, memory + (i - 1) * 128, memory);
106 blk = i;
107 }
108
109 for (size_t pass = 0; pass < t; pass++) {
110 for (size_t slice = 0; slice < 4; slice++) {
111 size_t start = (slice * m) / 4;
112 size_t end = ((slice + 1) * m) / 4;
113 for (size_t col = start; col < end; col++) {
114 size_t prev = col == 0 ? m - 1 : col - 1;
115 size_t ref = (col + prev) % m;
116 int with_xor = pass > 0;
117 fill_block_with_xor(memory + prev * 128, memory + ref * 128, memory + col * 128, with_xor);
118 blk = col;
119 }
120 }
121 }
122
123 uint64_t final[128];
124 memset(final, 0, sizeof(final));
125 for (size_t i = 0; i < m; i++) {
126 for (int j = 0; j < 128; j++) final[j] ^= memory[i * 128 + j];
127 }
128
129 uint8_t result[64];
130 for (int i = 0; i < 8; i++) {
131 for (int j = 0; j < 8; j++) result[i * 8 + j] = (uint8_t)(final[i] >> (j * 8));
132 }
133
134 size_t out_len = hlen;
135 memcpy(hash, result, out_len > 64 ? 64 : out_len);
136 if (out_len > 64) {
137 /* Extend by hashing output, simplified for now */
138 }
139 free(memory);
140 return 0;
141}
