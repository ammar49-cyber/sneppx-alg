174static void freelist_insert(SNEPPXSecureFreeNode* node) {
175 if (!g_freelist_head) {
176 g_freelist_head = node;
177 return;
178 }
179 SNEPPXSecureFreeNode* cur = g_freelist_head;
180 SNEPPXSecureFreeNode* prv = NULL;
181 while (cur && cur->base < node->base) {
182 prv = cur;
183 cur = cur->next;
184 }
185 if (prv) {
186 char* prv_end = (char*)prv->base + prv->total;
187 if (prv_end == node->base) {
188 prv->total += node->total;
189 prv->canary = SNEPPX_SECURE_FREELIST_CANARY;
190 prv->next = node->next;
191 if (node->next) node->next->prev = prv;
192 free(node);
193 node = prv;
194 }
195 }
196 if (cur) {
197 char* node_end = (char*)node->base + node->total;
198 if (node_end == cur->base) {
199 node->total += cur->total;
200 node->canary = SNEPPX_SECURE_FREELIST_CANARY;
201 node->next = cur->next;
202 if (cur->next) cur->next->prev = node;
203 if (g_freelist_head == cur) g_freelist_head = node;
204 free(cur);
205 }
206 }
207 if (node != prv && node->prev != prv) {
208 node->next = cur;
209 node->prev = prv;
210 if (prv) prv->next = node;
211 else g_freelist_head = node;
212 if (cur) cur->prev = node;
213 }
214}

265static int small_bin_find_free(int bin_idx) {
266 if (bin_idx < 0 || bin_idx >= SNEPPX_SMALL_BINS) return -1;
267 for (int w = 0; w < SNEPPX_SMALL_BITMAP_WORDS; w++) {
268 uint32_t bits = g_small_bins[bin_idx].bitmap[w];
269 if (bits != 0xFFFFFFFF) {
270 for (int b = 0; b < 32; b++) {
271 if (!(bits & (1u << b))) {
272 return w * 32 + b;
273 }
274 }
275 }
276 }
277 return -1;

346void* SNEPPX_secure_alloc(SNEPPXSecureAllocator* alloc, size_t bytes, size_t alignment) {
347 (void)alignment;
348 if (!alloc || bytes == 0) return NULL;
349
350 if (bytes <= SNEPPX_SMALL_OBJECT_MAX) {
351 int bin_idx = find_small_bin(bytes);
352 if (bin_idx >= 0) {
353 int slot = small_bin_find_free(bin_idx);
354 if (slot >= 0) {
355 small_bin_mark_used(bin_idx, slot);
356 void* ptr = g_small_bins[bin_idx].blocks[slot];
357 if (ptr) {
358 g_small_bins[bin_idx].count++;
359 add_record(ptr, g_small_bins[bin_idx].bin_size, 0, 0, 0);
360 alloc->total_allocated += g_small_bins[bin_idx].bin_size;
361 alloc->num_allocations++;
362 if (alloc->total_allocated > alloc->peak_allocated)
363 alloc->peak_allocated = alloc->total_allocated;
364 return ptr;
365 }
366 }
367 }
368 }
369
370 size_t alloc_bytes = bytes + SNEPPX_SECURE_OVERHEAD_SIZE;
371 size_t total = round_up_page(alloc_bytes + SNEPPX_SECURE_CANARY_SIZE);
372 size_t total_with_guards = total + SNEPPX_SECURE_PAGE_SIZE * 2;
373
374 void* base = NULL;
375#ifdef _WIN32
376 base = VirtualAlloc(NULL, total_with_guards, MEM_RESERVE, PAGE_NOACCESS);
377 if (!base) return NULL;
378 void* user_region = (char*)base + SNEPPX_SECURE_PAGE_SIZE;
379 void* user_committed = VirtualAlloc(user_region, total, MEM_COMMIT, PAGE_READWRITE);
380 if (!user_committed) { VirtualFree(base, 0, MEM_RELEASE); return NULL; }
381#else
382 base = mmap(NULL, total_with_guards, PROT_NONE, MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
383 if (base == MAP_FAILED) return NULL;
384 void* user_region = (char*)base + SNEPPX_SECURE_PAGE_SIZE;
385 if (mprotect(user_region, total, PROT_READ | PROT_WRITE) != 0) {
386 munmap(base, total_with_guards); return NULL;
387 }
388#endif
389
390 SNEPPXSecureAllocHeader* hdr = (SNEPPXSecureAllocHeader*)user_region;
391 hdr->magic = SNEPPX_SECURE_GUARD_MAGIC;
392 hdr->actual_size = bytes;
393 hdr->reserved = 0;
394
395 void* ptr = (char*)user_region + SNEPPX_SECURE_OVERHEAD_SIZE;
396
397 uint64_t canary_val = SNEPPX_secure_canary_generate();
398 uint64_t* canary_ptr = (uint64_t*)((char*)ptr + total - SNEPPX_SECURE_OVERHEAD_SIZE - SNEPPX_SECURE_CANARY_SIZE);
399 *canary_ptr = canary_val;
400
401 if (alloc->use_guard_pages) {
402 alloc->total_allocated += bytes;
403 alloc->num_allocations++;
404 if (alloc->total_allocated > alloc->peak_allocated)
405 alloc->peak_allocated = alloc->total_allocated;
406 }
407
408 add_record(ptr, bytes, SNEPPX_SECURE_PAGE_SIZE, SNEPPX_SECURE_PAGE_SIZE, canary_val);
409 return ptr;
410}

417void SNEPPX_secure_free(SNEPPXSecureAllocator* alloc, void* ptr) {
418 if (!alloc || !ptr) return;
419
420 for (int i = 0; i < g_small_bins_initialized && i < SNEPPX_SMALL_BINS; i++) {
421 for (int j = 0; j < SNEPPX_SMALL_BIN_CAPACITY; j++) {
422 if (g_small_bins[i].blocks[j] == ptr) {
423 small_bin_mark_free(i, j);
424 g_small_bins[i].count--;
425 alloc->total_allocated -= (alloc->total_allocated >= (size_t)g_small_bins[i].bin_size) ? (size_t)g_small_bins[i].bin_size : alloc->total_allocated;
426 g_stats_num_frees++;
427 return;
428 }
429 }
430 }
431
432 if (freed_set_check(ptr)) {
433 g_stats_num_double_free++;
434 if (alloc->on_overflow) {
435 for (int i = 0; i < g_record_count; i++) {
436 if (g_records[i].addr == ptr) {
437 alloc->on_overflow(&g_records[i]);
438 break;
439 }
440 }
441 }
442 return;
443 }
444
445 for (int i = 0; i < g_record_count; i++) {
446 if (g_records[i].addr == ptr && !g_records[i].is_freed) {
447 SNEPPXSecureAllocRecord* r = &g_records[i];
448 size_t alloc_bytes = r->size + SNEPPX_SECURE_OVERHEAD_SIZE;
449 size_t total = round_up_page(alloc_bytes + SNEPPX_SECURE_CANARY_SIZE);
450 size_t total_with_guards = total + r->guard_front + r->guard_back;
451 void* base = (char*)ptr - r->guard_front - SNEPPX_SECURE_OVERHEAD_SIZE;
452 void* user_region = (char*)base + r->guard_front;
453
454 SNEPPXSecureAllocHeader* hdr = (SNEPPXSecureAllocHeader*)user_region;
455 if (hdr->magic != SNEPPX_SECURE_GUARD_MAGIC) {
456 g_stats_num_canary_violations++;
457 if (alloc->on_overflow) alloc->on_overflow(r);
458 }
459
460 uint64_t* canary_ptr = (uint64_t*)((char*)ptr + total - SNEPPX_SECURE_OVERHEAD_SIZE - SNEPPX_SECURE_CANARY_SIZE);
461 if (*canary_ptr != r->canary) {
462 g_stats_num_canary_violations++;
463 if (alloc->on_overflow) alloc->on_overflow(r);
464 }
465
466 secure_scrub(ptr, r->size);
467 r->is_freed = 1;
468 alloc->total_allocated -= (alloc->total_allocated >= r->size) ? r->size : alloc->total_allocated;
469 g_stats_num_frees++;
470 freed_set_add(ptr);
471
472 SNEPPXSecureFreeNode* node = freelist_node_alloc(base, total_with_guards);
473 if (node) freelist_insert(node);
474
475#ifdef _WIN32
476 VirtualFree(base, 0, MEM_RELEASE);
477#else
478 munmap(base, total_with_guards);
479#endif
480 return;
481 }
482 }

564int SNEPPX_secure_freelist_check(SNEPPXSecureAllocator* alloc) {
565 (void)alloc;
566 if (!g_initialized) return -1;
567 int errors = 0;
568 SNEPPXSecureFreeNode* cur = g_freelist_head;
569 while (cur) {
570 if (cur->canary != SNEPPX_SECURE_FREELIST_CANARY) {
571 fprintf(stderr, "FREELIST ERROR: canary corrupted at node %p (base %p)\n", (void*)cur, cur->base);
572 errors++;
573 }
574 if (cur->prev && cur->prev->next != cur) {
575 fprintf(stderr, "FREELIST ERROR: broken prev link at node base %p\n", cur->base);
576 errors++;
577 }
578 if (cur->next && cur->next->prev != cur) {
579 fprintf(stderr, "FREELIST ERROR: broken next link at node base %p\n", cur->base);
580 errors++;
581 }
582 if (cur->next && (char*)cur->base + cur->total > (char*)cur->next->base) {
583 fprintf(stderr, "FREELIST ERROR: overlapping or out-of-order blocks at base %p and %p\n", cur->base, cur->next->base);
584 errors++;
585 }
586 if (cur->next && (char*)cur->base + cur->total == (char*)cur->next->base) {
587 fprintf(stderr, "FREELIST ERROR: adjacent blocks should be merged at base %p\n", cur->base);
588 errors++;
589 }
590 cur = cur->next;
591 }
592 return (errors == 0) ? 0 : -1;
593}

614 for (int i = 0; i < g_record_count; i++) {
615 if (g_records[i].addr == ptr && !g_records[i].is_freed) {
616 SNEPPXSecureAllocRecord* r = &g_records[i];
617 size_t alloc_bytes = r->size + SNEPPX_SECURE_OVERHEAD_SIZE;
618 size_t total = round_up_page(alloc_bytes + SNEPPX_SECURE_CANARY_SIZE);
619 size_t total_with_guards = total + r->guard_front + r->guard_back;
620 void* base = (char*)ptr - r->guard_front - SNEPPX_SECURE_OVERHEAD_SIZE;
621 void* user_region = (char*)base + r->guard_front;
622
623 SNEPPXSecureAllocHeader* hdr = (SNEPPXSecureAllocHeader*)user_region;
624 if (hdr->magic != SNEPPX_SECURE_GUARD_MAGIC) {
625 g_stats_num_canary_violations++;
626 if (alloc->on_overflow) alloc->on_overflow(r);
627 }
628
629 uint64_t* canary_ptr = (uint64_t*)((char*)ptr + total - SNEPPX_SECURE_OVERHEAD_SIZE - SNEPPX_SECURE_CANARY_SIZE);
630 if (*canary_ptr != r->canary) {
631 g_stats_num_canary_violations++;
632 if (alloc->on_overflow) alloc->on_overflow(r);
633 }
634
635 secure_scrub(ptr, r->size);
636 r->is_freed = 1;
637 alloc->total_allocated -= (alloc->total_allocated >= r->size) ? r->size : alloc->total_allocated;
638 g_stats_num_frees++;
639 freed_set_add(ptr);
640
641 SNEPPXSecureFreeNode* node = freelist_node_alloc(base, total_with_guards);
642 if (node) freelist_insert(node);
643
644 g_quarantine[g_quarantine_count].base = base;
645 g_quarantine[g_quarantine_count].total_with_guards = total_with_guards;
646 g_quarantine_count++;
647 return 0;
648 }
649 }
650 return -1;
651}

702double SNEPPX_secure_allocator_get_fragmentation(void) {
703 if (!g_freelist_head) return 0.0;
704 size_t total_free = 0;
705 size_t largest_free = 0;
706 int block_count = 0;
707 SNEPPXSecureFreeNode* cur = g_freelist_head;
708 while (cur) {
709 total_free += cur->total;
710 if (cur->total > largest_free) largest_free = cur->total;
711 block_count++;
712 cur = cur->next;
713 }
714 if (total_free == 0 || block_count <= 1) return 0.0;
715 return 1.0 - ((double)largest_free / (double)total_free);
716}

723int SNEPPX_secure_allocator_defragment(void) {
724 if (!g_freelist_head) return 0;
725 int merged = 0;
726 SNEPPXSecureFreeNode* cur = g_freelist_head;
727 while (cur && cur->next) {
728 char* cur_end = (char*)cur->base + cur->total;
729 if (cur_end == cur->next->base) {
730 cur->total += cur->next->total;
731 SNEPPXSecureFreeNode* victim = cur->next;
732 cur->next = victim->next;
733 if (victim->next) victim->next->prev = cur;
734 free(victim);
735 merged++;
736 } else {
737 cur = cur->next;
738 }
739 }
740 return merged;
741}

772static int small_bin_populate(int bin_idx) {
773 if (bin_idx < 0 || bin_idx >= SNEPPX_SMALL_BINS) return -1;
774 SNEPPXSmallObjectBin* bin = &g_small_bins[bin_idx];
775 int size = bin->bin_size;
776 int needed = 0;
777 for (int w = 0; w < SNEPPX_SMALL_BITMAP_WORDS; w++) {
778 uint32_t bits = bin->bitmap[w];
779 for (int b = 0; b < 32; b++) {
780 if (!(bits & (1u << b)) && bin->blocks[w * 32 + b] == NULL) {
781 needed++;
782 }
783 }
784 }
785 if (needed == 0) return 0;
786 size_t block_total = round_up_page((size_t)size * needed + SNEPPX_SECURE_OVERHEAD_SIZE + SNEPPX_SECURE_CANARY_SIZE);
787 size_t total_with_guards = block_total + SNEPPX_SECURE_PAGE_SIZE * 2;
788 void* base = NULL;
789#ifdef _WIN32
790 base = VirtualAlloc(NULL, total_with_guards, MEM_RESERVE, PAGE_NOACCESS);
791 if (!base) return -1;
792 void* user_region = (char*)base + SNEPPX_SECURE_PAGE_SIZE;
793 void* user_committed = VirtualAlloc(user_region, block_total, MEM_COMMIT, PAGE_READWRITE);
794 if (!user_committed) { VirtualFree(base, 0, MEM_RELEASE); return -1; }
795#else
796 base = mmap(NULL, total_with_guards, PROT_NONE, MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
797 if (base == MAP_FAILED) return -1;
798 void* user_region = (char*)base + SNEPPX_SECURE_PAGE_SIZE;
799 if (mprotect(user_region, block_total, PROT_READ | PROT_WRITE) != 0) {
800 munmap(base, total_with_guards); return -1;
801 }
802#endif
803 int slot = 0;
804 for (int w = 0; w < SNEPPX_SMALL_BITMAP_WORDS && slot < needed; w++) {
805 uint32_t bits = bin->bitmap[w];
806 for (int b = 0; b < 32 && slot < needed; b++) {
807 int idx = w * 32 + b;
808 if (!(bits & (1u << b)) && bin->blocks[idx] == NULL) {
809 bin->blocks[idx] = (char*)user_region + (size_t)slot * size;
810 SNEPPXSecureAllocHeader* hdr = (SNEPPXSecureAllocHeader*)bin->blocks[idx];
811 hdr->magic = SNEPPX_SECURE_GUARD_MAGIC;
812 hdr->actual_size = (size_t)size;
813 hdr->reserved = 0;
814 bin->blocks[idx] = (char*)bin->blocks[idx] + SNEPPX_SECURE_OVERHEAD_SIZE;
815 slot++;
816 }
817 }
818 }
819 return slot;
820}
821static int freelist_split_block(SNEPPXSecureFreeNode* node, size_t needed) {
822 if (!node || node->total < needed) return -1;
823 size_t remaining = node->total - needed;
824 if (remaining < SNEPPX_SECURE_PAGE_SIZE) return -1;
825 void* split_base = (char*)node->base + needed;
826 SNEPPXSecureFreeNode* split = freelist_node_alloc(split_base, remaining);
827 if (!split) return -1;
828 node->total = needed;
829 split->next = node->next;
830 split->prev = node;
831 if (node->next) node->next->prev = split;
832 node->next = split;
833 return 0;
834}
