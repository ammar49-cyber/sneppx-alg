112SNEPPXCognitiveMemory* SNEPPX_cognitive_memory_create(
113 const SNEPPXCognitiveMemoryConfig* config, unsigned int seed) {
114 if (!config) return NULL;
115 SNEPPXCognitiveMemory* cmem = SNEPPX_malloc(sizeof(SNEPPXCognitiveMemory), 64);
116 if (!cmem) return NULL;
117 memset(cmem, 0, sizeof(SNEPPXCognitiveMemory));
118 cmem->config = *config;
119 cmem->seed = seed;
120 cmem->episodic = sneppx_episodic_create(&config->episodic, &cmem->seed);
121 if (!cmem->episodic) {
122 SNEPPX_free(cmem, sizeof(SNEPPXCognitiveMemory));
123 return NULL;
124 }
125 cmem->semantic = sneppx_semantic_create(&config->semantic, &cmem->seed);
126 if (!cmem->semantic) {
127 sneppx_episodic_destroy(cmem->episodic);
128 SNEPPX_free(cmem, sizeof(SNEPPXCognitiveMemory));
129 return NULL;
130 }
131 cmem->working = sneppx_working_create(&config->working);
132 if (!cmem->working) {
133 sneppx_episodic_destroy(cmem->episodic);
134 sneppx_semantic_destroy(cmem->semantic);
135 SNEPPX_free(cmem, sizeof(SNEPPXCognitiveMemory));
136 return NULL;
137 }
138 cmem->procedural = sneppx_procedural_create(&config->procedural, &cmem->seed);
139 if (!cmem->procedural) {
140 sneppx_episodic_destroy(cmem->episodic);
141 sneppx_semantic_destroy(cmem->semantic);
142 sneppx_working_destroy(cmem->working);
143 SNEPPX_free(cmem, sizeof(SNEPPXCognitiveMemory));
144 return NULL;
145 }
146 return cmem;
147}

169int SNEPPX_cognitive_memory_forward(
170 SNEPPXCognitiveMemory* cmem,
171 const SNEPPXTensor* input,
172 SNEPPXTensor** output) {
173 if (!cmem || !input || !output) return -1;
174 sneppx_working_write(cmem->working, 0, input);
175 SNEPPXTensor* attended = NULL;
176 sneppx_working_attend(cmem->working, input, &attended);
177 SNEPPXTensor* semantic_result = NULL;
178 sneppx_semantic_retrieve(cmem->semantic, input, 3, &semantic_result, NULL);
179 SNEPPXTensor* procedural_result = NULL;
180 sneppx_procedural_recall(cmem->procedural, input, &procedural_result);
181 size_t out_dim = input->size + cmem->config.working.slot_dim +
182 cmem->config.semantic.embedding_dim +
183 cmem->config.procedural.skill_dim;
184 size_t shape[2] = {1, out_dim};
185 SNEPPXTensor* result = SNEPPX_tensor_zeros(shape, 2, SNEPPX_FLOAT32);
186 if (!result) {
187 if (attended) SNEPPX_tensor_destroy(attended);
188 if (semantic_result) SNEPPX_tensor_destroy(semantic_result);
189 if (procedural_result) SNEPPX_tensor_destroy(procedural_result);
190 return -1;
191 }
192 float* rd = (float*)result->data;
193 size_t offset = 0;
194 memcpy(rd + offset, input->data, input->size * sizeof(float));
195 offset += input->size;
196 if (attended) {
197 size_t copy_dim = cmem->config.working.slot_dim < attended->size ?
198 cmem->config.working.slot_dim : attended->size;
199 memcpy(rd + offset, attended->data, copy_dim * sizeof(float));
200 SNEPPX_tensor_destroy(attended);
201 }
202 offset += cmem->config.working.slot_dim;
203 if (semantic_result) {
204 float* sd = (float*)semantic_result->data;
205 size_t num_rows = semantic_result->shape[0];
206 size_t emb_dim = semantic_result->shape[1];
207 size_t copy_dim = cmem->config.semantic.embedding_dim < emb_dim ?
208 cmem->config.semantic.embedding_dim : emb_dim;
209 for (size_t r = 0; r < num_rows && r < 3; r++) {
210 for (size_t c = 0; c < copy_dim; c++) {
211 rd[offset + c] += sd[r * emb_dim + c] / (float)num_rows;
212 }
213 }
214 SNEPPX_tensor_destroy(semantic_result);
215 }
216 offset += cmem->config.semantic.embedding_dim;
217 if (procedural_result) {
218 size_t copy_dim = cmem->config.procedural.skill_dim < procedural_result->size ?
219 cmem->config.procedural.skill_dim : procedural_result->size;
220 memcpy(rd + offset, procedural_result->data, copy_dim * sizeof(float));
221 SNEPPX_tensor_destroy(procedural_result);
222 }
223 *output = result;
224 return 0;
225}
