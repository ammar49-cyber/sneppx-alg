55int SNEPPX_radam_step(SNEPPXRAdamState* state, float* params, const float* grads) {
56 if (!state || !params || !grads) return -1;
57 if (!state->m || !state->v) return -1;
58 size_t sz = state->size;
59 float b1 = state->beta1, b2 = state->beta2;
60 float eps = state->eps, lr = state->lr;
61 float* m = state->m, *v = state->v;
62
63 state->t++;
64 int t = state->t;
65
66 float b1_t = powf(b1, (float)t);
67 float b2_t = powf(b2, (float)t);
68 float rho_inf = 2.0f / (1.0f - b2) - 1.0f;
69 float rho_t = rho_inf - 2.0f * (float)t * b2_t / (1.0f - b2_t);
70
71 for (size_t i = 0; i < sz; i++) {
72 float g = grads[i];
73 m[i] = b1 * m[i] + (1.0f - b1) * g;
74 v[i] = b2 * v[i] + (1.0f - b2) * g * g;
75 float m_hat = m[i] / (1.0f - b1_t);
76 float update;
77 if (rho_t > 4.0f) {
78 float l = sqrtf((1.0f - b2_t) / (v[i] + eps));
79 float r_num = (rho_t - 4.0f) * (rho_t - 2.0f) * rho_inf;
80 float r_den = (rho_inf - 4.0f) * (rho_inf - 2.0f) * rho_t;
81 float r = sqrtf(r_num / r_den);
82 update = m_hat * r * l;
83 } else {
84 update = m_hat;
85 }
86 params[i] -= lr * update;
87 }
88 return 0;
