44 size_t idx[8]; size_t tmp = flat;
45 for (long i = (long)t->ndim - 1; i >= 0; i--) {
46 idx[i] = (t->strides[i] > 0) ? (tmp / t->strides[i]) % t->shape[i] : 0;

116int SNEPPX_grad_opt_step(SNEPPXGradientOptimizer* opt, SNEPPXTensor** params, SNEPPXTensor** grads, size_t num_params) {
117 if (!opt || !params || !grads) return -1;
118 ensure_buffers(opt, num_params);
119 opt->step_count++;
120 int is_adam = (opt->type == SNEPPX_OPTIMIZER_ADAM || opt->type == SNEPPX_OPTIMIZER_ADAMW || opt->type == SNEPPX_OPTIMIZER_RADAM || opt->type == SNEPPX_OPTIMIZER_SF_ADAMW);
121 double lr = opt->lr;
122 double wd = opt->weight_decay;
123 for (size_t p = 0; p < num_params; p++) {
124 SNEPPXTensor* param = params[p];
125 SNEPPXTensor* grad = grads[p];
126 if (!param || !grad) return -1;
127 size_t n = param->size < grad->size ? param->size : grad->size;
128 for (size_t i = 0; i < n; i++) {
129 float value = get_flat(param, i);
130 float g = get_flat(grad, i);
131 if (wd != 0.0) g += (float)(wd * value);
132 if (is_adam) {
133 float* m = &opt->buf[p];
134 float* v = &opt->buf2[p];
135 *m = (float)(opt->beta1 * (*m) + (1.0 - opt->beta1) * g);
136 *v = (float)(opt->beta2 * (*v) + (1.0 - opt->beta2) * g * g);
137 double bc1 = 1.0 - pow(opt->beta1, (double)opt->step_count);
138 double bc2 = 1.0 - pow(opt->beta2, (double)opt->step_count);
139 float mhat = (float)(*m / bc1);
140 float vhat = (float)(*v / bc2);
141 float denom = (float)sqrt((double)vhat) + (float)opt->epsilon;
142 float update = mhat / denom;
143 if (opt->type == SNEPPX_OPTIMIZER_ADAMW || opt->type == SNEPPX_OPTIMIZER_SF_ADAMW)
144 value -= (float)(lr * (wd * value + update));
145 else
146 value -= (float)(lr * update);
147 } else {
148 float g2 = g;
149 if (opt->momentum != 0.0) {
150 float prev = opt->buf[p];
151 float cur = (float)(opt->momentum * prev + g2);
152 if (opt->nesterov) g2 = (float)(opt->momentum * prev + g2);
153 else g2 = cur;
154 opt->buf[p] = cur;
155 }
156 value -= (float)(lr * g2);
157 }
158 set_flat(param, i, value);
159 }
160 }
161 return 0;
162}

231int SNEPPX_grad_opt_clip_grad_norm(SNEPPXTensor** grads, size_t num_grads, double max_norm) {
232 if (!grads) return -1;
233 double total = 0.0;
234 for (size_t g = 0; g < num_grads; g++) {
235 if (!grads[g]) continue;
236 float* d = (float*)grads[g]->data;
237 for (size_t i = 0; i < grads[g]->size; i++) total += (double)d[i] * d[i];
238 }
239 double norm = sqrt(total);
240 if (norm > max_norm && norm > 0.0) {
241 double scale = max_norm / norm;
242 for (size_t g = 0; g < num_grads; g++) {
243 if (!grads[g]) continue;
244 float* d = (float*)grads[g]->data;
245 for (size_t i = 0; i < grads[g]->size; i++) d[i] = (float)(d[i] * scale);
246 }
247 }
248 return 0;
249}
