59static THREAD_RET worker_thread_main(THREAD_ARG arg) {
60 SNEPPXWorker* w = (SNEPPXWorker*)arg;
61 SNEPPXThreadScheduler* sched = (SNEPPXThreadScheduler*)w->thread_handle;
62 SNEPPXTask task_buf; memset(&task_buf, 0, sizeof(task_buf));
63 while (1) {
64 if (sched->shutdown_flag && w->deque_head == w->deque_tail) break;
65 int got_task = 0;
66 if (w->deque_head < w->deque_tail) {
67 w->deque_head++;
68 task_buf = w->deque[(w->deque_head - 1) % w->deque_capacity];
69 got_task = 1;
70 }
71 if (!got_task) {
72 SCHED_ACQUIRE(&sched->global_lock);
73 if (sched->global_queue_size > 0) {
74 task_buf = sched->global_queue[--sched->global_queue_size];
75 got_task = 1;
76 }
77 SCHED_RELEASE(&sched->global_lock);
78 }
79 if (!got_task) {
80 for (int i = 0; i < sched->num_workers; i++) {
81 SNEPPXWorker* victim = sched->workers[i];
82 if (victim == w) continue;
83 SCHED_ACQUIRE(&sched->global_lock);
84 if (victim->deque_head < victim->deque_tail) {
85 victim->deque_tail--;
86 task_buf = victim->deque[victim->deque_tail % victim->deque_capacity];
87 got_task = 1;
88 w->tasks_stolen++;
89 }
90 SCHED_RELEASE(&sched->global_lock);
91 if (got_task) break;
92 }
93 }
94 if (got_task) {
95 w->tasks_executed++;
96 task_buf.fn(task_buf.arg);
97 } else {
98#ifdef _WIN32
99 SwitchToThread();
100#else
101 sched_yield();
102#endif
103 }
104 }
105 w->is_running = 0;
106 return 0;
107}

116int SNEPPX_scheduler_init(SNEPPXThreadScheduler* sched, int num_workers) {
117 if (!sched) return -1;
118 memset(sched, 0, sizeof(*sched));
119 if (num_workers <= 0) {
120#ifdef _WIN32
121 SYSTEM_INFO si; GetSystemInfo(&si); num_workers = (int)si.dwNumberOfProcessors;
122#else
123 num_workers = (int)sysconf(_SC_NPROCESSORS_ONLN);
124#endif
125 if (num_workers < 1) num_workers = 1;
126 }
127 sched->num_workers = num_workers;
128 sched->num_threads = 0;
129 sched->shutdown_flag = 0;
130 sched->global_lock = calloc(1, sizeof(SCHED_LOCK));
131 if (sched->global_lock) SCHED_INIT_LOCK(sched->global_lock);
132 sched->global_queue_capacity = INITIAL_DEQUE_CAPACITY;
133 sched->global_queue = (SNEPPXTask*)calloc(sched->global_queue_capacity, sizeof(SNEPPXTask));
134 if (!sched->global_queue) return -1;
135 sched->global_queue_size = 0;
136 sched->workers = (SNEPPXWorker**)calloc((size_t)num_workers, sizeof(SNEPPXWorker*));
137 if (!sched->workers) { free(sched->global_queue); sched->global_queue = NULL; return -1; }
138 for (int i = 0; i < num_workers; i++) {
139 sched->workers[i] = (SNEPPXWorker*)calloc(1, sizeof(SNEPPXWorker));
140 if (!sched->workers[i]) return -1;
141 SNEPPX_worker_init(sched->workers[i], i);
142 }
143 return 0;
144}

149void SNEPPX_scheduler_destroy(SNEPPXThreadScheduler* sched) {
150 if (!sched) return;
151 sched->shutdown_flag = 1;
152 for (int i = 0; i < sched->num_threads; i++) {

226int SNEPPX_scheduler_submit(SNEPPXThreadScheduler* sched, SNEPPXTaskFn fn, void* arg, int priority) {
227 if (!sched || !fn) return -1;
228 SNEPPXTask task;
229 task.fn = fn; task.arg = arg; task.priority = priority;
230 static uint64_t global_id = 0; task.id = ATOMIC_INC(&global_id);
231 int best_idx = 0;
232 size_t min_load = (size_t)-1;
233 for (int i = 0; i < sched->num_workers; i++) {

293int SNEPPX_worker_push_task(SNEPPXWorker* worker, SNEPPXTask task) {
294 if (!worker) return -1;
295 size_t tail = worker->deque_tail;
296 if (tail - worker->deque_head >= worker->deque_capacity) {
297 size_t new_cap = worker->deque_capacity * 2;
298 SNEPPXTask* new_deque = (SNEPPXTask*)calloc(new_cap, sizeof(SNEPPXTask));
299 if (!new_deque) return -1;
300 size_t count = worker->deque_tail - worker->deque_head;
301 for (size_t i = 0; i < count; i++)
302 new_deque[i] = worker->deque[(worker->deque_head + i) % worker->deque_capacity];
303 free(worker->deque);
304 worker->deque = new_deque;
305 worker->deque_capacity = new_cap;
306 worker->deque_head = 0;
307 worker->deque_tail = count;
308 tail = count;
309 }
310 worker->deque[tail % worker->deque_capacity] = task;
311 worker->deque_tail++;
312 return 0;
313}
