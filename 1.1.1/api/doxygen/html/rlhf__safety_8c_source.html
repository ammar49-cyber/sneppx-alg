62static double rlhf_detect_harm(const char *text, size_t len) {
63 double score = 0.0;
64 const char *harm_patterns[] = {
65 "bomb", "explosive", "weapon", "kill", "murder", "suicide",
66 "terrorist", "hack", "malware", "ransomware", "virus",
67 "child abuse", "grooming", "exploit child",
68 "nuclear", "biological weapon", "chemical weapon",
69 "methods to", "steps to", "instructions for",

72 "hate speech", "racial slurs", "discriminat",
73 NULL
74 };
75 for (int i = 0; harm_patterns[i]; i++) {
76 if (strstr(text, harm_patterns[i])) score += 0.15;

151int SNEPPX_rlhf_score(const char *input, const char *output, rlhf_score_t *score) {
152 if (!input || !output || !score) return -1;
153 memset(score, 0, sizeof(rlhf_score_t));
154 score->harmfulness = rlhf_detect_harm(output, strlen(output));
155 score->bias = rlhf_detect_bias(output, strlen(output));
156 score->factuality = rlhf_check_factual(output, strlen(output));
157 score->helpfulness = 1.0 - score->factuality * 0.3;
158 score->honesty = 1.0 - score->factuality * 0.2;
159 score->overall = (score->helpfulness * rlhf_config.helpfulness_weight) +
160 ((1.0 - score->harmfulness) * rlhf_config.harmlessness_weight) +
161 (score->honesty * rlhf_config.honesty_weight);
162 score->should_refuse = (score->harmfulness > rlhf_config.harm_threshold ||
163 score->bias > rlhf_config.bias_threshold);
164 if (score->should_refuse && rlhf_config.enable_refusal) total_refusals++;
165 return 0;
166}

176int SNEPPX_rlhf_check_policies(const char *text, rlhf_violation_t *violations, int max_violations) {
177 if (!text || !violations) return -1;
178 int found = 0;
179 for (int i = 0; i < policy_count && found < max_violations; i++) {
180 if (!policies[i].active) continue;
181 double score = 0;
182 switch (policies[i].rule_type) {
183 case RLHF_RULE_HARM: score = rlhf_detect_harm(text, strlen(text)); break;
184 case RLHF_RULE_BIAS: score = rlhf_detect_bias(text, strlen(text)); break;
185 case RLHF_RULE_FACTUAL: score = rlhf_check_factual(text, strlen(text)); break;
186 case RLHF_RULE_CUSTOM: score = rlhf_detect_harm(text, strlen(text)) * 0.5; break;
187 }
188 if (score > policies[i].threshold) {
189 strncpy(violations[found].policy_name, policies[i].name, 63);
190 violations[found].score = score;
191 violations[found].threshold = policies[i].threshold;
192 violations[found].severity = (score > 0.8) ? RLHF_SEV_CRITICAL :
193 (score > 0.6) ? RLHF_SEV_HIGH : RLHF_SEV_MEDIUM;
194 policies[i].violations++;
195 found++;
196 }
197 }
198 return found;
199}

231int SNEPPX_rlhf_generate_critique(const char *output, char *critique, size_t critique_len) {
232 if (!output || !critique) return -1;
233 double harm = rlhf_detect_harm(output, strlen(output));

240 if (bias > 0.3) pos += snprintf(critique + pos, critique_len - pos,

242 if (factual > 0.3) pos += snprintf(critique + pos, critique_len - pos,
243 "- May contain unverified claims (score: %.2f)\n", factual);
244 if (harm <= 0.3 && bias <= 0.3 && factual <= 0.3)
245 pos += snprintf(critique + pos, critique_len - pos, "- Output appears safe and factual.\n");
