36int SNEPPX_tensor_strided_copy(void* dst, const void* src,
37 const size_t* dst_strides, const size_t* src_strides,
38 const size_t* shape, size_t ndim, size_t elem_size) {
39 if (!dst || !src || !shape || ndim == 0) return -1;
40 if (ndim == 1) { memcpy(dst, src, shape[0] * elem_size); return 0; }
41 size_t* src_idx = (size_t*)calloc(ndim, sizeof(size_t));
42 size_t* dst_idx = (size_t*)calloc(ndim, sizeof(size_t));
43 if (!src_idx || !dst_idx) { free(src_idx); free(dst_idx); return -1; }
44 size_t total = 1;
45 for (size_t i = 0; i < ndim; i++) total *= shape[i];
46 for (size_t flat = 0; flat < total; flat++) {
47 size_t soff = 0, doff = 0, tmp = flat;
48 for (size_t d = ndim; d > 0; d--) {
49 size_t idx = tmp % shape[d - 1];
50 tmp /= shape[d - 1];
51 soff += idx * src_strides[d - 1];
52 doff += idx * dst_strides[d - 1];
53 }
54 memcpy((unsigned char*)dst + doff * elem_size,
55 (const unsigned char*)src + soff * elem_size, elem_size);
56 }
57 free(src_idx); free(dst_idx);
58 return 0;
59}

71int SNEPPX_tensor_broadcast_strides(const size_t* src_shape, size_t src_ndim,
72 const size_t* dst_shape, size_t dst_ndim,
73 size_t* out_strides) {
74 if (!src_shape || !dst_shape || !out_strides) return -1;
75 if (dst_ndim < src_ndim) return -1;
76 size_t diff = dst_ndim - src_ndim;
77 size_t stride = 1;
78 for (size_t d = dst_ndim; d > 0; d--) {
79 size_t s_dim = (d - 1 < diff) ? 1 : src_shape[d - 1 - diff];
80 size_t d_dim = dst_shape[d - 1];
81 if (s_dim != d_dim && s_dim != 1) return -1;
82 out_strides[d - 1] = (s_dim == 1) ? 0 : stride;
83 stride *= d_dim;
84 }
85 return 0;
86}

253int SNEPPX_tensor_convert_dtype(void* dst, int dst_dtype, const void* src, int src_dtype, size_t num_elements) {
254 if (!dst || !src || num_elements == 0) return 0;
255 if (src_dtype == dst_dtype) { memcpy(dst, src, num_elements * (size_t)dtype_size(src_dtype)); return 0; }
256 for (size_t i = 0; i < num_elements; i++) {
257 float val;
258 switch (src_dtype) {
259 case 0: val = ((const float*)src)[i]; break;
260 case 1: val = ((const signed char*)src)[i]; break;
261 case 2: val = ((const unsigned char*)src)[i]; break;
262 case 3: val = ((const short*)src)[i]; break;
263 case 4: val = ((const unsigned short*)src)[i]; break;
264 case 5: val = (float)((const double*)src)[i]; break;
265 case 7: val = ((const unsigned char*)src)[i] ? 1.0f : 0.0f; break;
266 default: val = ((const float*)src)[i]; break;
267 }
268 switch (dst_dtype) {
269 case 0: ((float*)dst)[i] = val; break;
270 case 1: ((signed char*)dst)[i] = (signed char)val; break;
271 case 2: ((unsigned char*)dst)[i] = (unsigned char)val; break;
272 case 3: ((short*)dst)[i] = (short)val; break;
273 case 4: ((unsigned short*)dst)[i] = (unsigned short)val; break;
274 case 5: ((double*)dst)[i] = (double)val; break;
275 case 7: ((unsigned char*)dst)[i] = val != 0.0f; break;
276 default: ((float*)dst)[i] = val; break;
277 }
278 }
279 return 0;

354int SNEPPX_tensor_parallel_for(SNEPPXTensorKernel1D kernel, void* ctx,
355 size_t total_work, size_t min_grain) {
356 if (!kernel || total_work == 0) return 0;
357 if (total_work <= min_grain) { kernel(ctx, 0, total_work); return 0; }
358 size_t num_chunks = (total_work + min_grain - 1) / min_grain;
359 if (num_chunks > 8) num_chunks = 8;
360 size_t chunk_size = (total_work + num_chunks - 1) / num_chunks;
361 for (size_t c = 0; c < num_chunks; c++) {
362 size_t begin = c * chunk_size;
363 size_t end = begin + chunk_size;
364 if (end > total_work) end = total_work;
365 kernel(ctx, begin, end);
366 }
367 return 0;
368}
