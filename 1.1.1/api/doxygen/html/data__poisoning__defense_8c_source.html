71int SNEPPX_poison_detector_train(SNEPPXPoisonDetector* pd,
72 const double* samples, int sample_count) {
73 if (!pd || !samples || sample_count < 2) return -1;
74 int f = pd->feature_count;
75 for (int j = 0; j < f; j++) {
76 double sum = 0.0, sum_sq = 0.0;
77 for (int i = 0; i < sample_count; i++) {
78 double v = samples[i * f + j];
79 sum += v;
80 sum_sq += v * v;
81 }
82 pd->feature_means[j] = sum / sample_count;
83 double var = (sum_sq / sample_count) - (pd->feature_means[j] * pd->feature_means[j]);
84 pd->feature_stds[j] = (var > 0) ? sqrt(var) : 1.0;
85 }
86 pd->trained = 1;
87 poison_sample_count = sample_count;
88 if (ensemble_count < SNEPPX_POISON_ENSEMBLE_K) {
89 for (int j = 0; j < f; j++) {
90 ensemble_means[ensemble_count][j] = pd->feature_means[j];
91 ensemble_stds[ensemble_count][j] = pd->feature_stds[j];
92 }
93 ensemble_trained[ensemble_count] = 1;
94 ensemble_count++;
95 }
96 return 0;
97}

144int SNEPPX_poison_detector_update_incremental(SNEPPXPoisonDetector* pd,
145 const double* sample) {
146 if (!pd || !sample || !pd->trained) return -1;
147 int f = pd->feature_count;
148 double n = (double)poison_sample_count;
149 for (int j = 0; j < f; j++) {
150 double old_mean = pd->feature_means[j];
151 pd->feature_means[j] = (old_mean * n + sample[j]) / (n + 1.0);
152 double old_var = pd->feature_stds[j] * pd->feature_stds[j];
153 double new_var = (n * (old_var + old_mean * old_mean) + sample[j] * sample[j]) / (n + 1.0) - pd->feature_means[j] * pd->feature_means[j];
154 pd->feature_stds[j] = (new_var > 0) ? sqrt(new_var) : 1.0;
155 }
156 poison_sample_count++;
157 return 0;
158}

222int SNEPPX_poison_detector_incremental_update(SNEPPXPoisonDetector* pd, const double* sample, double learning_rate) {
223 if (!pd || !sample || !pd->trained) return -1;
224 if (learning_rate <= 0.0 || learning_rate > 1.0) learning_rate = 0.1;
225 int f = pd->feature_count;
226 for (int j = 0; j < f; j++) {
227 double diff = sample[j] - pd->feature_means[j];
228 pd->feature_means[j] += learning_rate * diff;
229 double var = pd->feature_stds[j] * pd->feature_stds[j];
230 var += learning_rate * (diff * diff - var);
231 pd->feature_stds[j] = (var > 0) ? sqrt(var) : 1.0;
232 }
233 poison_sample_count++;
234 return 0;
