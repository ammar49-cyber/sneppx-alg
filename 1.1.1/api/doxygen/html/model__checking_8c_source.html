90SNEPPXModelCheckResult SNEPPX_model_check(SNEPPXFormalModel* model) {
91 SNEPPXModelCheckResult result;
92 memset(&result, 0, sizeof(result));
93 if (!model) { result.property_satisfied = 0; return result; }
94
95 result.total_states = model->state_count;
96 int visited[SNEPPX_MODEL_MAX_STATES] = {0};
97 int stack[SNEPPX_MODEL_MAX_STATES];
98 int stack_top = 0;
99 stack[stack_top++] = model->initial_state;
100 visited[model->initial_state] = 1;
101
102 while (stack_top > 0) {
103 int cur = stack[--stack_top];
104 result.reachable_states++;
105 for (int i = 0; i < model->state_count; i++) {
106 if (model->states[i].state_id != (uint32_t)cur) continue;
107 if (model->states[i].is_error) result.error_states++;
108 if (model->states[i].next_count == 0) result.deadlock_states++;
109 for (int j = 0; j < model->states[i].next_count; j++) {
110 uint32_t next = model->states[i].next_states[j];
111 int found = 0;
112 for (int k = 0; k < model->state_count; k++) {
113 if (model->states[k].state_id == next) { found = 1; break; }
114 }
115 if (found && !visited[next]) {
116 visited[next] = 1;
117 if (stack_top < SNEPPX_MODEL_MAX_STATES) stack[stack_top++] = next;
118 }
119 }
120 }
121 }
122
123 result.property_satisfied = (result.error_states == 0) ? 1 : 0;
124 snprintf(result.counterexample, sizeof(result.counterexample), "%s", result.property_satisfied ? "ok" : "error state reachable");
125 return result;
126}

262int SNEPPX_model_find_trace(SNEPPXFormalModel* model, uint32_t from, uint32_t to, uint32_t* trace, int max_len) {
263 if (!model || !trace || max_len <= 0) return -1;
264 int visited[SNEPPX_MODEL_MAX_STATES] = {0};
265 int parent[SNEPPX_MODEL_MAX_STATES];
266 for (int i = 0; i < SNEPPX_MODEL_MAX_STATES; i++) parent[i] = -1;
267 int queue[SNEPPX_MODEL_MAX_STATES];
268 int qh = 0, qt = 0;
269 queue[qt++] = from;
270 visited[from] = 1;
271 while (qh < qt) {
272 int cur = queue[qh++];
273 if ((uint32_t)cur == to) {
274 int len = 0, node = cur;
275 while (node != -1 && len < max_len) {
276 trace[len++] = (uint32_t)node;
277 node = parent[node];
278 }
279 for (int i = 0; i < len / 2; i++) {
280 uint32_t t = trace[i]; trace[i] = trace[len - 1 - i]; trace[len - 1 - i] = t;
281 }
282 return len;
283 }
284 for (int i = 0; i < model->state_count; i++) {
285 if (model->states[i].state_id != (uint32_t)cur) continue;
286 for (int j = 0; j < model->states[i].next_count; j++) {
287 uint32_t next = model->states[i].next_states[j];
288 if (!visited[next]) {
289 visited[next] = 1;
290 parent[next] = cur;
291 if (qt < SNEPPX_MODEL_MAX_STATES) queue[qt++] = next;
292 }
293 }
294 }
295 }
296 return -1;
297}

324int SNEPPX_model_has_cycle(SNEPPXFormalModel* model) {
325 if (!model) return 0;
326 int visited[SNEPPX_MODEL_MAX_STATES] = {0};
327 int rec_stack[SNEPPX_MODEL_MAX_STATES] = {0};
328 int stack[SNEPPX_MODEL_MAX_STATES];
329 int sp = 0;
330 stack[sp++] = model->initial_state;
331 while (sp > 0) {
332 int cur = stack[--sp];
333 if (!visited[cur]) {
334 visited[cur] = 1;
335 rec_stack[cur] = 1;
336 }
337 for (int i = 0; i < model->state_count; i++) {
338 if (model->states[i].state_id != (uint32_t)cur) continue;
339 for (int j = 0; j < model->states[i].next_count; j++) {
340 uint32_t next = model->states[i].next_states[j];
341 if (!visited[next]) {
342 if (sp < SNEPPX_MODEL_MAX_STATES) stack[sp++] = next;
343 } else if (rec_stack[next]) {
344 return 1;
345 }
346 }
347 }
348 rec_stack[cur] = 0;
349 }
350 return 0;
