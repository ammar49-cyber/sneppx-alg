45static int compare_priority_desc(const void* a, const void* b) {
46 const SNEPPXEpisodicExperience* ea = (const SNEPPXEpisodicExperience*)a;
47 const SNEPPXEpisodicExperience* eb = (const SNEPPXEpisodicExperience*)b;

87 const SNEPPXTensor* action, float reward,
88 const SNEPPXTensor* next_state) {
89 if (!buf || !state) return -1;
90 size_t idx = buf->count < buf->capacity ? buf->count : buf->head;
91 SNEPPXEpisodicExperience* exp = &buf->experiences[idx];
92 if (exp->valid) {
93 if (exp->state) SNEPPX_tensor_destroy(exp->state);
94 if (exp->action) SNEPPX_tensor_destroy(exp->action);
95 if (exp->next_state) SNEPPX_tensor_destroy(exp->next_state);
96 }
97 exp->state = SNEPPX_tensor_copy(state);
98 if (!exp->state) return -1;
99 if (action) {
100 exp->action = SNEPPX_tensor_copy(action);
101 if (!exp->action) { SNEPPX_tensor_destroy(exp->state); exp->state = NULL; return -1; }
102 } else {
103 exp->action = NULL;
104 }
105 exp->reward = reward;
106 if (next_state) {
107 exp->next_state = SNEPPX_tensor_copy(next_state);
108 if (!exp->next_state) {
109 SNEPPX_tensor_destroy(exp->state); exp->state = NULL;
110 if (exp->action) { SNEPPX_tensor_destroy(exp->action); exp->action = NULL; }
111 return -1;
112 }
113 } else {
114 exp->next_state = NULL;
115 }
116 exp->timestamp = buf->next_timestamp++;
117 exp->priority = reward > 0 ? reward : 0.01f;
118 exp->valid = 1;
119 if (buf->count < buf->capacity) {
120 buf->count++;
121 } else {
122 buf->head = (buf->head + 1) % buf->capacity;
123 }
124 if (exp->timestamp - buf->last_consolidation >= buf->consolidation_interval) {
125 sneppx_episodic_consolidate(buf);
126 }
127 return 0;

131 uint64_t time_end, SNEPPXTensor** output) {

146 size_t shape[2] = {match_count, state_dim};

148 if (!result) return -1;

150 float* data = (float*)result->data;
151 for (size_t i = 0; i < buf->capacity; i++) {
152 if (buf->experiences[i].valid &&
153 buf->experiences[i].timestamp >= time_start &&
154 buf->experiences[i].timestamp <= time_end) {
155 if (buf->experiences[i].state) {
156 memcpy(data + row * state_dim, buf->experiences[i].state->data,
157 state_dim * sizeof(float));
158 }
159 row++;
160 }
161 }

166int sneppx_episodic_sample(SNEPPXEpisodicBuffer* buf, size_t batch_size,
167 SNEPPXTensor** states, SNEPPXTensor** actions,
168 float* rewards, SNEPPXTensor** next_states) {
169 if (!buf || !states || !rewards || batch_size == 0) return -1;
170 size_t available = 0;
171 for (size_t i = 0; i < buf->capacity; i++) {
172 if (buf->experiences[i].valid) available++;
173 }
174 if (available == 0) return -1;
175 size_t actual_batch = batch_size < available ? batch_size : available;
176 SNEPPXEpisodicExperience* sorted = SNEPPX_malloc(available * sizeof(SNEPPXEpisodicExperience), 64);
177 if (!sorted) return -1;
178 size_t idx = 0;
179 for (size_t i = 0; i < buf->capacity; i++) {
180 if (buf->experiences[i].valid) {
181 sorted[idx++] = buf->experiences[i];
182 }
183 }
184 qsort(sorted, available, sizeof(SNEPPXEpisodicExperience), compare_priority_desc);
185 float total_priority = 0.0f;
186 for (size_t i = 0; i < available; i++) total_priority += sorted[i].priority;
187 if (total_priority < 1e-10f) total_priority = 1.0f;
188 for (size_t i = 0; i < actual_batch; i++) {
189 float r = frand(buf->seed_ptr) * total_priority;
190 float cum = 0.0f;
191 size_t chosen = 0;
192 for (size_t j = 0; j < available; j++) {
193 cum += sorted[j].priority;
194 if (r <= cum) { chosen = j; break; }
195 }
196 SNEPPXEpisodicExperience* exp = &sorted[chosen];
197 if (exp->state) {
198 states[i] = SNEPPX_tensor_copy(exp->state);
199 if (!states[i]) {
200 for (size_t k = 0; k < i; k++) SNEPPX_tensor_destroy(states[k]);
201 SNEPPX_free(sorted, available * sizeof(SNEPPXEpisodicExperience));
202 return -1;
203 }
204 } else {
205 states[i] = NULL;
206 }
207 if (actions && exp->action) {
208 actions[i] = SNEPPX_tensor_copy(exp->action);
209 } else if (actions) {
210 actions[i] = NULL;
211 }
212 rewards[i] = exp->reward;
213 if (next_states && exp->next_state) {
214 next_states[i] = SNEPPX_tensor_copy(exp->next_state);
215 } else if (next_states) {
216 next_states[i] = NULL;
217 }
218 sorted[chosen].priority *= 0.9f;
219 }
220 SNEPPX_free(sorted, available * sizeof(SNEPPXEpisodicExperience));
221 return (int)actual_batch;
222}

224int sneppx_episodic_consolidate(SNEPPXEpisodicBuffer* buf) {
225 if (!buf) return -1;
226 buf->last_consolidation = buf->next_timestamp;
227 if (buf->count < 2) return 0;
228 float mean_reward = 0.0f;
229 float max_reward = -1e10f;
230 float min_reward = 1e10f;
231 size_t valid_count = 0;
232 for (size_t i = 0; i < buf->capacity; i++) {
233 if (buf->experiences[i].valid) {
234 mean_reward += buf->experiences[i].reward;
235 if (buf->experiences[i].reward > max_reward) max_reward = buf->experiences[i].reward;
236 if (buf->experiences[i].reward < min_reward) min_reward = buf->experiences[i].reward;
237 valid_count++;
238 }
239 }
240 if (valid_count == 0) return 0;
241 mean_reward /= (float)valid_count;
242 for (size_t i = 0; i < buf->capacity; i++) {
243 if (buf->experiences[i].valid) {
244 float age = (float)(buf->next_timestamp - buf->experiences[i].timestamp);
245 if (age > 1000.0f && buf->experiences[i].reward < mean_reward * 0.5f) {
246 buf->experiences[i].priority *= 0.5f;
247 }
248 }
249 }
250 return 0;
