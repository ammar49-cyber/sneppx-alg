109int SNEPPX_sample_from_logits(const float* logits, size_t vocab_size, SNEPPXGenerationConfig* cfg) {
110 if (!logits || vocab_size == 0 || !cfg) return -1;
111
112 size_t n = vocab_size;
113 float* probs = (float*)SNEPPX_malloc(n * sizeof(float), 64);
114 if (!probs) return SNEPPX_argmax(logits, n);
115
116 int* indices = (int*)SNEPPX_malloc(n * sizeof(int), 64);
117 if (!indices) { SNEPPX_free(probs, n * sizeof(float)); return SNEPPX_argmax(logits, n); }
118
119 for (size_t i = 0; i < n; i++) {
120 probs[i] = logits[i] / (cfg->temperature > 0.0f ? cfg->temperature : 1.0f);
121 }
122
123 float max_logit = probs[0];
124 for (size_t i = 1; i < n; i++) if (probs[i] > max_logit) max_logit = probs[i];
125 float sum = 0.0f;
126 for (size_t i = 0; i < n; i++) {
127 probs[i] = expf(probs[i] - max_logit);
128 sum += probs[i];
129 }
130 float inv_sum = 1.0f / (sum + 1e-10f);
131 for (size_t i = 0; i < n; i++) probs[i] *= inv_sum;
132
133 if (cfg->top_k > 0 && cfg->top_k < (int)n) {
134 for (size_t i = 0; i < n; i++) indices[i] = (int)i;
135 for (int i = 0; i < (int)n - 1; i++) {
136 for (int j = i + 1; j < (int)n; j++) {
137 if (probs[indices[i]] < probs[indices[j]]) {
138 int t = indices[i]; indices[i] = indices[j]; indices[j] = t;
139 }
140 }
141 }
142 float kth_prob = probs[indices[cfg->top_k - 1]];
143 for (size_t i = 0; i < n; i++) {
144 if (probs[i] < kth_prob) probs[i] = 0.0f;
145 }
146 float new_sum = 0.0f;
147 for (size_t i = 0; i < n; i++) new_sum += probs[i];
148 inv_sum = 1.0f / (new_sum + 1e-10f);
149 for (size_t i = 0; i < n; i++) probs[i] *= inv_sum;
150 }
151
152 if (cfg->top_p > 0.0f && cfg->top_p < 1.0f) {
153 for (size_t i = 0; i < n; i++) indices[i] = (int)i;
154 for (size_t i = 0; i < n - 1; i++) {
155 for (size_t j = i + 1; j < n; j++) {
156 if (probs[indices[i]] < probs[indices[j]]) {
157 int t = indices[i]; indices[i] = indices[j]; indices[j] = t;
158 }
159 }
160 }
161 float cum = 0.0f;
162 for (size_t i = 0; i < n; i++) {
163 if (cum >= cfg->top_p) probs[indices[i]] = 0.0f;
164 cum += probs[indices[i]];
165 }
166 float new_sum = 0.0f;
167 for (size_t i = 0; i < n; i++) new_sum += probs[i];
168 inv_sum = 1.0f / (new_sum + 1e-10f);
169 for (size_t i = 0; i < n; i++) probs[i] *= inv_sum;
170 }
171
172 float r = (float)rand() / ((float)RAND_MAX + 1.0f);
173 float cumulative = 0.0f;
174 int result = 0;
175 for (size_t i = 0; i < n; i++) {
176 cumulative += probs[i];
177 if (r < cumulative) { result = (int)i; break; }
178 }
179
180 SNEPPX_free(probs, n * sizeof(float));
181 SNEPPX_free(indices, n * sizeof(int));
182 return result;
183}

261 SNEPPX_tensor_destroy(attn_out);

265 SNEPPX_tensor_destroy(flat);
266
267 if (!logits) break;
268 float* ld = (float*)logits->data;
269 size_t last_idx = logits->size / vocab_size - 1;
270 float* last_logits = ld + last_idx * vocab_size;
271
272 int next_id = SNEPPX_sample_from_logits(last_logits, vocab_size, cfg);
273 SNEPPX_tensor_destroy(logits);
274
275 if (next_id < 0) break;
276 output_ids[generated] = next_id;
277 generated++;
278
279 if (cfg->eos_token_id >= 0 && next_id == cfg->eos_token_id) break;
280 }
281
282 SNEPPX_kv_cache_destroy(cache);
283 return (int)generated;
284}
