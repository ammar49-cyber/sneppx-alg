134SNEPPXFilterResult SNEPPX_prompt_filter_scan(SNEPPXPromptFilter* pf,
135 const char* prompt, size_t len) {
136 if (!pf || !pf->enabled || !prompt) return SNEPPX_FILTER_CLEAN;
137 total_scans++;
138 char lower[SNEPPX_PATTERN_MAX_LEN];
139 size_t plen = (len < SNEPPX_PATTERN_MAX_LEN - 1) ? len : SNEPPX_PATTERN_MAX_LEN - 1;
140 for (size_t i = 0; i < plen; i++) lower[i] = (char)tolower((unsigned char)prompt[i]);
141 lower[plen] = '\0';
142
143 SNEPPXFilterResult highest = SNEPPX_FILTER_CLEAN;
144 int highest_prio = -1;
145 for (int i = 0; i < pf->pattern_count; i++) {
146 if (!pf->patterns[i].is_active) continue;
147 if (strstr(lower, pf->patterns[i].pattern)) {
148 if (pattern_priorities[i] > highest_prio) {
149 highest_prio = pattern_priorities[i];
150 highest = pf->patterns[i].classification;
151 }
152 }
153 }
154 if (highest != SNEPPX_FILTER_CLEAN) total_blocks++;

224int SNEPPX_prompt_filter_match_wildcard(const char* text, const char* pattern) {
225 if (!text || !pattern) return 0;
226 if (pattern[0] == '\0') return text[0] == '\0';
227 if (pattern[0] == '*') {
228 for (int i = 0; text[i]; i++)
229 if (SNEPPX_prompt_filter_match_wildcard(text + i, pattern + 1)) return 1;
230 return SNEPPX_prompt_filter_match_wildcard(text, pattern + 1);
231 }
232 if (pattern[0] == '?' && text[0]) {
233 return SNEPPX_prompt_filter_match_wildcard(text + 1, pattern + 1);
234 }
235 if (pattern[0] == text[0]) {
236 return SNEPPX_prompt_filter_match_wildcard(text + 1, pattern + 1);
237 }
238 return 0;
239}

429static void to_lower_utf8(char* dst, const char* src, size_t max_len) {
430 if (!dst || !src) return;
431 size_t i = 0, j = 0;
432 while (src[i] && j < max_len - 1) {
433 int clen = utf8_char_len((unsigned char)src[i]);
434 if (clen == 1) {
435 dst[j++] = (char)tolower((unsigned char)src[i++]);
436 } else {
437 for (int k = 0; k < clen && src[i + k] && j < max_len - 1; k++)
438 dst[j++] = src[i + k];
439 i += clen;
440 }
441 }
442 dst[j] = '\0';
443}

585static size_t hex_decode_local(const char* in, size_t in_len, char* out) {
586 size_t out_pos = 0;
587 for (size_t i = 0; i + 1 < in_len; i += 2) {
588 unsigned char c1 = (unsigned char)in[i], c2 = (unsigned char)in[i + 1];
589 unsigned char v1 = (c1 >= 'a' ? c1 - 'a' + 10 : (c1 >= 'A' ? c1 - 'A' + 10 : c1 - '0'));
590 unsigned char v2 = (c2 >= 'a' ? c2 - 'a' + 10 : (c2 >= 'A' ? c2 - 'A' + 10 : c2 - '0'));
591 out[out_pos++] = (char)((v1 << 4) | v2);
592 }
593 return out_pos;
594}
595static const char* b64_alphabet_local = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
596static int b64_rev_local[256];
597static int b64_init_local = 0;
598static void b64_build_rev_local(void) {
599 if (b64_init_local) return; b64_init_local = 1;
600 for (int i = 0; i < 256; i++) b64_rev_local[i] = -1;
601 for (int i = 0; i < 64; i++) b64_rev_local[(int)b64_alphabet_local[i]] = i;
602}
603static size_t b64_decode_local(const char* in, size_t in_len, char* out) {
604 b64_build_rev_local();
605 size_t out_pos = 0; int val = 0, valb = -8;
606 for (size_t i = 0; i < in_len; i++) {
607 unsigned char c = (unsigned char)in[i];
608 if (c == '=') break;
609 int d = b64_rev_local[c];
610 if (d == -1) continue;
611 val = (val << 6) | d; valb += 6;
612 if (valb >= 0) { out[out_pos++] = (char)((val >> valb) & 0xFF); valb -= 8; }
613 }
614 return out_pos;
615}
