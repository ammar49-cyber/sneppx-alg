27static float frand(unsigned int* seed) {
28 *seed = *seed * 1103515245U + 12345U;
29 return (float)((*seed >> 16) & 0x7FFF) / 32767.0f;

32static float kaiming_init(unsigned int* seed, size_t fan) {
33 float std = sqrtf(2.0f / (float)fan);
34 float u1 = frand(seed), u2 = frand(seed);
35 return std * sqrtf(-2.0f * logf(u1 + 1e-10f)) * cosf(6.2831853f * u2);
36}

142SNEPPXTensor* SNEPPX_rope_precompute(size_t seq_len, size_t head_dim, float base) {
143 size_t shape[] = {seq_len, head_dim};
144 SNEPPXTensor* t = SNEPPX_tensor_create(shape, 2, SNEPPX_FLOAT32);
145 if (!t) return NULL;
146 float* d = (float*)t->data;
147 for (size_t pos = 0; pos < seq_len; pos++) {
148 for (size_t j = 0; j < head_dim / 2; j++) {
149 float theta = (float)pos / powf(base, 2.0f * j / (float)head_dim);
150 d[pos * head_dim + j] = cosf(theta);
151 d[pos * head_dim + j + head_dim / 2] = sinf(theta);
152 }
153 }
154 return t;
155}

165void SNEPPX_rope_apply(SNEPPXTensor* q, SNEPPXTensor* k, const SNEPPXTensor* cos, const SNEPPXTensor* sin,
166 size_t offset) {
167 if (!q || !k || !cos) return;
168 size_t d = q->shape[q->ndim - 1];
169 size_t half = d / 2;
170 if (half == 0) return;
171 size_t total = q->size;
172 float* qd = (float*)q->data;
173 float* kd = (float*)k->data;
174 const float* cd = (const float*)cos->data;
175 size_t cd_last = cos->shape[cos->ndim - 1];
176 (void)sin;
177 /* SNEPPX_rope_precompute returns a combined [S, d] table: first half =
178 cos, second half = sin. Every caller passes this combined table as
179 `cos` (and `sin` is NULL or the same pointer). Process each (x_i,
180 x_{i+half}) pair together from the ORIGINAL values so the in-place
181 rotation is not contaminated by earlier writes. */
182 /* The [B,H,S,hd] layout means base/d is the flat [B*H*S] index, not the
183 sequence position. pos must be (base/d)%S so multi-head tensors with
184 S=1 do not index past the [S,hd] cos table. */
185 size_t seq_len = (q->ndim >= 2) ? q->shape[q->ndim - 2] : 1;
186 for (size_t base = 0; base < total; base += d) {
187 size_t pos = ((base / d) % seq_len) + offset;
188 for (size_t j = 0; j < half; j++) {
189 float cosv = cd[pos * cd_last + j];
190 float sinv = cd[pos * cd_last + half + j];
191 float qv = qd[base + j]; float qvp = qd[base + j + half];
192 float kv = kd[base + j]; float kvp = kd[base + j + half];
193 qd[base + j] = qv * cosv - qvp * sinv;
194 qd[base + j + half] = qv * sinv + qvp * cosv;
195 kd[base + j] = kv * cosv - kvp * sinv;
196 kd[base + j + half] = kv * sinv + kvp * cosv;
197 }
198 }
199}

201/* Correct RoPE rotation using original values only (no in-place contamination).
202 x shape [..., d], cos_table shape [S, d]. First d/2 entries of each row: cos, second d/2: sin.
203 Returns new tensor same shape as x. */
211SNEPPXTensor* SNEPPX_tensor_rope(const SNEPPXTensor* x, const SNEPPXTensor* cos_table) {
212 if (!x || !cos_table) return NULL;
213 size_t d = x->shape[x->ndim - 1];
214 size_t half = d / 2;
215 size_t total = x->size;
216 SNEPPXTensor* out = SNEPPX_tensor_copy(x);
217 if (!out) return NULL;
218 float* xd = (float*)x->data;
219 float* od = (float*)out->data;
220 float* cd = (float*)cos_table->data;
221 size_t cd_last = cos_table->shape[cos_table->ndim - 1];
222 for (size_t i = 0; i < total; i++) {
223 size_t pos = i / d;
224 size_t j = i % d;
225 size_t pair = j % half;
226 float cos_val = cd[pos * cd_last + pair];
227 float sin_val = cd[pos * cd_last + half + pair];
228 size_t base = pos * d;
229 size_t partner = (j < half) ? (base + j + half) : (base + j - half);
230 float x_i = xd[i];
231 float x_p = xd[partner];
232 if (j < half)
233 od[i] = x_i * cos_val - x_p * sin_val;
234 else
235 od[i] = x_p * sin_val + x_i * cos_val;
236 }
237 return out;
238}

251SNEPPXTensor* SNEPPX_batched_matmul(const SNEPPXTensor* a, const SNEPPXTensor* b,
252 int transpose_b, int transpose_a) {
253 (void)transpose_a;
254 if (!a || !b || a->ndim != 4 || b->ndim != 4) return NULL;
255 size_t B = a->shape[0], H = a->shape[1], S = a->shape[2];
256 size_t D, S2;
257 if (transpose_b) {
258 /* a @ b^T: a=[B,H,S,D], b=[B,H,S2,D] → out=[B,H,S,S2] */
259 D = a->shape[3];
260 if (b->shape[3] != D) return NULL;
261 S2 = b->shape[2];
262 } else {
263 /* a @ b: a=[B,H,S,D1], b=[B,H,D1,D2] → out=[B,H,S,D2] */
264 D = a->shape[3];
265 if (b->shape[2] != D) return NULL;
266 S2 = b->shape[3];
267 }
268 size_t out_shape[] = {B, H, S, S2};
269 SNEPPXTensor* out = SNEPPX_tensor_zeros(out_shape, 4, SNEPPX_FLOAT32);
270 if (!out) return NULL;
271 float* ad = (float*)a->data;
272 float* bd = (float*)b->data;
273 float* od = (float*)out->data;
274 if (transpose_b) {
275 for (size_t b_ = 0; b_ < B; b_++) {
276 for (size_t h = 0; h < H; h++) {
277 for (size_t s = 0; s < S; s++) {
278 for (size_t s2 = 0; s2 < S2; s2++) {
279 float sum = 0.0f;
280 for (size_t d = 0; d < D; d++) {
281 sum += ad[((b_ * H + h) * S + s) * D + d]
282 * bd[((b_ * H + h) * S2 + s2) * D + d];
283 }
284 od[((b_ * H + h) * S + s) * S2 + s2] = sum;
285 }
286 }
287 }
288 }
289 } else {
290 for (size_t b_ = 0; b_ < B; b_++) {
291 for (size_t h = 0; h < H; h++) {
292 for (size_t s = 0; s < S; s++) {
293 for (size_t d2 = 0; d2 < S2; d2++) {
294 float sum = 0.0f;
295 for (size_t d = 0; d < D; d++) {
296 sum += ad[((b_ * H + h) * S + s) * D + d]
297 * bd[((b_ * H + h) * D + d) * S2 + d2];
298 }
299 od[((b_ * H + h) * S + s) * S2 + d2] = sum;
300 }
301 }
302 }
303 }
304 }
305 return out;
306}

322/* ========== 3D Matmul Helper ========== */
323
324static SNEPPXTensor* matmul_3d(const SNEPPXTensor* a, const SNEPPXTensor* b) {
325 if (!a || !b) return NULL;
326 /* a: [B, S, D], b: [D, Dk] → result: [B, S, Dk] */
327 size_t B = a->shape[0], S = a->shape[1], D = a->shape[2];
328 size_t Dk = b->shape[1];
329 size_t result_shape[] = {B, S, Dk};
330 SNEPPXTensor* result = SNEPPX_tensor_zeros(result_shape, 3, SNEPPX_FLOAT32);
331 if (!result) return NULL;
332 float* ad = (float*)a->data;
333 float* bd = (float*)b->data;
334 float* rd = (float*)result->data;
335 for (size_t batch = 0; batch < B; batch++) {
336 for (size_t s = 0; s < S; s++) {
337 for (size_t k = 0; k < Dk; k++) {
338 float sum = 0.0f;
339 for (size_t d = 0; d < D; d++)
340 sum += ad[(batch * S + s) * D + d] * bd[d * Dk + k];
341 rd[(batch * S + s) * Dk + k] = sum;
342 }
343 }
344 }
345 return result;
346}

359SNEPPXTensor* SNEPPX_attn_forward(const SNEPPXAttentionWeights* w, const SNEPPXTensor* x,
360 SNEPPXTensor* cos_t, SNEPPXTensor* sin_t) {
361 if (!w || !x) return NULL;
362 /* Accept a 2D [seq_len, d_model] input as a single batch element. */
363 int orig_ndim = x->ndim;
364 SNEPPXTensor* x_work = (SNEPPXTensor*)x;
365 if (x->ndim == 2) {
366 size_t shp3[] = {1, x->shape[0], x->shape[1]};
367 x_work = SNEPPX_tensor_reshape(x, shp3, 3);
368 if (!x_work) return NULL;
369 }
370 size_t B = x_work->shape[0], S = x_work->shape[1], D = w->config.d_model;
371 size_t H = w->config.num_heads, hd = w->config.head_dim;
372
373 /* Q, K, V projections: [B, S, D] → [B, S, H*hd] */
374 SNEPPXTensor* q = matmul_3d(x_work, w->w_q);
375 SNEPPXTensor* k = matmul_3d(x_work, w->w_k);
376 SNEPPXTensor* v = matmul_3d(x_work, w->w_v);
377 if (!q || !k || !v) {
378 SNEPPX_tensor_destroy(q); SNEPPX_tensor_destroy(k); SNEPPX_tensor_destroy(v);
379 return NULL;
380 }
381
382 /* Reshape to [B, H, S, hd] */
383 size_t qkv_shape[] = {B, H, S, hd};
384 SNEPPXTensor* q_r = SNEPPX_tensor_reshape(q, qkv_shape, 4);
385 SNEPPXTensor* k_r = SNEPPX_tensor_reshape(k, qkv_shape, 4);
386 SNEPPXTensor* v_r = SNEPPX_tensor_reshape(v, qkv_shape, 4);
387 SNEPPX_tensor_destroy(q); SNEPPX_tensor_destroy(k); SNEPPX_tensor_destroy(v);
388
389 /* RoPE */
390 if (w->config.use_rope && cos_t && sin_t)
391 SNEPPX_rope_apply(q_r, k_r, cos_t, sin_t, 0);
392
393 /* Attention scores: Q @ K^T, [B, H, S, S] */
394 SNEPPXTensor* scores = SNEPPX_batched_matmul(q_r, k_r, 1, 0);
395 if (!scores) { SNEPPX_tensor_destroy(q_r); SNEPPX_tensor_destroy(k_r); SNEPPX_tensor_destroy(v_r); return NULL; }
396
397 /* Scale */
398 float scale = 1.0f / sqrtf((float)hd);
399 float* sd = (float*)scores->data;
400 for (size_t i = 0; i < scores->size; i++) sd[i] *= scale;
401
402 /* Causal mask */
403 if (w->config.use_causal_mask && S > 1) {
404 SNEPPXTensor* mask = causal_mask(S);
405 if (mask) {
406 float* md = (float*)mask->data;
407 for (size_t b = 0; b < B; b++)
408 for (size_t h = 0; h < H; h++)
409 for (size_t i = 0; i < S; i++)
410 for (size_t j = 0; j < S; j++)
411 sd[((b * H + h) * S + i) * S + j] += md[i * S + j];
412 SNEPPX_tensor_destroy(mask);
413 }
414 }
415
416 /* Softmax over last dim */
417 for (size_t i = 0; i < scores->size; i += S) {
418 float maxv = sd[i];
419 for (size_t j = 1; j < S; j++)
420 if (sd[i + j] > maxv) maxv = sd[i + j];
421 float sum = 0.0f;
422 for (size_t j = 0; j < S; j++) {
423 sd[i + j] = expf(sd[i + j] - maxv);
424 sum += sd[i + j];
425 }
426 float inv = 1.0f / (sum + 1e-10f);
427 for (size_t j = 0; j < S; j++) sd[i + j] *= inv;
428 }
429
430 /* Apply to V: scores @ V, [B, H, S, hd] */
431 SNEPPXTensor* out = SNEPPX_batched_matmul(scores, v_r, 0, 0);
432 SNEPPX_tensor_destroy(scores);
433 if (!out) { SNEPPX_tensor_destroy(q_r); SNEPPX_tensor_destroy(k_r); SNEPPX_tensor_destroy(v_r); return NULL; }
434
435 /* Reshape back to [B, S, D] */
436 size_t out_sh[] = {B, S, D};
437 SNEPPXTensor* out_r = SNEPPX_tensor_reshape(out, out_sh, 3);
438 SNEPPX_tensor_destroy(out);
439
440 /* Output projection: [B, S, D] → [B, S, D] */
441 SNEPPXTensor* result = matmul_3d(out_r, w->w_o);
442 SNEPPX_tensor_destroy(out_r);
443 SNEPPX_tensor_destroy(q_r); SNEPPX_tensor_destroy(k_r); SNEPPX_tensor_destroy(v_r);
444 if (orig_ndim == 2) {
445 size_t shp2[] = {x->shape[0], x->shape[1]};
446 SNEPPXTensor* squeezed = SNEPPX_tensor_reshape(result, shp2, 2);
447 SNEPPX_tensor_destroy(result);
448 if (x_work != x) SNEPPX_tensor_destroy(x_work);
449 return squeezed;
450 }
451 if (x_work != x) SNEPPX_tensor_destroy(x_work);
452 return result;
453}

507SNEPPXTensor* SNEPPX_attn_forward_cached(const SNEPPXAttentionWeights* w, const SNEPPXTensor* x,
508 SNEPPXKVCache* cache, SNEPPXTensor* cos, SNEPPXTensor* sin) {
509 if (!w || !x || !cache) return NULL;
510 size_t B = x->shape[0], S = x->shape[1], D = w->config.d_model;
511 size_t H = w->config.num_heads, hd = w->config.head_dim;
512 size_t offset = cache->seq_len;
513 size_t total_seq = offset + S;
514
515 /* Q, K, V */
516 SNEPPXTensor* q = SNEPPX_tensor_matmul(x, w->w_q);
517 SNEPPXTensor* k = SNEPPX_tensor_matmul(x, w->w_k);
518 SNEPPXTensor* v = SNEPPX_tensor_matmul(x, w->w_v);
519
520 size_t qkv_shape[] = {B, H, S, hd};
521 SNEPPXTensor* q_r = SNEPPX_tensor_reshape(q, qkv_shape, 4);
522 SNEPPXTensor* k_r = SNEPPX_tensor_reshape(k, qkv_shape, 4);
523 SNEPPXTensor* v_r = SNEPPX_tensor_reshape(v, qkv_shape, 4);
524 SNEPPX_tensor_destroy(q); SNEPPX_tensor_destroy(k); SNEPPX_tensor_destroy(v);
525
526 if (w->config.use_rope && cos && sin)
527 SNEPPX_rope_apply(q_r, k_r, cos, sin, offset);
528
529 /* Append to cache */
530 float* kd = (float*)k_r->data;
531 float* vd = (float*)v_r->data;
532 float* kcd = (float*)cache->k_cache->data;
533 float* vcd = (float*)cache->v_cache->data;
534 size_t K_cache = cache->k_cache->shape[2];
535 for (size_t b = 0; b < B; b++) {
536 for (size_t h = 0; h < H; h++) {
537 for (size_t s = 0; s < S; s++) {
538 for (size_t d = 0; d < hd; d++) {
539 size_t dst = ((b * H + h) * K_cache + offset + s) * hd + d;
540 size_t src = ((b * H + h) * S + s) * hd + d;
541 kcd[dst] = kd[src];
542 vcd[dst] = vd[src];
543 }
544 }
545 }
546 }
547 SNEPPX_tensor_destroy(k_r); SNEPPX_tensor_destroy(v_r);
548 cache->seq_len = total_seq;
549
550 /* Use full K,V from cache */
551 size_t full_kv_shape[] = {B, H, total_seq, hd};
552 SNEPPXTensor* k_full = SNEPPX_tensor_as_strided(cache->k_cache, 0, full_kv_shape, 4,
553 cache->k_cache->strides);
554 SNEPPXTensor* v_full = SNEPPX_tensor_as_strided(cache->v_cache, 0, full_kv_shape, 4,
555 cache->v_cache->strides);
556
557 /* Attention scores */
558 SNEPPXTensor* scores = SNEPPX_batched_matmul(q_r, k_full, 1, 0);
559 float scale = 1.0f / sqrtf((float)hd);
560 float* sd = (float*)scores->data;
561 for (size_t i = 0; i < scores->size; i++) sd[i] *= scale;
562
563 /* Causal mask for full sequence */
564 if (w->config.use_causal_mask && total_seq > 1) {
565 for (size_t b = 0; b < B; b++) {
566 for (size_t h = 0; h < H; h++) {
567 for (size_t i = 0; i < S; i++) {
568 for (size_t j = 0; j < total_seq; j++) {
569 if (j > offset + i)
570 sd[((b * H + h) * S + i) * total_seq + j] = -1e9f;
571 }
572 }
573 }
574 }
575 }
576
577 /* Softmax */
578 for (size_t i = 0; i < scores->size; i += total_seq) {
579 float maxv = sd[i];
580 for (size_t j = 1; j < total_seq; j++)
581 if (sd[i + j] > maxv) maxv = sd[i + j];
582 float sum = 0.0f;
583 for (size_t j = 0; j < total_seq; j++) {
584 sd[i + j] = expf(sd[i + j] - maxv);
585 sum += sd[i + j];
586 }
587 float inv = 1.0f / (sum + 1e-10f);
588 for (size_t j = 0; j < total_seq; j++) sd[i + j] *= inv;
589 }
590
591 /* Apply to V */
592 SNEPPXTensor* attn_out = SNEPPX_batched_matmul(scores, v_full, 0, 0);
593 SNEPPX_tensor_destroy(scores);
594 SNEPPX_tensor_destroy(k_full); SNEPPX_tensor_destroy(v_full);
595
596 size_t out_shape[] = {B, S, D};
597 SNEPPXTensor* out_r = SNEPPX_tensor_reshape(attn_out, out_shape, 3);
598 SNEPPX_tensor_destroy(attn_out);
599
600 SNEPPXTensor* result = SNEPPX_tensor_matmul(out_r, w->w_o);
601 SNEPPX_tensor_destroy(out_r);
602 SNEPPX_tensor_destroy(q_r);
603 return result;
604}

606/* ========== Autodiff Training Graph ========== */
607
621int SNEPPX_attn_build_train_graph(SNEPPXAttentionWeights* w, SNEPPXTape* tape,
622 SNEPPXVariable* input_var,
623 SNEPPXVariable** weight_vars, size_t num_weights,
624 SNEPPXVariable** output_var,
625 SNEPPXTensor* cos, SNEPPXTensor* sin) {
626 if (!w || !tape || !input_var || !weight_vars || !output_var || num_weights < 8) return -1;
627 SNEPPXVariable* wq_v = weight_vars[0]; SNEPPXVariable* bq_v = weight_vars[1];
628 SNEPPXVariable* wk_v = weight_vars[2]; SNEPPXVariable* bk_v = weight_vars[3];
629 SNEPPXVariable* wv_v = weight_vars[4]; SNEPPXVariable* bv_v = weight_vars[5];
630 SNEPPXVariable* wo_v = weight_vars[6]; SNEPPXVariable* bo_v = weight_vars[7];
631 (void)bq_v; (void)bk_v; (void)bv_v; (void)bo_v;
632
633 size_t D = w->config.d_model, H = w->config.num_heads, hd = w->config.head_dim;
634 size_t B = input_var->data->shape[0], S = input_var->data->shape[1];
635 size_t dk = H * hd;
636
637 /* Flatten input: [B, S, D] -> [B*S, D] */
638 size_t flat_sh[] = {B * S, D};
639 SNEPPXVariable* x_flat = SNEPPX_reshape(tape, input_var, flat_sh, 2);
640
641 /* Q, K, V projections: [B*S, D] @ [D, dk] -> [B*S, dk] (all 2D matmuls) */
642 SNEPPXVariable* q_flat = SNEPPX_matmul(tape, x_flat, wq_v);
643 SNEPPXVariable* k_flat = SNEPPX_matmul(tape, x_flat, wk_v);
644 SNEPPXVariable* v_flat = SNEPPX_matmul(tape, x_flat, wv_v);
645
646 /* Single-head: Q, K are [B*S, dk]; K^T = [dk, B*S]; scores = Q @ K^T = [B*S, B*S] */
647 size_t k_t_sh[] = {dk, B * S};
648 SNEPPXVariable* k_t = SNEPPX_reshape(tape, k_flat, k_t_sh, 2);
649 SNEPPXVariable* scores = SNEPPX_matmul(tape, q_flat, k_t);
650
651 /* Scale by 1/sqrt(dk) */
652 float scale = 1.0f / sqrtf((float)dk);
653 size_t one_s[] = {1};
654 SNEPPXTensor* scale_t = SNEPPX_tensor_ones(one_s, 1, SNEPPX_FLOAT32);
655 if (!scale_t) return -1;
656 ((float*)scale_t->data)[0] = scale;
657 SNEPPXVariable* scale_v = SNEPPX_variable_create(scale_t, 0);
658 SNEPPX_tape_record(tape, scale_v);
659 scores = SNEPPX_mul(tape, scores, scale_v);
660
661 /* Causal mask */
662 if (w->config.use_causal_mask && S > 1) {
663 size_t mask_sh[] = {B * S, B * S};
664 SNEPPXTensor* mask_t = SNEPPX_tensor_create(mask_sh, 2, SNEPPX_FLOAT32);
665 if (!mask_t) return -1;
666 float* md = (float*)mask_t->data;
667 for (size_t i = 0; i < B * S; i++)
668 for (size_t j = 0; j < B * S; j++)
669 md[i * (B * S) + j] = (j <= i) ? 0.0f : -1e9f;
670 SNEPPXVariable* mask_v = SNEPPX_variable_create(mask_t, 0);
671 SNEPPX_tape_record(tape, mask_v);
672 scores = SNEPPX_add(tape, scores, mask_v);
673 }
674
675 /* Softmax over last dim */
676 SNEPPXVariable* attn = SNEPPX_softmax(tape, scores, 1);
677
678 /* attn @ V: [B*S, B*S] @ [B*S, dk] -> [B*S, dk] */
679 SNEPPXVariable* ctx = SNEPPX_matmul(tape, attn, v_flat);
680
681 /* Output projection: [B*S, dk] @ [dk, D] -> [B*S, D] */
682 SNEPPXVariable* out_flat = SNEPPX_matmul(tape, ctx, wo_v);
683
684 /* Reshape back to [B, S, D] */
685 size_t out_sh[] = {B, S, D};
686 *output_var = SNEPPX_reshape(tape, out_flat, out_sh, 3);
687 return 0;
688}
