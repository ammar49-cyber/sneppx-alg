80static int parse_csv_line(const char* line, float* out, size_t max_cols) {
81 size_t col = 0;
82 const char* p = line;
83 while (*p && col < max_cols) {
84 while (*p == ' ' || *p == '\t') p++;
85 if (*p == '\0' || *p == '\n' || *p == '\r') break;
86 char* end = NULL;
87 out[col++] = (float)strtod(p, &end);
88 if (end == p) break;
89 p = end;
90 while (*p == ',' || *p == ' ' || *p == '\t') p++;
91 }
92 return (int)col;
93}

206int SNEPPX_data_pipeline_get_batch(SNEPPXDataPipeline* pipe, SNEPPXTensor** batch, SNEPPXTensor** labels) {
207 if (!pipe || !pipe->data_buffer || !batch || !labels) return -1;
208 if (pipe->num_samples == 0) return -1;
209 size_t actual = pipe->batch_size;
210 if (pipe->current_pos + actual > pipe->num_samples) actual = pipe->num_samples - pipe->current_pos;
211 if (actual == 0) { pipe->current_pos = 0; actual = pipe->batch_size; if (actual > pipe->num_samples) actual = pipe->num_samples; }
212
213 float* batch_data = (float*)SNEPPX_malloc(actual * pipe->num_features * sizeof(float), 64);
214 float* batch_labels = (float*)SNEPPX_malloc(actual * pipe->num_labels * sizeof(float), 64);
215 if (!batch_data || !batch_labels) {
216 SNEPPX_free(batch_data, actual * pipe->num_features * sizeof(float));
217 SNEPPX_free(batch_labels, actual * pipe->num_labels * sizeof(float));
218 return -1;
219 }
220
221 for (size_t i = 0; i < actual; i++) {
222 size_t idx = pipe->shuffle_indices ? pipe->shuffle_indices[pipe->current_pos + i] : pipe->current_pos + i;
223 memcpy(batch_data + i * pipe->num_features, pipe->data_buffer + idx * pipe->num_features, pipe->num_features * sizeof(float));
224 memcpy(batch_labels + i * pipe->num_labels, pipe->label_buffer + idx * pipe->num_labels, pipe->num_labels * sizeof(float));
225 }
226 pipe->current_pos += actual;
227
228 {
229 size_t shape[] = {actual, pipe->num_features};
230 *batch = SNEPPX_tensor_empty(shape, 2, SNEPPX_FLOAT32);
231 if (*batch) memcpy((*batch)->data, batch_data, actual * pipe->num_features * sizeof(float));
232 }
233 {
234 size_t shape[] = {actual, pipe->num_labels};
235 *labels = SNEPPX_tensor_empty(shape, 2, SNEPPX_FLOAT32);
236 if (*labels) memcpy((*labels)->data, batch_labels, actual * pipe->num_labels * sizeof(float));
237 }
238 SNEPPX_free(batch_data, actual * pipe->num_features * sizeof(float));
239 SNEPPX_free(batch_labels, actual * pipe->num_labels * sizeof(float));
240 return 0;
241}

285SNEPPXTextDataset* SNEPPX_text_dataset_create(const char* path, SNEPPXTokenizer* tok,
286 size_t seq_len, int line_by_line) {
287 if (!path || !tok || seq_len == 0) return NULL;
288 FILE* f = fopen(path, "rb");
289 if (!f) return NULL;
290 fseek(f, 0, SEEK_END);
291 long fsize = ftell(f);
292 if (fsize <= 0) { fclose(f); return NULL; }
293 fseek(f, 0, SEEK_SET);
294 char* text = (char*)SNEPPX_malloc((size_t)fsize + 1, 64);
295 if (!text) { fclose(f); return NULL; }
296 size_t nread = fread(text, 1, (size_t)fsize, f);
297 fclose(f);
298 text[nread] = '\0';
299
300 size_t max_lines = 0;
301 for (size_t i = 0; i < nread; i++) if (text[i] == '\n') max_lines++;
302 if (max_lines == 0) { SNEPPX_free(text, (size_t)fsize + 1); return NULL; }
303
304 size_t max_token_buf = (size_t)fsize * sizeof(int);
305 size_t max_samples = (size_t)fsize;
306 size_t alloc_size = max_token_buf + max_samples * sizeof(size_t);
307 unsigned char* token_data = (unsigned char*)SNEPPX_malloc(alloc_size, 64);
308 if (!token_data) { SNEPPX_free(text, (size_t)fsize + 1); return NULL; }
309 int* td = (int*)token_data;
310 size_t* offsets = (size_t*)(token_data + max_token_buf);
311 size_t actual_samples = 0;
312 size_t token_cursor = 0;
313
314 if (line_by_line) {
315 char* line_start = text;
316 for (size_t i = 0; i <= nread && actual_samples < max_lines; i++) {
317 if (i == nread || text[i] == '\n') {
318 text[i] = '\0';
319 if (line_start < text + i) {
320 int n = SNEPPX_tokenizer_encode(tok, line_start, (int*)(td + token_cursor), (int)(max_token_buf / sizeof(int) - token_cursor));
321 if (n > 0) {
322 offsets[actual_samples] = token_cursor * sizeof(int);
323 token_cursor += (size_t)n;
324 actual_samples++;
325 }
326 }
327 line_start = text + i + 1;
328 }
329 }
330 } else {
331 int total_tokens = SNEPPX_tokenizer_encode(tok, text, (int*)td, (int)(max_token_buf / sizeof(int)));
332 if (total_tokens <= 0) { SNEPPX_free(token_data, alloc_size); SNEPPX_free(text, (size_t)fsize + 1); return NULL; }
333 size_t num_chunks = (size_t)total_tokens / seq_len;
334 if (num_chunks == 0) { SNEPPX_free(token_data, alloc_size); SNEPPX_free(text, (size_t)fsize + 1); return NULL; }
335 for (size_t i = 0; i < num_chunks; i++) {
336 offsets[i] = i * seq_len * sizeof(int);
337 actual_samples++;
338 }
339 }
340
341 SNEPPX_free(text, (size_t)fsize + 1);
342
343 SNEPPXTextDataset* ds = (SNEPPXTextDataset*)SNEPPX_malloc(sizeof(SNEPPXTextDataset), 64);
344 if (!ds) { SNEPPX_free(token_data, alloc_size); return NULL; }
345 ds->token_data = token_data;
346 ds->sample_offsets = offsets;
347 ds->num_samples = actual_samples;
348 ds->seq_len = seq_len;
349 ds->alloc_size = alloc_size;
350 ds->tok = tok;
351 return ds;
352}

382int SNEPPX_text_dataset_get_batch(const SNEPPXTextDataset* ds, size_t start_idx, size_t batch_size,
383 SNEPPXTensor** input_ids, SNEPPXTensor** target_ids) {
384 if (!ds || !input_ids || !target_ids) return 1;

388
389 size_t in_shape[] = {actual, ds->seq_len};
390 SNEPPXTensor* in_t = SNEPPX_tensor_empty(in_shape, 2, SNEPPX_INT32);
391 SNEPPXTensor* tgt_t = SNEPPX_tensor_empty(in_shape, 2, SNEPPX_INT32);
392 if (!in_t || !tgt_t) {
393 SNEPPX_tensor_destroy(in_t); SNEPPX_tensor_destroy(tgt_t);
394 return 1;
395 }
396
397 int* in_d = (int*)in_t->data;
398 int* tgt_d = (int*)tgt_t->data;
399 int* src = (int*)ds->token_data;
400
401 for (size_t i = 0; i < actual; i++) {
402 size_t sidx = start_idx + i;
403 size_t sample_offset = ds->sample_offsets[sidx] / sizeof(int);
404 for (size_t j = 0; j + 1 < ds->seq_len && sample_offset + j + 1 < ds->num_samples * ds->seq_len; j++) {
405 in_d[i * ds->seq_len + j] = src[sample_offset + j];
406 tgt_d[i * ds->seq_len + j] = src[sample_offset + j + 1];
407 }
408 }
409
410 *input_ids = in_t;
411 *target_ids = tgt_t;
412 return 0;
