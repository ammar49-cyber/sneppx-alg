89typedef struct WorkNode {
90 SNEPPXTask task;
91 struct WorkNode* next;
92} WorkNode;
93
94typedef struct {
95 WorkNode* head; /* front (stolen from) */
96 WorkNode* tail; /* back (owner pushes) */
97 SNEPPX_mutex_t lock;
98} WorkDeque;
99
100static void deque_init(WorkDeque* q) {
101 q->head = q->tail = NULL;
102 SNEPPX_mutex_init(&q->lock);
103}
104
105static void deque_destroy(WorkDeque* q) {
106 /* Drain remaining */
107 SNEPPX_mutex_lock(&q->lock);
108 WorkNode* n = q->head;
109 while (n) { WorkNode* next = n->next; free(n); n = next; }
110 q->head = q->tail = NULL;
111 SNEPPX_mutex_unlock(&q->lock);
112 SNEPPX_mutex_destroy(&q->lock);
113}
114
115/* Owner pushes to the back (LIFO for locality). */
116static int deque_push(WorkDeque* q, SNEPPXTask task) {
117 WorkNode* n = (WorkNode*)malloc(sizeof(WorkNode));
118 if (!n) return -1;
119 n->task = task;
120 n->next = NULL;
121 SNEPPX_mutex_lock(&q->lock);
122 if (q->tail) {
123 q->tail->next = n;
124 q->tail = n;
125 } else {
126 q->head = q->tail = n;
127 }
128 SNEPPX_mutex_unlock(&q->lock);
129 return 0;
130}
131
132/* Owner pops from the back. Returns 0 on success, -1 if empty. */
133static int deque_pop(WorkDeque* q, SNEPPXTask* out) {
134 SNEPPX_mutex_lock(&q->lock);
135 WorkNode* prev = NULL;
136 WorkNode* cur = q->head;
137 while (cur && cur != q->tail) { prev = cur; cur = cur->next; }
138 if (!cur) { SNEPPX_mutex_unlock(&q->lock); return -1; }
139 *out = cur->task;
140 if (prev) prev->next = NULL; else q->head = NULL;
141 q->tail = prev;
142 free(cur);
143 SNEPPX_mutex_unlock(&q->lock);
144 return 0;
145}
146
147/* Steal from the front. Returns 0 on success, -1 if empty. */
148static int deque_steal(WorkDeque* q, SNEPPXTask* out) {
149 SNEPPX_mutex_lock(&q->lock);
150 WorkNode* n = q->head;
151 if (!n) { SNEPPX_mutex_unlock(&q->lock); return -1; }
152 *out = n->task;
153 q->head = n->next;
154 if (!q->head) q->tail = NULL;
155 free(n);
156 SNEPPX_mutex_unlock(&q->lock);
157 return 0;
158}
159
160/* Number of tasks currently in the deque. */
161static int deque_count(WorkDeque* q) {
162 SNEPPX_mutex_lock(&q->lock);
163 int cnt = 0;
164 for (WorkNode* n = q->head; n; n = n->next) cnt++;
165 SNEPPX_mutex_unlock(&q->lock);
166 return cnt;
167}

286/* Worker thread entry point */
287#ifdef _WIN32
288static DWORD WINAPI worker_entry(LPVOID arg)
289#else
290static void* worker_entry(void* arg)
291#endif
292{
293 SNEPPXThreadPool* pool = (SNEPPXThreadPool*)((void**)arg)[0];
294 int tid = (int)(intptr_t)((void**)arg)[1];
295 set_thread_id(tid);
296 free(arg);
297
298 SNEPPXTask task;
299 while (!pool->shutdown) {
300 int got = 0;
301
302 /* 1. Try own queue (pop back — LIFO) */
303 if (deque_pop(&pool->queues[tid], &task) == 0) {
304 got = 1;
305 } else {
306 /* 2. Steal from other queues (front — FIFO) */
307 for (size_t i = 1; i < pool->num_threads; i++) {
308 size_t victim = (tid + i) % pool->num_threads;
309 if (deque_steal(&pool->queues[victim], &task) == 0) {
310 got = 1;
311 break;
312 }
313 }
314 }
315
316 if (got) {
317 task.func(task.arg);
318 SNEPPX_mutex_lock(&pool->global_lock);
319 pool->active_tasks--;
320 SNEPPX_mutex_unlock(&pool->global_lock);
321 continue;
322 }
323
324 /* 3. No work — sleep until signaled (predicate check avoids lost wakeup) */
325 SNEPPX_mutex_lock(&pool->global_lock);
326 pool->waiting_count++;
327 while (!pool->shutdown && pool->active_tasks == 0) {
328 SNEPPX_cond_wait(&pool->global_cond, &pool->global_lock);
329 }
330 pool->waiting_count--;
331 SNEPPX_mutex_unlock(&pool->global_lock);
332 }
333 return 0;
334}

341SNEPPXThreadPool* SNEPPX_threadpool_create(size_t num_threads) {
342 if (num_threads == 0) num_threads = SNEPPX_threadpool_default_count();
343
344 SNEPPXThreadPool* pool = (SNEPPXThreadPool*)malloc(sizeof(SNEPPXThreadPool));
345 if (!pool) return NULL;
346
347 pool->num_threads = num_threads;
348 pool->shutdown = 0;
349 pool->active_tasks = 0;
350 pool->waiting_count = 0;
351
352 pool->threads = (uintptr_t*)malloc(num_threads * sizeof(uintptr_t));
353 pool->queues = (WorkDeque*)malloc(num_threads * sizeof(WorkDeque));
354 if (!pool->threads || !pool->queues) {
355 free(pool->threads); free(pool->queues); free(pool);
356 return NULL;
357 }
358
359 SNEPPX_mutex_init(&pool->global_lock);
360 SNEPPX_cond_init(&pool->global_cond);
361
362 for (size_t i = 0; i < num_threads; i++)
363 deque_init(&pool->queues[i]);
364
365 /* Start worker threads */
366 size_t i;
367 for (i = 0; i < num_threads; i++) {
368 void** args = (void**)malloc(2 * sizeof(void*));
369 if (!args) goto fail;
370 args[0] = (void*)pool;
371 args[1] = (void*)(intptr_t)i;
372 if (SNEPPX_thread_create(&pool->threads[i], worker_entry, args) != 0) {
373 free(args);
374 goto fail;
375 }
376 }
377
378 return pool;
379
380fail:
381 for (size_t j = 0; j < num_threads; j++) {
382 if (j < i) {
383 SNEPPX_thread_join(pool->threads[j]);
384 deque_destroy(&pool->queues[j]);
385 } else {
386 deque_destroy(&pool->queues[j]);
387 }
388 }
389 SNEPPX_cond_destroy(&pool->global_cond);
390 SNEPPX_mutex_destroy(&pool->global_lock);
391 free(pool->threads); free(pool->queues); free(pool);
392 return NULL;
393}

413/* Submit a task to a thread's queue (round-robin via atomic index or random). */
414static volatile long g_rr_counter = 0;
415
423int SNEPPX_threadpool_submit(SNEPPXThreadPool* pool, SNEPPXTask task) {
424 size_t idx;
425 if (g_thread_id >= 0) {
426 idx = (size_t)g_thread_id; /* own queue */
427 } else {
428#ifdef _WIN32
429 idx = (size_t)(_InterlockedExchangeAdd(&g_rr_counter, 1) % (long)pool->num_threads);
430#else
431 idx = (size_t)(__sync_fetch_and_add(&g_rr_counter, 1) % (long)pool->num_threads);
432#endif
433 }
434 if (idx >= pool->num_threads) idx = 0;
435
436 int ret = deque_push(&pool->queues[idx], task);
437 if (ret != 0) return ret;
438
439 SNEPPX_mutex_lock(&pool->global_lock);
440 pool->active_tasks++;
441 /* Wake one sleeper */
442 if (pool->waiting_count > 0)
443 SNEPPX_cond_signal(&pool->global_cond);
444 SNEPPX_mutex_unlock(&pool->global_lock);
445 return 0;
446}

483void SNEPPX_threadpool_wait(SNEPPXThreadPool* pool) {
484 /* Busy-wait until all tasks are done. Workers are guaranteed to wake
485 * for every submitted task because the predicate-based cond_wait loop
486 * cannot lose signals. */
487 while (1) {
488 int all_empty = 1;
489 for (size_t i = 0; i < pool->num_threads; i++) {
490 if (deque_count(&pool->queues[i]) > 0) {
491 all_empty = 0;
492 break;
493 }
494 }
495 if (all_empty && pool->active_tasks == 0) break;
496#ifdef _WIN32
497 SwitchToThread();
498#else
499 sched_yield();
500#endif
501 }
502}

564void SNEPPX_parallel_for(SNEPPXThreadPool* pool,
565 size_t start, size_t end,
566 SNEPPXRangeFunc func, void* arg) {
567 size_t range = end - start;
568 size_t nthreads = pool->num_threads;
569 if (range == 0) return;
570 if (nthreads <= 1 || range < 64) {
571 func(start, end, arg);
572 return;
573 }
574
575 size_t chunk_size = (range + nthreads - 1) / nthreads;
576 size_t num_tasks = 0;
577
578 for (size_t s = start; s < end; s += chunk_size) {
579 size_t e = s + chunk_size;
580 if (e > end) e = end;
581 ForTaskArg* fta = (ForTaskArg*)malloc(sizeof(ForTaskArg));
582 if (!fta) { func(s, e, arg); continue; }
583 fta->func = func;
584 fta->arg = arg;
585 fta->start = s;
586 fta->end = e;
587 fta->chunk = chunk_size;
588
589 SNEPPXTask task;
590 task.func = for_task_func;
591 task.arg = fta;
592 if (SNEPPX_threadpool_submit(pool, task) == 0)
593 num_tasks++;
594 else
595 func(s, e, arg);
596 }
597
598 SNEPPX_threadpool_wait(pool);
599}

634void SNEPPX_parallel_reduce(SNEPPXThreadPool* pool,
635 size_t start, size_t end,
636 void* init, size_t elem_size,
637 SNEPPXReduceFunc reduce_func,
638 SNEPPXCombineFunc combine_func,
639 void* result,
640 void* user_arg) {
641 size_t range = end - start;
642 size_t nthreads = pool->num_threads;
643
644 if (init) memcpy(result, init, elem_size);
645 else memset(result, 0, elem_size);
646 if (range == 0) return;
647
648 /* Single-threaded path */
649 if (nthreads <= 1 || range < 64) {
650 memset(result, 0, elem_size);
651 reduce_func(start, end, user_arg, result);
652 return;
653 }
654
655 size_t chunk_size = (range + nthreads - 1) / nthreads;
656 size_t num_chunks = (range + chunk_size - 1) / chunk_size;
657 size_t max_chunks = num_chunks < nthreads ? num_chunks : nthreads;
658
659 /* Heap-allocate partial results */
660 unsigned char* partials = (unsigned char*)calloc(max_chunks, elem_size);
661 if (!partials) { reduce_func(start, end, NULL, result); return; }
662
663 size_t chunks_submitted = 0;
664 for (size_t s = start; s < end; s += chunk_size) {
665 size_t e = s + chunk_size;
666 if (e > end) e = end;
667
668 ReduceTaskArg* rta = (ReduceTaskArg*)malloc(sizeof(ReduceTaskArg));
669 if (!rta) { reduce_func(s, e, NULL, result); free(partials); return; }
670 rta->reduce_func = reduce_func;
671 rta->combine_func = combine_func;
672 rta->arg = user_arg;
673 rta->start = s;
674 rta->end = e;
675 rta->result = partials + chunks_submitted * elem_size;
676 rta->elem_size = elem_size;
677
678 SNEPPXTask task;
679 task.func = reduce_task_func;
680 task.arg = rta;
681 SNEPPX_threadpool_submit(pool, task);
682 chunks_submitted++;
683 }
684
685 SNEPPX_threadpool_wait(pool);
686
687 /* Combine partials */
688 if (chunks_submitted > 0) {
689 memcpy(result, partials, elem_size);
690 for (size_t i = 1; i < chunks_submitted; i++) {
691 combine_func(result, partials + i * elem_size);
692 }
693 }
694
695 free(partials);
696}
