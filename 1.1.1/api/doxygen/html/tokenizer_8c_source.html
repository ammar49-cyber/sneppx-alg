153int SNEPPX_tokenizer_add_token(SNEPPXTokenizer* tok, const char* token, int id) {
154 if (!tok || id < 0 || id >= tok->capacity) return -1;

173int SNEPPX_tokenizer_encode(const SNEPPXTokenizer* tok, const char* text, int* out_ids, size_t max_len) {
174 if (!tok || !text || !out_ids || max_len == 0) return -1;
175 size_t text_len = strlen(text);
176 IntArray* ids = intarr_create((int)(text_len + 1));
177 if (!ids) return -1;
178 for (size_t i = 0; i < text_len; i++) {
179 int id = find_token(tok, text + i, 1);
180 if (id < 0) id = tok->special.unk_id >= 0 ? tok->special.unk_id : (int)text[i];
181 intarr_push(ids, id);
182 }
183 int changed = 1;
184 while (changed) {
185 changed = 0;
186 for (int r = 0; r < tok->num_merges; r++) {
187 int l = tok->merges[r].left_id;
188 int ri = tok->merges[r].right_id;
189 int nid = tok->merges[r].new_id;
190 for (int i = 0; i < ids->len - 1; i++) {
191 if (ids->ids[i] == l && ids->ids[i + 1] == ri) {
192 ids->ids[i] = nid;
193 for (int j = i + 1; j < ids->len - 1; j++)
194 ids->ids[j] = ids->ids[j + 1];
195 ids->len--;
196 changed = 1;
197 i--;
198 }
199 }
200 }
201 }
202 size_t n = (size_t)ids->len < max_len ? (size_t)ids->len : max_len;
203 memcpy(out_ids, ids->ids, n * sizeof(int));
204 int result = ids->len;
205 intarr_free(ids);
206 return result;
207}

283SNEPPXTokenizer* SNEPPX_tokenizer_train_bpe(const char** texts, size_t num_texts, size_t target_vocab) {
284 if (!texts || num_texts == 0 || target_vocab <= 256) return NULL;
285 SNEPPXTokenizer* tok = SNEPPX_tokenizer_create((int)target_vocab + 256);

293 SNEPPX_free(corpus, num_texts * sizeof(IntArray*));
294 SNEPPX_tokenizer_destroy(tok); return NULL;

302 if (!pair_counts) break;

304 count_pairs(corpus, num_texts, pair_counts, pair_dim, NULL);
305 long long best_count = 0;
306 int best_l = -1, best_r = -1;
307 for (int l = 0; l < pair_dim; l++) {
308 for (int r = 0; r < pair_dim; r++) {
309 long long c = pair_counts[(size_t)l * (size_t)pair_dim + (size_t)r];
310 if (c > best_count) { best_count = c; best_l = l; best_r = r; }
311 }
312 }
313 SNEPPX_free(pair_counts, (size_t)pair_dim * (size_t)pair_dim * sizeof(long long));
314 if (best_l < 0 || best_count == 0) break;
315 int new_id = tok->vocab_size;
316 if (new_id >= tok->capacity) break;
317 int l_len = tok->token_lens[best_l];
318 int r_len = tok->token_lens[best_r];
319 char* new_bytes = tok->token_bytes + (size_t)new_id * MAX_TOKEN_LEN;
320 memcpy(new_bytes, tok->token_bytes + (size_t)best_l * MAX_TOKEN_LEN, (size_t)l_len);
321 memcpy(new_bytes + l_len, tok->token_bytes + (size_t)best_r * MAX_TOKEN_LEN, (size_t)r_len);
322 new_bytes[l_len + r_len] = '\0';
323 tok->token_lens[new_id] = l_len + r_len;
324 tok->vocab_size++;
325 if (tok->num_merges >= tok->merge_capacity) {
326 int nc = tok->merge_capacity ? tok->merge_capacity * 2 : 1024;
327 BpeMerge* nm = SNEPPX_realloc(tok->merges, (size_t)tok->merge_capacity * sizeof(BpeMerge),
328 (size_t)nc * sizeof(BpeMerge), 64);
329 if (!nm) break;
330 tok->merges = nm; tok->merge_capacity = nc;
331 }
332 tok->merges[tok->num_merges].left_id = best_l;
333 tok->merges[tok->num_merges].right_id = best_r;
334 tok->merges[tok->num_merges].new_id = new_id;
335 tok->num_merges++;
336 current_max_id = new_id + 1;
337 for (size_t t = 0; t < num_texts; t++)
338 apply_merge(corpus[t], best_l, best_r, new_id);
339 }
340 for (size_t t = 0; t < num_texts; t++) intarr_free(corpus[t]);
341 SNEPPX_free(corpus, num_texts * sizeof(IntArray*));

392 tok->vocab_size = vs;

396 tok->token_lens[i] = len;

398 SNEPPX_tokenizer_destroy(tok); fclose(f); return NULL;
399 }
400 }
401 for (int i = 0; i < nm; i++) {
402 BpeMerge m;

404 || fread(&m.new_id, sizeof(int), 1, f) != 1) {

406 }

415 }
