254static void json_escape(char* out, size_t out_size, const char* in) {
255 size_t j = 0;
256 for (size_t i = 0; in[i] && j < out_size - 1; i++) {
257 if (in[i] == '"' || in[i] == '\\') {
258 if (j < out_size - 2) out[j++] = '\\';
259 }
260 out[j++] = in[i];
261 }
262 out[j] = '\0';
263}

496int SNEPPX_llm_config_extend_context(SNEPPXLLMConfig* cfg, size_t target_len) {
497 if (!cfg || target_len == 0) return -1;
498
499 switch (cfg->family) {
500 case SNEPPX_MODEL_LLAMA_2:
501 case SNEPPX_MODEL_LLAMA_3: {
502 SNEPPXLlamaConfig* l = &cfg->config.llama;
503 if (target_len <= l->max_position_embeddings) return 0;
504 float scale = (float)target_len / (float)l->max_position_embeddings;
505 l->rope_theta *= scale > 1.0f ? scale : 1.0f;
506 l->use_scaled_rope = 1;
507 l->max_position_embeddings = target_len;
508 return 0;
509 }
510 case SNEPPX_MODEL_MISTRAL: {
511 SNEPPXMistralConfig* m = &cfg->config.mistral;
512 if (target_len <= m->max_position_embeddings) return 0;
513 float scale = (float)target_len / (float)m->max_position_embeddings;
514 m->rope_theta *= scale > 1.0f ? scale : 1.0f;
515 m->max_position_embeddings = target_len;
516 return 0;
517 }
518 case SNEPPX_MODEL_QWEN_2: {
519 SNEPPXQwen2Config* q = &cfg->config.qwen2;
520 if (target_len <= q->max_position_embeddings) return 0;
521 float scale = (float)target_len / (float)q->max_position_embeddings;
522 q->rope_theta *= scale > 1.0f ? scale : 1.0f;
523 q->rope_scaling_factor = scale;
524 q->use_rope_scaling = 1;
525 q->max_position_embeddings = target_len;
526 return 0;
527 }
528 case SNEPPX_MODEL_DEEPSEEK_V2: {
529 SNEPPXDeepSeekV2Config* d = &cfg->config.deepseek_v2;
530 if (target_len <= d->max_position_embeddings) return 0;
531 float scale = (float)target_len / (float)d->max_position_embeddings;
532 d->rope_theta *= scale > 1.0f ? scale : 1.0f;
533 d->max_position_embeddings = target_len;
534 return 0;
535 }
536 default:
537 return -1;
538 }
539}
