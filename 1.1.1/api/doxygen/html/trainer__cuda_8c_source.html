218int SNEPPX_trainer_cuda_optimizer_step(SNEPPXOptimizer* opt, SNEPPXTensor** params, SNEPPXTensor** grads, size_t n) {
219#if defined(SNEPPX_HAS_CUDA)
220 if (!g_trainer_cuda.initialized || !opt || !params || !grads) return -1;
221
222 float lr = opt->learning_rate;
223 float wd = opt->weight_decay;
224 float beta1 = opt->beta1;
225 float beta2 = opt->beta2;
226 float eps = opt->epsilon;
227 int step = (int)opt->step_count + 1;
228
229 size_t m = n < g_trainer_cuda.num_params ? n : g_trainer_cuda.num_params;
230
231 for (size_t i = 0; i < m; i++) {
232 if (!params[i] || !grads[i] || !g_trainer_cuda.d_params[i] || !g_trainer_cuda.d_grads[i]) continue;
233 float* grad_data = (float*)grads[i]->data;
234 size_t numel = params[i]->size;
235 size_t bytes = numel * sizeof(float);
236
237 cudaMemcpyAsync(g_trainer_cuda.d_grads[i], grad_data, bytes,
238 cudaMemcpyHostToDevice, g_trainer_cuda.stream);
239
240 if (opt->type == SNEPPX_OPTIMIZER_SGD) {
241 float alpha = -lr;
242 cublasStatus_t cs = cublasSaxpy(g_trainer_cuda.blas, (int)numel, &alpha,
243 (const float*)g_trainer_cuda.d_grads[i], 1,
244 (float*)g_trainer_cuda.d_params[i], 1);
245 if (cs != CUBLAS_STATUS_SUCCESS) return -1;
246
247 if (wd != 0.0f) {
248 float wd_alpha = -lr * wd;
249 cs = cublasSaxpy(g_trainer_cuda.blas, (int)numel, &wd_alpha,
250 (const float*)g_trainer_cuda.d_params[i], 1,
251 (float*)g_trainer_cuda.d_params[i], 1);
252 if (cs != CUBLAS_STATUS_SUCCESS) return -1;
253 }
254 } else {
255 /* AdamW via cuBLAS-based element-wise operations */
256 float* d_p = (float*)g_trainer_cuda.d_params[i];
257 const float* d_g = (const float*)g_trainer_cuda.d_grads[i];
258 float* d_m = g_trainer_cuda.d_state1[i];
259 float* d_v = g_trainer_cuda.d_state2[i];
260 int N = (int)numel;
261
262 /* m = beta1*m + (1-beta1)*g */
263 float one_minus_b1 = 1.0f - beta1;
264 cublasSscal(g_trainer_cuda.blas, N, &beta1, d_m, 1);
265 cublasSaxpy(g_trainer_cuda.blas, N, &one_minus_b1, d_g, 1, d_m, 1);
266
267 /* v = beta2*v + (1-beta2)*g*g */
268 float one_minus_b2 = 1.0f - beta2;
269 cublasSscal(g_trainer_cuda.blas, N, &beta2, d_v, 1);
270 /* For g*g, need an element-wise kernel — fall back to host for now */
271 float* h_g = (float*)malloc(bytes);
272 float* h_v = (float*)malloc(bytes);
273 if (!h_g || !h_v) { free(h_g); free(h_v); return -1; }
274 cudaMemcpy(h_g, d_g, bytes, cudaMemcpyDeviceToHost);
275 cudaMemcpy(h_v, d_v, bytes, cudaMemcpyDeviceToHost);
276 for (size_t j = 0; j < numel; j++) {
277 h_v[j] += one_minus_b2 * h_g[j] * h_g[j];
278 }
279 cudaMemcpy(d_v, h_v, bytes, cudaMemcpyHostToDevice);
280 free(h_g); free(h_v);
281
282 /* bias-corrected lr */
283 float lr_corrected = lr * sqrtf(1.0f - powf(beta2, (float)step)) / (1.0f - powf(beta1, (float)step));
284
285 /* p = p - lr_corrected * m / (sqrt(v) + eps) - lr * wd * p */
286 if (wd != 0.0f) {
287 float neg_lr_wd = -lr * wd;
288 cublasSaxpy(g_trainer_cuda.blas, N, &neg_lr_wd, d_p, 1, d_p, 1);
289 }
290
291 /* Need element-wise division: p -= lr_corrected * m / (sqrt(v) + eps) */
292 float* h_p = (float*)malloc(bytes);
293 float* h_m = (float*)malloc(bytes);
294 if (!h_p || !h_m) { free(h_p); free(h_m); return -1; }
295 cudaMemcpy(h_p, d_p, bytes, cudaMemcpyDeviceToHost);
296 cudaMemcpy(h_m, d_m, bytes, cudaMemcpyDeviceToHost);
297 for (size_t j = 0; j < numel; j++) {
298 h_p[j] -= lr_corrected * h_m[j] / (sqrtf(h_v[j]) + eps);
299 }
300 cudaMemcpy(d_p, h_p, bytes, cudaMemcpyHostToDevice);
301 free(h_p); free(h_m);
302 }
303 }
304
305 cudaStreamSynchronize(g_trainer_cuda.stream);
306
307 for (size_t i = 0; i < m; i++) {
308 if (!params[i] || !g_trainer_cuda.d_params[i]) continue;
309 cudaMemcpyAsync(params[i]->data, g_trainer_cuda.d_params[i],
310 g_trainer_cuda.sizes[i], cudaMemcpyDeviceToHost,
311 g_trainer_cuda.stream);
312 }
313 cudaStreamSynchronize(g_trainer_cuda.stream);
314
315 return 0;
316#else
317 (void)opt; (void)params; (void)grads; (void)n;
318 return -1;
319#endif
