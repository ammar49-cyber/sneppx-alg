38int sneppx_distributed_sampler_init(SNEPPX_DistributedSampler** sampler,
39 int64_t num_samples, int world_size,
40 int rank, int shuffle, int64_t seed) {
41 if (!sampler || world_size <= 0) return -1;
42 SNEPPX_DistributedSampler* s = (SNEPPX_DistributedSampler*)calloc(1, sizeof(SNEPPX_DistributedSampler));
43 if (!s) return -1;
44 s->num_samples = num_samples;
45 s->world_size = world_size;
46 s->rank = rank;
47 s->shuffle = shuffle;
48 s->seed = seed;
49 s->epoch = 0;
50
51 // Compute partition
52 int64_t total = num_samples;
53 int64_t per_rank = (total + world_size - 1) / world_size;
54 s->num_samples_per_rank = per_rank;
55 s->indices = (int64_t*)malloc(per_rank * sizeof(int64_t));
56
57 for (int64_t i = 0; i < per_rank; i++) {
58 int64_t idx = rank * per_rank + i;
59 if (idx >= total) idx = total - 1;
60 s->indices[i] = idx;
61 }
62
63 *sampler = s;
64 return 0;
65}

73void sneppx_distributed_sampler_set_epoch(SNEPPX_DistributedSampler* sampler,
74 int64_t epoch) {
75 if (!sampler) return;
76 sampler->epoch = epoch;
77 if (sampler->shuffle) {
78 // Fisher-Yates shuffle with epoch-based seed
79 srand((unsigned int)(sampler->seed + epoch));
80 for (int64_t i = sampler->num_samples_per_rank - 1; i > 0; i--) {
81 int64_t j = rand() % (i + 1);
82 int64_t tmp = sampler->indices[i];
83 sampler->indices[i] = sampler->indices[j];
84 sampler->indices[j] = tmp;
85 }
86 }
87}
