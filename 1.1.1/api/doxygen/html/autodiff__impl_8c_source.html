110static void toposort_dfs(SNEPPXGradNode* node, SNEPPXGradNode*** order, size_t* count, size_t* capacity, int* visited) {
111 if (!node || visited[node->id % 65536]) return;
112 visited[node->id % 65536] = 1;

130int SNEPPX_grad_graph_toposort(SNEPPXGradGraph* graph) {
131 if (!graph || graph->num_nodes == 0) return 0;
132 free(graph->topological_order);
133 graph->topological_order = NULL;
134 size_t capacity = graph->num_nodes > 0 ? graph->num_nodes : 1;
135 graph->topological_order = (SNEPPXGradNode**)malloc(capacity * sizeof(SNEPPXGradNode*));
136 if (!graph->topological_order) return -1;
137 size_t count = 0;
138 int visited[65536] = {0};
139 for (size_t i = 0; i < graph->num_nodes; i++) {
140 if (graph->nodes[i])
141 toposort_dfs(graph->nodes[i], &graph->topological_order, &count, &capacity, visited);
142 }
143 graph->needs_rebuild = 0;
144 return 0;
145}

154int SNEPPX_grad_graph_backward(SNEPPXGradGraph* graph, SNEPPXGradNode* root) {
155 if (!graph || !root) return -1;
156 if (graph->needs_rebuild) SNEPPX_grad_graph_toposort(graph);
157 if (!graph->topological_order) return -1;

159 for (size_t i = graph->num_nodes; i > 0; i--) {
160 SNEPPXGradNode* node = graph->topological_order[i - 1];

162 ((void (*)(void*, void*))node->grad_fn)(node->saved_tensors, NULL);
163 }
164 return 0;
