35 * GPTQ (Post-Training Quantization):

55int SNEPPX_gptq_compute_hessian(float* hessian, const float* activation,
56 size_t n, size_t dim)
57{
58 if (!hessian || !activation || n == 0 || dim == 0) return -1;
59 memset(hessian, 0, dim * dim * sizeof(float));
60 for (size_t i = 0; i < n; i++) {
61 const float* row = activation + i * dim;
62 for (size_t r = 0; r < dim; r++) {
63 for (size_t c = r; c < dim; c++) {
64 hessian[r * dim + c] += row[r] * row[c];
65 }
66 }
67 }
68 for (size_t r = 0; r < dim; r++) {
69 for (size_t c = 0; c < r; c++) {
70 hessian[r * dim + c] = hessian[c * dim + r];
71 }
72 }
73 float reg = 1e-5f * hessian[0];
74 if (reg < 1e-8f) reg = 1e-5f;
75 for (size_t i = 0; i < dim; i++) {
76 hessian[i * dim + i] += reg;
77 }
78 return 0;
79}

81static int _cholesky_inv(float* h, size_t dim)
82{
83 float* L = (float*)calloc(dim * dim, sizeof(float));
84 if (!L) return -1;
85 for (size_t j = 0; j < dim; j++) {
86 float s = 0.0f;
87 for (size_t k = 0; k < j; k++) {
88 s += L[j * dim + k] * L[j * dim + k];
89 }
90 float d = h[j * dim + j] - s;
91 if (d < 1e-10f) { free(L); return -1; }
92 L[j * dim + j] = sqrtf(d);
93 for (size_t i = j + 1; i < dim; i++) {
94 s = 0.0f;
95 for (size_t k = 0; k < j; k++) {
96 s += L[i * dim + k] * L[j * dim + k];
97 }
98 L[i * dim + j] = (h[i * dim + j] - s) / L[j * dim + j];
99 }
100 }
101 for (size_t i = 0; i < dim; i++) {
102 for (size_t j = 0; j < dim; j++) {
103 if (j > i) {
104 h[i * dim + j] = 0.0f;
105 } else if (j == i) {
106 h[i * dim + j] = 1.0f / L[i * dim + i];
107 } else {
108 float s = 0.0f;
109 for (size_t k = j; k < dim; k++) {
110 s += L[i * dim + k] * h[k * dim + j];
111 }
112 h[i * dim + j] = -s / L[i * dim + i];
113 }
114 }
115 }
116 free(L);
117 return 0;
118}

134int SNEPPX_gptq_quantize_block(float* w, float* h_inv,
135 float* qw, float* scale, int32_t* zero,
136 size_t block_size, int bits, int sym)
137{
138 if (!w || !h_inv || !qw || !scale) return -1;
139 int qmax = (1 << (bits - 1)) - 1;
140 int qmin = -qmax;
141 if (!sym) { qmax = (1 << bits) - 1; qmin = 0; }
142 size_t cols = block_size;
143 float max_abs = 0.0f;
144 for (size_t j = 0; j < cols; j++) {
145 float a = fabsf(w[j]);
146 if (a > max_abs) max_abs = a;
147 }
148 if (max_abs < FLT_MIN) {
149 *scale = 1.0f;
150 if (zero) *zero = 0;
151 memset(qw, 0, cols * sizeof(float));
152 return 0;
153 }
154 *scale = max_abs / (float)qmax;
155 if (zero) *zero = 0;
156 float inv_scale = 1.0f / (*scale);
157 for (size_t j = 0; j < cols; j++) {
158 int qi = (int)roundf(w[j] * inv_scale);
159 qi = SNEPPX_CLAMP(qi, qmin, qmax);
160 float q = (float)qi * (*scale);
161 float err = q - w[j];
162 qw[j] = q;
163 float h_col = h_inv[j * cols + j];
164 if (fabsf(h_col) > 1e-10f) {
165 float coef = -err / h_col;
166 for (size_t k = j + 1; k < cols; k++) {
167 w[k] += coef * h_inv[k * cols + j];
168 }
169 }
170 }
171 return 0;
172}

189int SNEPPX_gptq_quantize(float* weights, size_t rows, size_t cols,
190 int group_size, int bits,
191 SNEPPXDtype* qweight, float* scales,
192 int32_t* zeros, int sym)
193{
194 if (!weights || !qweight || !scales || rows == 0 || cols == 0) return -1;
195 if (group_size <= 0) group_size = 128;
196 if (bits <= 0 || bits > 16) bits = 8;
197 size_t num_groups = (cols + group_size - 1) / group_size;
198 for (size_t r = 0; r < rows; r++) {
199 float* row = weights + r * cols;
200 for (size_t g = 0; g < num_groups; g++) {
201 size_t g_start = g * group_size;
202 size_t g_end = SNEPPX_MIN(g_start + group_size, cols);
203 size_t gs = g_end - g_start;
204 float* block = (float*)malloc(gs * sizeof(float));
205 float* h_inv = (float*)malloc(gs * gs * sizeof(float));
206 float* qblock = (float*)malloc(gs * sizeof(float));
207 if (!block || !h_inv || !qblock) {
208 free(block); free(h_inv); free(qblock);
209 return -1;
210 }
211 memcpy(block, row + g_start, gs * sizeof(float));
212 memset(h_inv, 0, gs * gs * sizeof(float));
213 for (size_t i = 0; i < gs; i++) h_inv[i * gs + i] = 1.0f;
214 if (_cholesky_inv(h_inv, gs) != 0) {
215 for (size_t i = 0; i < gs; i++) {
216 for (size_t j = 0; j < gs; j++) {
217 h_inv[i * gs + j] = (i == j) ? 1.0f : 0.0f;
218 }
219 }
220 }
221 float scale;
222 int32_t zero = 0;
223 SNEPPX_gptq_quantize_block(block, h_inv, qblock,
224 &scale, &zero, gs, bits, sym);
225 scales[r * num_groups + g] = scale;
226 if (zeros) zeros[r * num_groups + g] = zero;
227 size_t elem_size = (bits <= 8) ? 1 : 2;
228 if (elem_size == 1) {
229 for (size_t j = 0; j < gs; j++) {
230 int qi = (int)roundf(block[j] / scale);
231 ((int8_t*)qweight)[r * cols + g_start + j] =
232 (int8_t)SNEPPX_CLAMP(qi, -128, 127);
233 }
234 }
235 free(block); free(h_inv); free(qblock);
236 }
237 }
238 return 0;
