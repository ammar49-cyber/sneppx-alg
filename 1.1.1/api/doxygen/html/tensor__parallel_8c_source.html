57// Output: partial [M, N/tp], all-reduce to get full [M, N]
58int sneppx_tp_linear_forward(SNEPPX_TensorParallel* tp,

66 cublasSetStream(handle, stream);
67
68 // Each rank computes partial output
69 int n_local = n / tp->tp_size;
70 int rank_offset = tp->tp_rank * n_local;
71
72 float alpha = 1.0f, beta = 0.0f;
73 cublasSgemm(handle, CUBLAS_OP_T, CUBLAS_OP_N,
