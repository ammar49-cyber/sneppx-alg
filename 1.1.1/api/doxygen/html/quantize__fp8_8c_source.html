41uint8_t SNEPPX_float_to_fp8_e4m3(float value)
42{
43 uint32_t b;
44 memcpy(&b, &value, sizeof(b));
45 uint32_t sign = (b >> 31) & 1;
46 uint32_t exp_bits = (b >> 23) & 0xFF;
47
48 /* Handle special float32 values */
49 if (exp_bits == 0xFF) {
50 /* Inf or NaN -> FP8 Inf (0x78/0xF8) */
51 return (uint8_t)((sign << 7) | 0x78);
52 }
53 if (exp_bits == 0) {
54 /* Subnormal or zero -> FP8 zero */
55 return (uint8_t)(sign << 7);
56 }
57 int32_t exp = (int32_t)exp_bits - 127;
58 uint32_t mant = (b >> 20) & 0x07;
59
60 if (exp < -6) {
61 return (uint8_t)(sign << 7);
62 }
63 if (exp > 8) {
64 return (uint8_t)((sign << 7) | 0x78); /* Inf */
65 }
66 uint8_t e4m3_exp = (uint8_t)(exp + 7);
67 uint8_t e4m3 = (uint8_t)((sign << 7) | (e4m3_exp << 3) | mant);
68 return e4m3;
69}
