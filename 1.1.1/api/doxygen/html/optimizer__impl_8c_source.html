80int SNEPPX_optimizer_sgd_step(SNEPPXOptimizerState* state) {
81 if (!state) return -1;

83 float wd = state->weight_decay;
84 size_t n = state->param_size / sizeof(float);
85 for (size_t i = 0; i < state->num_params; i++) {
86 float* p = (float*)state->param_data[i];

104int SNEPPX_optimizer_adam_step(SNEPPXOptimizerState* state) {
105 if (!state) return -1;
106 float lr = state->lr;
107 float b1 = state->beta1, b2 = state->beta2;
108 float ep = state->eps, wd = state->weight_decay;
109 uint64_t t = state->step + 1;
110 float bias_corr1 = 1.0f / (1.0f - powf(b1, (float)t));
111 float bias_corr2 = 1.0f / (1.0f - powf(b2, (float)t));
112 float adj_lr = lr * bias_corr1;
113 size_t n = state->param_size / sizeof(float);
114 for (size_t i = 0; i < state->num_params; i++) {
115 float* p = (float*)state->param_data[i];
116 float* g = (float*)state->grad_data[i];
117 float* m = get_or_create(&state->momentum_buf, i, state->param_size);
118 float* v = get_or_create(&state->velocity_buf, i, state->param_size);
119 if (!p || !g || !m || !v) continue;
120 for (size_t j = 0; j < n; j++) {
121 float grad = g[j];
122 if (wd != 0.0f) grad += wd * p[j];
123 m[j] = b1 * m[j] + (1.0f - b1) * grad;
124 v[j] = b2 * v[j] + (1.0f - b2) * grad * grad;
125 float m_hat = m[j] * bias_corr1;
126 float v_hat = v[j] * bias_corr2;
127 p[j] -= adj_lr * m_hat / (sqrtf(v_hat) + ep);
128 }
129 }
130 state->step++;
131 return 0;
132}

173int SNEPPX_optimizer_lamb_step(SNEPPXOptimizerState* state) {
174 if (!state) return -1;
175 float lr = state->lr;
176 float b1 = state->beta1, b2 = state->beta2;
177 float ep = state->eps, wd = state->weight_decay;
178 uint64_t t = state->step + 1;
179 float bias_corr1 = 1.0f / (1.0f - powf(b1, (float)t));
180 float bias_corr2 = 1.0f / (1.0f - powf(b2, (float)t));
181 size_t n = state->param_size / sizeof(float);
182 for (size_t i = 0; i < state->num_params; i++) {
183 float* p = (float*)state->param_data[i];
184 float* g = (float*)state->grad_data[i];
185 float* m = get_or_create(&state->momentum_buf, i, state->param_size);
186 float* v = get_or_create(&state->velocity_buf, i, state->param_size);
187 if (!p || !g || !m || !v) continue;
188 float grad_norm = 0.0f, param_norm = 0.0f;
189 for (size_t j = 0; j < n; j++) {
190 float grad = g[j];
191 if (wd != 0.0f) grad += wd * p[j];
192 m[j] = b1 * m[j] + (1.0f - b1) * grad;
193 v[j] = b2 * v[j] + (1.0f - b2) * grad * grad;
194 float m_hat = m[j] * bias_corr1;
195 float v_hat = v[j] * bias_corr2;
196 float update = m_hat / (sqrtf(v_hat) + ep);
197 grad_norm += update * update;
198 param_norm += p[j] * p[j];
199 p[j] -= lr * update;
200 }
201 float gn = sqrtf(grad_norm);
202 float pn = sqrtf(param_norm);
203 if (gn > 0.0f && pn > 0.0f) {
204 float trust_ratio = (pn > 0.0f) ? (pn / gn) : 1.0f;
205 if (trust_ratio > 10.0f) trust_ratio = 10.0f;
206 for (size_t j = 0; j < n; j++) {
207 float grad = g[j];
208 if (wd != 0.0f) grad += wd * p[j];
209 float m_hat = m[j] * bias_corr1;
210 float v_hat = v[j] * bias_corr2;
211 p[j] += lr * trust_ratio * m_hat / (sqrtf(v_hat) + ep) - lr * m_hat / (sqrtf(v_hat) + ep);
212 p[j] -= lr * trust_ratio * m_hat / (sqrtf(v_hat) + ep);
213 }
214 }
215 }
216 state->step++;
217 return 0;
218}

229int SNEPPX_gradient_clip_norm(void** grads, size_t num_params, size_t param_size, float max_norm) {
230 if (!grads || num_params == 0 || param_size == 0) return 0;
231 float total_norm = 0.0f;
232 size_t n = param_size / sizeof(float);
233 for (size_t i = 0; i < num_params; i++) {
234 float* g = (float*)grads[i];
235 if (!g) continue;
236 for (size_t j = 0; j < n; j++) total_norm += g[j] * g[j];
237 }
238 total_norm = sqrtf(total_norm);
239 if (total_norm <= max_norm || total_norm == 0.0f) return 0;
240 float scale = max_norm / total_norm;
241 for (size_t i = 0; i < num_params; i++) {
242 float* g = (float*)grads[i];
243 if (!g) continue;
244 for (size_t j = 0; j < n; j++) g[j] *= scale;
245 }
246 return 0;

298float SNEPPX_impl_lr_scheduler_step(SNEPPXLRScheduler* sched) {
299 if (!sched) return 0.0f;
300 size_t step = sched->step_count;
301 float progress = (sched->total_steps > 0) ? (float)step / (float)sched->total_steps : 0.0f;
302 if (step < sched->warmup_steps && sched->warmup_steps > 0) {
303 sched->current_lr = sched->base_lr * (float)(step + 1) / (float)sched->warmup_steps;
304 } else {
305 switch (sched->type) {
306 case SNEPPX_SCHEDULE_CONSTANT:
307 sched->current_lr = sched->base_lr;
308 break;
309 case SNEPPX_SCHEDULE_COSINE:
310 sched->current_lr = sched->min_lr + 0.5f * (sched->base_lr - sched->min_lr) * (1.0f + cosf((float)3.14159265f * progress));
311 break;
312 case SNEPPX_SCHEDULE_LINEAR:
313 sched->current_lr = sched->base_lr + (sched->min_lr - sched->base_lr) * progress;
314 break;
315 case SNEPPX_SCHEDULE_STEP_DECAY: {
316 size_t milestone = sched->total_steps / 3;
317 if (milestone == 0) milestone = 1;
318 size_t decay_count = step / milestone;
319 float decay_factor = 1.0f;
320 for (size_t d = 0; d < decay_count && d < 3; d++) decay_factor *= 0.1f;
321 sched->current_lr = sched->base_lr * decay_factor;
322 break;
323 }
324 case SNEPPX_SCHEDULE_WARMUP_COSINE:
325 progress = (float)(step - sched->warmup_steps) / (float)(sched->total_steps - sched->warmup_steps + 1);
326 if (progress > 1.0f) progress = 1.0f;
327 sched->current_lr = sched->min_lr + 0.5f * (sched->base_lr - sched->min_lr) * (1.0f + cosf((float)3.14159265f * progress));
328 break;
329 default:
330 sched->current_lr = sched->base_lr;
331 break;
332 }
333 }
334 sched->step_count++;
335 return sched->current_lr;
336}
