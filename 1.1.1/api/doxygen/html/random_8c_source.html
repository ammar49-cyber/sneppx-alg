59static int test_rng_generate(uint8_t* out, size_t len) {
60 size_t produced = 0;
61 while (produced < len) {
62 if (g_test_state_used >= 64) {
63 SNEPPXSHA512Context ctx;
64 SNEPPX_sha512_init(&ctx);
65 SNEPPX_sha512_update(&ctx, g_test_state, 64);
66 SNEPPX_sha512_finish(&ctx, g_test_state);

70 size_t take = (len - produced < avail) ? (len - produced) : avail;

72 g_test_state_used += take;
