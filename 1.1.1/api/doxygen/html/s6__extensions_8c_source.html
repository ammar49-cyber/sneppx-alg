28static void gf_init(void) {
29 if (gf_tables_initialized) return;
30 int x = 1;
31 for (int i = 0; i < 255; i++) {
32 gf_exp[i] = (uint8_t)x;
33 gf_log[x] = (uint8_t)i;
34 x = (x << 1) ^ (x & 0x80 ? 0x11b : 0);
35 }
36 for (int i = 255; i < 512; i++) gf_exp[i] = gf_exp[i - 255];
37 gf_tables_initialized = 1;
38}

40static uint8_t gf_mul(uint8_t a, uint8_t b) {
41 if (a == 0 || b == 0) return 0;
42 return gf_exp[gf_log[a] + gf_log[b]];
43}

45static uint8_t gf_inv(uint8_t a) {
46 if (a == 0) return 0;
47 return gf_exp[255 - gf_log[a]];
48}

53static uint8_t gf_poly_eval(uint8_t* coeff, int k, uint8_t x) {
54 uint8_t result = coeff[k - 1];
55 for (int i = k - 2; i >= 0; i--) {
56 result = gf_add(gf_mul(result, x), coeff[i]);
57 }
58 return result;

258 gf_init();

260 if (*secret_len < len) return -1;
261 *secret_len = len;
262 uint8_t x_vals[16];
263 for (int i = 0; i < k; i++) x_vals[i] = shares[i][0];

362static void dash_parse_json(const char* json) {
363 if (!json) return;
364 int brace_depth = 0;
365 int key_count = 0;
366 int arr_count = 0;
367 int colon_count = 0;
368 int comma_count = 0;

371 while (*json) {
372 char c = *json;

374 if (!in_string) {
375 if (c == '{') brace_depth++;
376 if (c == '}') brace_depth--;
377 if (c == '[') arr_count++;
378 if (c == ']') arr_count--;

380 if (c == ',') comma_count++;
381 }
382 prev = c;
383 json++;
384 }
385 dash_state.total_requests++;
386 dash_state.active_connections = arr_count > 0 ? arr_count : dash_state.active_connections;
387 if (comma_count > 20) dash_state.errors_logged++;
388 if (key_count > 8) dash_state.critical_alerts++;
389 else if (key_count > 4) dash_state.warnings++;
