149SNEPPXModel* SNEPPX_model_create(const SNEPPXArchConfig* config) {
150 if (!config) return NULL;
151 SNEPPXModel* model = (SNEPPXModel*)SNEPPX_malloc(sizeof(SNEPPXModel), 64);
152 if (!model) return NULL;
153 memset(model, 0, sizeof(SNEPPXModel));
154
155 model->config = *config;
156 unsigned int seed = config->seed ? config->seed : 42;
157 size_t in_dim = config->input_dim ? config->input_dim : 16;
158 size_t vocab = config->vocab_size ? config->vocab_size : 256;
159
160 if (config->enable_hss) {
161 SNEPPXHSSConfig hss_cfg = config->hss_config;
162 if (hss_cfg.input_dim == 0) { hss_cfg = SNEPPX_hss_config_default(); hss_cfg.input_dim = in_dim; }
163 if (hss_cfg.output_dim == 0) hss_cfg.output_dim = in_dim;
164 model->hss_model = SNEPPX_hss_model_create(&hss_cfg, seed);
165 if (!model->hss_model) { SNEPPX_model_destroy(model); return NULL; }
166 }
167
168 if (config->enable_ser) {
169 SNEPPXSERConfig ser_cfg = config->ser_config;
170 if (ser_cfg.num_experts == 0) { ser_cfg = SNEPPX_ser_config_default(); ser_cfg.input_dim = in_dim; }
171 if (ser_cfg.output_dim == 0) ser_cfg.output_dim = in_dim;
172 model->ser_model = SNEPPX_ser_model_create(&ser_cfg, seed + 1, 1);
173 if (!model->ser_model) { SNEPPX_model_destroy(model); return NULL; }
174 }
175
176 if (config->enable_arc) {
177 SNEPPXARCConfig arc_cfg = config->arc_config;
178 if (arc_cfg.input_guard_strength == 0.0f) arc_cfg = SNEPPX_arc_config_default();
179 model->arc_layer = SNEPPX_arc_layer_create(&arc_cfg, in_dim, in_dim, seed + 2);
180 if (!model->arc_layer) { SNEPPX_model_destroy(model); return NULL; }
181 }
182
183 if (config->enable_npe) {
184 SNEPPXNPEConfig npe_cfg = config->npe_config;
185 if (npe_cfg.max_program_length == 0) npe_cfg = SNEPPX_npe_config_default();
186 model->npe_program = SNEPPX_npe_compile_mlp(in_dim, in_dim * 2);
187 if (!model->npe_program) { SNEPPX_model_destroy(model); return NULL; }
188 unsigned long s = seed;
189 size_t total_w = in_dim * (in_dim * 2) + (in_dim * 2) + (in_dim * 2) * in_dim + in_dim;
190 size_t limit = total_w < model->npe_program->memory->size ? total_w : model->npe_program->memory->size;
191 for (size_t i = 0; i < limit; i++) {
192 s = s * 1103515245UL + 12345UL;
193 ((float*)model->npe_program->memory->data)[i] = ((float)((s >> 16) & 0x7FFF) / 32767.0f - 0.5f) * 0.1f;
194 }
195 model->npe_vm = SNEPPX_npe_vm_create(&npe_cfg);
196 if (!model->npe_vm) { SNEPPX_model_destroy(model); return NULL; }
197 SNEPPX_npe_vm_load(model->npe_vm, model->npe_program);
198 }
199
200 if (config->enable_fm) {
201 SNEPPXFMConfig fm_cfg = config->fm_config;
202 if (fm_cfg.num_nodes == 0) fm_cfg = SNEPPX_fm_config_default();
203 model->fm_controller = SNEPPX_fm_controller_create(&fm_cfg);
204 if (!model->fm_controller) { SNEPPX_model_destroy(model); return NULL; }
205 }
206
207 if (config->enable_attention) {
208 SNEPPXAttentionConfig attn_cfg = config->attention_config;
209 if (attn_cfg.d_model == 0) { attn_cfg = SNEPPX_attn_config_default(); }
210 attn_cfg.d_model = in_dim;
211 if (attn_cfg.head_dim == 0) attn_cfg.head_dim = in_dim / (attn_cfg.num_heads > 0 ? attn_cfg.num_heads : 1);
212 model->attention = SNEPPX_attn_weights_create(attn_cfg, seed + 3);
213 if (!model->attention) { SNEPPX_model_destroy(model); return NULL; }
214
215 size_t esh[] = {vocab, in_dim};
216 model->embed_weight = SNEPPX_tensor_create(esh, 2, SNEPPX_FLOAT32);
217 if (model->embed_weight) {
218 unsigned long s = seed + 4;
219 for (size_t i = 0; i < vocab * in_dim; i++) {
220 s = s * 1103515245UL + 12345UL;
221 ((float*)model->embed_weight->data)[i] = ((float)((s >> 16) & 0x7FFF) / 32767.0f - 0.5f) * 0.1f;
222 }
223 }
224 size_t ush[] = {in_dim, vocab};
225 model->unembed_weight = SNEPPX_tensor_create(ush, 2, SNEPPX_FLOAT32);
226 if (model->unembed_weight) {
227 unsigned long s = seed + 5;
228 for (size_t i = 0; i < in_dim * vocab; i++) {
229 s = s * 1103515245UL + 12345UL;
230 ((float*)model->unembed_weight->data)[i] = ((float)((s >> 16) & 0x7FFF) / 32767.0f - 0.5f) * 0.1f;
231 }
232 }
233 }
234
235 return model;
236}

263int SNEPPX_model_forward(SNEPPXModel* model, const SNEPPXTensor* input, SNEPPXTensor** output) {
264 if (!model || !input || !output) return 1;
265
266 SNEPPXTensor* current = NULL;
267
268 /* For attention models with embedding, do embedding lookup on token indices */
269 if (model->attention && model->embed_weight && input->ndim == 2) {
270 /* Input: [B, S] token IDs stored as floats */
271 size_t B = input->shape[0], S = input->shape[1];
272 size_t D = model->attention->config.d_model;
273 (void)D;
274 SNEPPXTensor* idx_t = SNEPPX_tensor_create(input->shape, input->ndim, SNEPPX_INT32);
275 if (!idx_t) return 1;
276 size_t n = B * S;
277 int* id = (int*)idx_t->data;
278 float* fd = (float*)input->data;
279 for (size_t i = 0; i < n; i++) id[i] = (int)fd[i];
280 current = SNEPPX_tensor_embedding(model->embed_weight, idx_t);
281 SNEPPX_tensor_destroy(idx_t);
282 if (current) {
283 size_t sh3[] = {B, S, D};
284 SNEPPXTensor* r = SNEPPX_tensor_reshape(current, sh3, 3);
285 if (r) { SNEPPX_tensor_destroy(current); current = r; }
286 }
287 } else if (input->ndim == 3) {
288 size_t b = input->shape[0], s = input->shape[1], d = input->shape[2];
289 size_t sh[] = {b * s, d};
290 current = SNEPPX_tensor_create(sh, 2, SNEPPX_FLOAT32);
291 if (current) memcpy(current->data, input->data, b * s * d * sizeof(float));
292 } else if (input->ndim == 2) {
293 current = SNEPPX_tensor_create(input->shape, input->ndim, SNEPPX_FLOAT32);
294 if (current) memcpy(current->data, input->data, input->size * sizeof(float));
295 }
296 if (!current) return 1;
297
298 if (model->attention && model->embed_weight) {
299 size_t S = current->ndim >= 3 ? current->shape[1] : current->shape[0];
300 SNEPPXTensor* cos_full = SNEPPX_rope_precompute(S, model->attention->config.head_dim, model->attention->config.rope_base);
301 if (cos_full) {
302 SNEPPXTensor* attn_out = SNEPPX_attn_forward(model->attention, current, cos_full, cos_full);
303 SNEPPX_tensor_destroy(cos_full);
304 if (attn_out) {
305 SNEPPX_tensor_destroy(current);
306 current = attn_out;
307 }
308 }
309 }
310
311 if (model->hss_model) {
312 SNEPPXTensor* hss_out = NULL;
313 if (SNEPPX_hss_forward(model->hss_model, current, &hss_out) == 0 && hss_out) {
314 SNEPPX_tensor_destroy(current);
315 current = hss_out;
316 }
317 }
318
319 if (model->ser_model) {
320 size_t seq = current->ndim == 2 ? current->shape[0] : current->shape[1];
321 size_t dim = current->shape[current->ndim - 1];
322 SNEPPXTensor flat;
323 size_t fsh[] = {seq, dim};
324 flat.data = current->data; flat.shape = fsh; flat.ndim = 2;
325 flat.size = seq * dim; flat.item_size = sizeof(float);
326 flat.dtype = SNEPPX_FLOAT32; flat.strides = NULL;
327 SNEPPXTensor* ser_out = NULL;
328 SNEPPX_ser_forward(model->ser_model->layers[0], &flat, &ser_out);
329 if (ser_out) {
330 SNEPPX_tensor_destroy(current);
331 current = ser_out;
332 }
333 }
334
335 if (model->arc_layer) {
336 float metrics[4];
337 SNEPPXTensor* arc_out = NULL;
338 SNEPPX_arc_forward(model->arc_layer, current, &arc_out, metrics);
339 if (arc_out) {
340 SNEPPX_tensor_destroy(current);
341 current = arc_out;
342 }
343 }
344
345 if (model->npe_vm && model->npe_program) {
346 SNEPPXTensor* npe_out = NULL;
347 SNEPPX_npe_vm_run(model->npe_vm, current, &npe_out);
348 if (npe_out) {
349 SNEPPX_tensor_destroy(current);
350 current = npe_out;
351 }
352 }
353
354 if (model->fm_controller) {
355 SNEPPXTensor* fm_out = NULL;
356 int ret = SNEPPX_fm_forward(model->fm_controller, 0, current, &fm_out);
357 if (ret == 0 && fm_out) {
358 SNEPPX_tensor_destroy(current);
359 current = fm_out;
360 }
361 }
362
363 *output = current;
364 return current ? 0 : 1;
365}

375size_t SNEPPX_model_get_params(const SNEPPXModel* model, SNEPPXTensor** out, size_t max_out) {
376 if (!model) return 0;
377 size_t total = 0;
378
379 if (model->attention) {
380 total += 8; /* 8 weight tensors: w_q, b_q, w_k, b_k, w_v, b_v, w_o, b_o */
381 }
382 if (model->hss_model) {
383 total += SNEPPX_hss_get_params(model->hss_model, NULL, 0);
384 }
385 if (model->ser_model) {
386 total += SNEPPX_ser_get_params(model->ser_model, NULL, 0);
387 }
388 if (model->arc_layer) {
389 total += SNEPPX_arc_get_params(model->arc_layer, NULL, 0);
390 }
391 if (model->npe_program) {
392 total += SNEPPX_npe_get_params(model->npe_program, NULL, 0);
393 }
394 if (model->fm_controller) {
395 total += SNEPPX_fm_get_params(model->fm_controller, NULL, 0);
396 }
397 if (model->embed_weight) total++;
398 if (model->unembed_weight) total++;
399
400 size_t idx = 0;
401 if (out) {
402 if (model->attention) {
403 SNEPPXTensor* attn_params[8];
404 int n = SNEPPX_attn_get_params(model->attention, attn_params, 8);
405 for (int i = 0; i < n && idx < max_out; i++, idx++) out[idx] = attn_params[i];
406 }
407 if (model->hss_model) {
408 size_t n = SNEPPX_hss_get_params(model->hss_model, out + idx, max_out > idx ? max_out - idx : 0);
409 idx += n;
410 }
411 if (model->ser_model) {
412 size_t n = SNEPPX_ser_get_params(model->ser_model, out + idx, max_out > idx ? max_out - idx : 0);
413 idx += n;
414 }
415 if (model->arc_layer) {
416 size_t n = SNEPPX_arc_get_params(model->arc_layer, out + idx, max_out > idx ? max_out - idx : 0);
417 idx += n;
418 }
419 if (model->npe_program) {
420 size_t n = SNEPPX_npe_get_params(model->npe_program, out + idx, max_out > idx ? max_out - idx : 0);
421 idx += n;
422 }
423 if (model->fm_controller) {
424 size_t n = SNEPPX_fm_get_params(model->fm_controller, out + idx, max_out > idx ? max_out - idx : 0);
425 idx += n;
426 }
427 if (model->embed_weight && idx < max_out) out[idx++] = model->embed_weight;
428 if (model->unembed_weight && idx < max_out) out[idx++] = model->unembed_weight;
429 }
430 return total;
431}
