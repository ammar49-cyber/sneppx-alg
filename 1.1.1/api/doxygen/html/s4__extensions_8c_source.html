784int SNEPPX_mtls_authenticate(const uint8_t* cert_der, size_t cert_len, const uint8_t* key_der, size_t key_len) {
785 if (!cert_der || !key_der || cert_len < 10 || key_len < 4) return 0;
786 if (cert_der[0] != 0x30) return 0;
787 size_t off = 2;
788 size_t outer_len = 0;
789 if (cert_der[off] < 0x80) { outer_len = cert_der[off]; off++; }
790 else { int lbytes = cert_der[off] & 0x7f; off++; for (int i = 0; i < lbytes && i < 4; i++) outer_len = (outer_len << 8) | cert_der[off++]; }
791 if (off + outer_len > cert_len) return 0;
792 if (off + 2 > cert_len) return 0;
793 if (cert_der[off] != 0x30) return 0;
794 off++;
795 size_t tbs_len = 0;
796 if (cert_der[off] < 0x80) { tbs_len = cert_der[off]; off++; }
797 else { int lbytes = cert_der[off] & 0x7f; off++; for (int i = 0; i < lbytes; i++) tbs_len = (tbs_len << 8) | cert_der[off++]; }
798 if (off + tbs_len > cert_len) return 0;
799 size_t tbs_end = off + tbs_len;
800 int found_key = 0;
801 for (size_t i = off; i + key_len <= tbs_end; i++) { if (memcmp(cert_der + i, key_der, key_len) == 0) { found_key = 1; break; } }
802 if (!found_key) return 0;
803 size_t sig_start = tbs_end;
804 if (sig_start + 2 > cert_len) return 0;
805 if (cert_der[sig_start] != 0x30) return 0;
806 size_t sig_algo_len = 0;
807 if (cert_der[sig_start+1] < 0x80) sig_algo_len = cert_der[sig_start+1];
808 sig_start += 2 + sig_algo_len;
809 if (sig_start + 4 > cert_len) return 0;
810 if (cert_der[sig_start] != 0x03 && cert_der[sig_start] != 0x04) return 0;
811 return 1;
812}
