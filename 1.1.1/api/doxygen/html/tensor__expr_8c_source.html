66static SNEPPXExprNode* new_node(SNEPPXExprGraph* g, SNEPPXExprOp op, int n_inputs) {
67 if (g->num_nodes >= g->capacity) {
68 g->capacity *= 2;
69 g->nodes = (SNEPPXExprNode**)realloc(g->nodes, g->capacity * sizeof(SNEPPXExprNode*));
70 }
71 SNEPPXExprNode* n = (SNEPPXExprNode*)calloc(1, sizeof(SNEPPXExprNode));
72 if (!n) return NULL;
73 n->op = op;
74 n->id = g->next_id++;
75 n->n_inputs = n_inputs;
76 n->inputs = n_inputs > 0 ? (SNEPPXExprNode**)calloc(n_inputs, sizeof(SNEPPXExprNode*)) : NULL;
77 n->fusion_group = -1;
78 n->fuseable = 0;
79 n->materialized = 0;
80 g->nodes[g->num_nodes++] = n;
81 return n;

272int SNEPPX_expr_plan_fusions(SNEPPXExprGraph* g) {
273 if (!g->enable_fusion) { g->fused_kernel_count = g->num_nodes; return g->num_nodes; }
274 int group = 0;
275 for (int i = 0; i < g->num_nodes; i++) {
276 SNEPPXExprNode* n = g->nodes[i];
277 if (n->fusion_group >= 0) continue;
278 if (!n->fuseable) { n->fusion_group = group++; continue; }
279 /* start a fusion group */
280 n->fusion_group = group;
281 /* absorb downstream consumers that are also fuseable and single-use */
282 for (int j = i + 1; j < g->num_nodes; j++) {
283 SNEPPXExprNode* m = g->nodes[j];
284 if (m->fuseable && m->fusion_group < 0) {
285 int single_use = 1;
286 for (int k = 0; k < g->num_nodes; k++) {
287 for (int ii = 0; ii < g->nodes[k]->n_inputs; ii++) {
288 if (g->nodes[k]->inputs[ii] == m && g->nodes[k] != m) {
289 single_use = 0;
290 }
291 }
292 }
293 if (single_use) m->fusion_group = group;
294 }
295 }
296 group++;
297 }
298 g->fused_kernel_count = group;
299 return group;
300}

347static void eval_unary(SNEPPXExprGraph* g, SNEPPXExprNode* n, float* out) {
348 eval_node(g, n->inputs[0]);
349 float* in = (float*)n->inputs[0]->payload;
350 int sz = shape_size(n->shape, n->ndim);
351 for (int i = 0; i < sz; i++) {
352 float x = in[i];
353 switch (n->op) {
354 case SNEPPX_EXPR_RELU: out[i] = x > 0 ? x : 0; break;
355 case SNEPPX_EXPR_GELU: { float c=0.79788456f; out[i]=0.5f*x*(1+tanhf(c*(x+0.044715f*x*x*x))); break; }
356 case SNEPPX_EXPR_SILU: out[i] = x/(1+expf(-x)); break;
357 case SNEPPX_EXPR_TANH: out[i] = tanhf(x); break;
358 case SNEPPX_EXPR_SIGMOID: out[i] = 1/(1+expf(-x)); break;
359 case SNEPPX_EXPR_EXP: out[i] = expf(x); break;
360 case SNEPPX_EXPR_LOG: out[i] = logf(x > 1e-12f ? x : 1e-12f); break;
361 case SNEPPX_EXPR_SQRT: out[i] = sqrtf(x > 0 ? x : 0); break;
362 case SNEPPX_EXPR_CLAMP: out[i] = x < 0 ? 0 : (x > 1 ? 1 : x); break;
363 default: out[i] = x; break;
364 }
365 }

368static void eval_binary(SNEPPXExprGraph* g, SNEPPXExprNode* n, float* out) {
369 eval_node(g, n->inputs[0]);
370 eval_node(g, n->inputs[1]);
371 float* a = (float*)n->inputs[0]->payload;
372 float* b = (float*)n->inputs[1]->payload;
373 int sz = shape_size(n->shape, n->ndim);
374 for (int i = 0; i < sz; i++) {
375 float av = a[i];
376 float bv = (n->inputs[1]->op == SNEPPX_EXPR_CONST) ? n->inputs[1]->scalar : b[i];
377 switch (n->op) {
378 case SNEPPX_EXPR_ADD: out[i] = av + bv; break;
379 case SNEPPX_EXPR_SUB: out[i] = av - bv; break;
380 case SNEPPX_EXPR_MUL: out[i] = av * bv; break;
381 case SNEPPX_EXPR_DIV: out[i] = av / (bv != 0 ? bv : 1e-12f); break;
382 default: out[i] = av; break;
383 }
384 }

387static void eval_node(SNEPPXExprGraph* g, SNEPPXExprNode* n) {
388 if (n->materialized) return;
389 float* out = alloc_payload(n);
390 switch (n->op) {
391 case SNEPPX_EXPR_INPUT:
392 case SNEPPX_EXPR_CONST:
393 if (n->op == SNEPPX_EXPR_CONST) {
394 int sz = shape_size(n->shape, n->ndim);
395 for (int i = 0; i < sz; i++) out[i] = n->scalar;
396 }
397 break;
398 case SNEPPX_EXPR_ADD:
399 case SNEPPX_EXPR_SUB:
400 case SNEPPX_EXPR_MUL:
401 case SNEPPX_EXPR_DIV:
402 eval_binary(g, n, out);
403 break;
404 case SNEPPX_EXPR_RELU:
405 case SNEPPX_EXPR_GELU:
406 case SNEPPX_EXPR_SILU:
407 case SNEPPX_EXPR_TANH:
408 case SNEPPX_EXPR_SIGMOID:
409 case SNEPPX_EXPR_EXP:
410 case SNEPPX_EXPR_LOG:
411 case SNEPPX_EXPR_SQRT:
412 case SNEPPX_EXPR_CLAMP:
413 eval_unary(g, n, out);
414 break;
415 case SNEPPX_EXPR_MATMUL: {
416 eval_node(g, n->inputs[0]);
417 eval_node(g, n->inputs[1]);
418 float* a = (float*)n->inputs[0]->payload;
419 float* b = (float*)n->inputs[1]->payload;
420 SNEPPX_gemm(a, b, out, n->shape[0], n->shape[1],
421 n->inputs[0]->shape[1], 1.0f, 0.0f);
422 break;
423 }
424 case SNEPPX_EXPR_BIAS_ADD: {
425 eval_node(g, n->inputs[0]);
426 eval_node(g, n->inputs[1]);
427 float* a = (float*)n->inputs[0]->payload;
428 float* w = (float*)n->inputs[1]->payload;
429 float* bias = (n->n_inputs > 2) ? (float*)n->inputs[2]->payload : NULL;
430 SNEPPX_fused_linear_bias_avx2(a, w, bias, out,
431 n->shape[0], n->shape[1],
432 n->inputs[0]->shape[1], n->act_kind);
433 break;
434 }
435 case SNEPPX_EXPR_SUM:
436 case SNEPPX_EXPR_MEAN: {
437 eval_node(g, n->inputs[0]);
438 float* in = (float*)n->inputs[0]->payload;
439 int sz = shape_size(n->inputs[0]->shape, n->inputs[0]->ndim);
440 float s = 0;
441 for (int i = 0; i < sz; i++) s += in[i];
442 if (n->op == SNEPPX_EXPR_MEAN) s /= (sz > 0 ? sz : 1);
443 out[0] = s;
444 break;
445 }
446 default:
447 break;
448 }
449}
