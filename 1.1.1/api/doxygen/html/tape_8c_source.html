91static void tape_topological_sort(SNEPPXTape* tape, SNEPPXVariable** sorted, size_t* num_sorted) {
92 if (!tape || !sorted) return;
93 size_t n = tape->num_vars;
94 if (n == 0) return;
95 int* visited = (int*)SNEPPX_malloc(n * sizeof(int), 64);
96 if (!visited) return;
97 memset(visited, 0, n * sizeof(int));
98 *num_sorted = 0;
99
100 /* DFS-based topological sort: for each node, visit all parents first */
101 for (size_t i = 0; i < n; i++) {
102 if (visited[i]) continue;
103 /* Simple stack-based DFS post-order */
104 size_t stack_cap = 64;
105 size_t* stack = (size_t*)SNEPPX_malloc(stack_cap * sizeof(size_t), 64);
106 int* on_stack = (int*)SNEPPX_malloc(n * sizeof(int), 64);
107 if (!stack || !on_stack) { SNEPPX_free(stack, stack_cap*sizeof(size_t)); SNEPPX_free(on_stack, n*sizeof(int)); break; }
108 memset(on_stack, 0, n * sizeof(int));
109 size_t sp = 0;
110 stack[sp++] = i;
111 on_stack[i] = 1;
112
113 size_t loop_guard = 0;
114 while (sp > 0) {
115 if (++loop_guard > 1000000) { fprintf(stderr, "ERROR: tape_topological_sort infinite loop at i=%zu sp=%zu n=%zu\n", i, sp, n); break; }
116 size_t idx = stack[sp - 1];
117 SNEPPXVariable* var = tape->vars[idx];
118 int all_parents_visited = 1;
119 if (var && var->parents) {
120 for (size_t p = 0; p < var->num_parents; p++) {
121 SNEPPXVariable* parent = var->parents[p];
122 if (!parent) continue;
123 /* Find parent index in tape vars */
124 int found = 0;
125 for (size_t pi = 0; pi < n; pi++) {
126 if (tape->vars[pi] == parent) {
127 found = 1;
128 if (!visited[pi]) {
129 if (!on_stack[pi]) {
130 /* Grow stack if needed */
131 if (sp >= stack_cap) {
132 stack_cap *= 2;
133 size_t* ns = (size_t*)SNEPPX_malloc(stack_cap * sizeof(size_t), 64);
134 if (!ns) break;
135 memcpy(ns, stack, sp * sizeof(size_t));
136 SNEPPX_free(stack, stack_cap/2 * sizeof(size_t));
137 stack = ns;
138 }
139 stack[sp++] = pi;
140 on_stack[pi] = 1;
141 all_parents_visited = 0;
142 }
143 }
144 break;
145 }
146 }
147 /* If parent not found on tape, treat as visited (no backward needed) */
148 if (!found) continue;
149 if (!all_parents_visited) break;
150 }
151 }
152 if (all_parents_visited) {
153 visited[idx] = 1;
154 sorted[(*num_sorted)++] = var;
155 on_stack[idx] = 0;
156 sp--;
157 }
158 }
159 SNEPPX_free(on_stack, n * sizeof(int));
160 SNEPPX_free(stack, stack_cap * sizeof(size_t));
161 }
162 SNEPPX_free(visited, n * sizeof(int));
163}

170void SNEPPX_tape_backward(SNEPPXTape* tape, SNEPPXVariable* loss) {
171 if (!tape || !loss) return;
172
173 if (loss->grad) SNEPPX_tensor_destroy(loss->grad);
174 loss->grad = SNEPPX_tensor_ones(loss->data->shape, loss->data->ndim, SNEPPX_FLOAT32);
175
176 SNEPPXVariable** sorted = NULL;
177 size_t n = tape->num_vars;
178 size_t num_sorted = 0;
179 if (n > 0) {
180 sorted = (SNEPPXVariable**)SNEPPX_malloc(n * sizeof(SNEPPXVariable*), 64);
181 if (sorted) {
182 tape_topological_sort(tape, sorted, &num_sorted);
183 }
184 }
185
186 for (size_t i = num_sorted; i > 0; i--) {
187 SNEPPXVariable* var = sorted ? sorted[i - 1] : tape->vars[i - 1];
188 if (!var || !var->backward_fn || !var->grad) continue;
189 if (!var->backward_ctx && var->checkpointed && var->recompute_ctx) {
190 var->backward_ctx = var->recompute_ctx(var, var->params, var->param_count);
191 }
192 if (!var->backward_ctx) continue;
193 var->backward_fn(var->backward_ctx, var->grad);
194 if (var->backward_ctx && var->free_ctx) {
195 var->free_ctx(var->backward_ctx);
196 var->backward_ctx = NULL;
197 }
198 for (size_t p = 0; p < var->num_parents; p++) {
199 SNEPPXVariable* parent = var->parents[p];
200 if (!parent) continue;
201 parent->ref_count--;
202 }
203 }
204
205 if (sorted) SNEPPX_free(sorted, n * sizeof(SNEPPXVariable*));
206}

226float SNEPPX_tape_global_norm(SNEPPXTape* tape) {
227 if (!tape) return 0.0f;
228 float sum_sq = 0.0f;
229 for (size_t i = 0; i < tape->num_vars; i++) {
230 SNEPPXVariable* var = tape->vars[i];
231 if (!var || !var->grad) continue;
232 float* gd = (float*)var->grad->data;
233 size_t sz = var->grad->size;
234 for (size_t j = 0; j < sz; j++) sum_sq += gd[j] * gd[j];
235 }
236 return sqrtf(sum_sq);
237}
