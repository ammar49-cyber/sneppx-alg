64static uint32_t fuzz_crc32(const uint8_t *data, size_t len) {
65 uint32_t crc = 0xFFFFFFFF;
66 static uint32_t table[256];
67 static int table_init = 0;
68 if (!table_init) {
69 for (int i = 0; i < 256; i++) {
70 uint32_t c = i;
71 for (int j = 0; j < 8; j++)
72 c = (c >> 1) ^ (c & 1 ? 0xEDB88320 : 0);
73 table[i] = c;
74 }
75 table_init = 1;
76 }
77 for (size_t i = 0; i < len; i++)
78 crc = table[(crc ^ data[i]) & 0xFF] ^ (crc >> 8);
79 return crc ^ 0xFFFFFFFF;
80}

82static int fuzz_mutate_flip(uint8_t *buf, size_t *len, size_t max_len) {
83 if (*len == 0) return 0;
84 int pos = rand() % (*len);
85 int bits = (rand() % 8) + 1;
86 uint8_t mask = 0;
87 for (int i = 0; i < bits; i++) mask |= 1 << (rand() % 8);
88 buf[pos] ^= mask;
89 return 1;

92static int fuzz_mutate_erase(uint8_t *buf, size_t *len) {
93 if (*len < 2) return 0;
94 int pos = rand() % (*len);
95 int count = (rand() % (*len - pos)) + 1;
96 memmove(buf + pos, buf + pos + count, *len - pos - count);
97 *len -= count;
98 return 1;
99}

101static int fuzz_mutate_insert(uint8_t *buf, size_t *len, size_t max_len) {
102 if (*len + 16 >= max_len) return 0;
103 int pos = rand() % (*len + 1);
104 int count = (rand() % 16) + 1;
105 memmove(buf + pos + count, buf + pos, *len - pos);
106 for (int i = 0; i < count; i++) buf[pos + i] = rand() & 0xFF;
107 *len += count;
108 return 1;

111static int fuzz_mutate_splice(uint8_t *buf, size_t *len, size_t max_len) {
112 if (corpus_count < 2 || *len < 4) return 0;
113 int src_idx = rand() % corpus_count;
114 if (corpus[src_idx].len < 4) return 0;
115 int pos1 = rand() % (*len - 2) + 1;
116 int pos2 = rand() % (corpus[src_idx].len - 2) + 1;
117 int count = (rand() % ((*len - pos1 > corpus[src_idx].len - pos2) ?
118 corpus[src_idx].len - pos2 : *len - pos1)) + 1;
119 memcpy(buf + pos1, corpus[src_idx].data + pos2, count);
120 return 1;
121}

123static int fuzz_mutate_havoc(uint8_t *buf, size_t *len, size_t max_len) {
124 int ops = (rand() % 8) + 1;
125 for (int i = 0; i < ops; i++) {
126 int choice = rand() % 4;
127 switch (choice) {
128 case 0: fuzz_mutate_flip(buf, len, max_len); break;
129 case 1: fuzz_mutate_erase(buf, len); break;
130 case 2: fuzz_mutate_insert(buf, len, max_len); break;
131 case 3: if (*len > 0) buf[rand() % (*len)] = rand() & 0xFF; break;
132 }
133 }
134 return 1;
135}

231int SNEPPX_fuzzer_generate(uint8_t *buf, size_t *len, size_t max_len) {
232 if (!buf || !len || max_len < 16) return -1;
233 if (corpus_count > 0 && rand() % 3 > 0) {
234 int idx = rand() % corpus_count;
235 if (corpus[idx].len > 0 && corpus[idx].len <= max_len) {
236 memcpy(buf, corpus[idx].data, corpus[idx].len);
237 *len = corpus[idx].len;
238 double r = (double)rand() / RAND_MAX;
239 if (r < fuzzer_config.mutate_prob_flip) fuzz_mutate_flip(buf, len, max_len);
240 else if (r < fuzzer_config.mutate_prob_flip + fuzzer_config.mutate_prob_erase) fuzz_mutate_erase(buf, len);
241 else if (r < fuzzer_config.mutate_prob_flip + fuzzer_config.mutate_prob_erase + fuzzer_config.mutate_prob_insert) fuzz_mutate_insert(buf, len, max_len);
242 else if (r < fuzzer_config.mutate_prob_flip + fuzzer_config.mutate_prob_erase + fuzzer_config.mutate_prob_insert + fuzzer_config.mutate_prob_splice) fuzz_mutate_splice(buf, len, max_len);
243 else fuzz_mutate_havoc(buf, len, max_len);
244 total_mutations++;
245 return 0;
246 }
247 }
248 switch (current_protocol) {
249 case FUZZ_PROTO_TLS: return fuzz_generate_tls_client_hello(buf, len, max_len);
250 case FUZZ_PROTO_HTTP: return fuzz_generate_http_request(buf, len, max_len);
251 case FUZZ_PROTO_QUIC: return fuzz_generate_quic_packet(buf, len, max_len);
252 default:
253 for (size_t i = 0; i < 64 && i < max_len; i++) buf[i] = rand() & 0xFF;
254 *len = 64;
255 return 0;
256 }

267int SNEPPX_fuzzer_execute(const uint8_t *input, size_t len, fuzz_result_t *result) {
268 if (!input || !result) return -1;
269 total_executions++;
270 memset(result, 0, sizeof(fuzz_result_t));
271 result->input_crc = fuzz_crc32(input, len);
272 result->execution_id = total_executions;
273 int crash_detected = 0;
274 if (fuzzer_config.enable_crash_detection) {
275 if (len > 0 && input[len - 1] == 0xFF) crash_detected = 1;
276 for (size_t i = 0; i < len - 1 && !crash_detected; i++)
277 if (input[i] == 'C' && input[i+1] == 'R' && input[i+2] == 'A' && input[i+3] == 'S' && input[i+4] == 'H')
278 crash_detected = 1;
279 }
280 if (crash_detected) {
281 result->crashed = 1;
282 result->crash_type = FUZZ_CRASH_SEGFAULT;
283 total_crashes++;
284 return 0;
285 }
286 if (fuzzer_config.enable_coverage && coverage_count < 65536) {
287 uint32_t hash = fuzz_crc32(input, len > 16 ? 16 : len);
288 int found = 0;
289 for (int i = 0; i < coverage_count; i++)
290 if (coverage_map[i].pc == hash) { coverage_map[i].hit_count++; found = 1; break; }
291 if (!found) {
292 coverage_map[coverage_count].pc = hash;
293 coverage_map[coverage_count].hit_count = 1;
294 coverage_count++;
295 }
296 result->new_coverage = !found;
297 }
298 if (corpus_count < FUZZER_CORPUS_SIZE && (result->new_coverage || rand() % 100 < 5)) {
299 corpus[corpus_count].data = (uint8_t *)malloc(len);
300 if (corpus[corpus_count].data) {
301 memcpy(corpus[corpus_count].data, input, len);
302 corpus[corpus_count].len = len;
303 corpus[corpus_count].crc = result->input_crc;
304 corpus[corpus_count].fitness = result->new_coverage ? 10 : 1;
305 corpus_count++;
306 }
307 }
308 return 0;
309}

366int SNEPPX_fuzzer_fuzz_target(const uint8_t *target_data, size_t target_len, int iterations, fuzz_result_t *results, int max_results) {
367 if (!target_data || !results) return -1;
368 int found = 0;
369 uint8_t buf[FUZZER_MAX_INPUT_SIZE];
370 size_t buf_len;

376 if (results[found].crashed || results[found].new_coverage) found++;
377 }
378 return found;
