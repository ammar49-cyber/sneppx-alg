55SNEPPXTrainer* SNEPPX_trainer_create(SNEPPXModel* model, const SNEPPXTrainConfig* config) {
56 if (!model || !config) return NULL;
57 SNEPPXTrainer* trainer = (SNEPPXTrainer*)SNEPPX_malloc(sizeof(SNEPPXTrainer), 64);
58 if (!trainer) return NULL;
59 memset(trainer, 0, sizeof(SNEPPXTrainer));
60
61 trainer->model = model;
62
63 SNEPPXOptimizerConfig opt_cfg = SNEPPX_optimizer_config_default();
64 opt_cfg.learning_rate = config->learning_rate;
65 trainer->optimizer = SNEPPX_optimizer_create(&opt_cfg);
66 if (!trainer->optimizer) { SNEPPX_free(trainer, sizeof(SNEPPXTrainer)); return NULL; }
67
68 size_t hist_shape[] = {1024};
69 trainer->loss_history = SNEPPX_tensor_zeros(hist_shape, 1, SNEPPX_FLOAT32);
70 if (!trainer->loss_history) {
71 SNEPPX_optimizer_destroy(trainer->optimizer);
72 SNEPPX_free(trainer, sizeof(SNEPPXTrainer));
73 return NULL;
74 }
75
76 trainer->step_count = 0;
77
78 if (config->use_cuda_optimizer && SNEPPX_trainer_cuda_available()) {
79 size_t nw = SNEPPX_model_get_params(trainer->model, NULL, 0);
80 if (nw > 0) {
81 SNEPPXTensor** params = (SNEPPXTensor**)SNEPPX_malloc(nw * sizeof(SNEPPXTensor*), 64);
82 if (params) {
83 SNEPPX_model_get_params(trainer->model, params, nw);
84 if (SNEPPX_trainer_cuda_init(params, nw) != 0) {
85 printf("Warning: CUDA optimizer init failed, falling back to CPU\n");
86 }
87 SNEPPX_free(params, nw * sizeof(SNEPPXTensor*));
88 }
89 }
90 }

114float SNEPPX_trainer_train_step(SNEPPXTrainer* trainer, const SNEPPXTensor* batch_input, const SNEPPXTensor* batch_target) {
115 if (!trainer || !batch_input || !batch_target) return -1.0f;
116
117 size_t nw = SNEPPX_model_get_params(trainer->model, NULL, 0);
118
119 SNEPPXTensor** params = (SNEPPXTensor**)SNEPPX_malloc(nw * sizeof(SNEPPXTensor*), 64);
120 if (!params) return -1.0f;
121 SNEPPX_model_get_params(trainer->model, params, nw);
122
123 SNEPPXTape* tape = SNEPPX_tape_create();
124
125 SNEPPXVariable** wv = (SNEPPXVariable**)SNEPPX_malloc(nw * sizeof(SNEPPXVariable*), 64);
126 if (!wv) { SNEPPX_free(params, nw * sizeof(SNEPPXTensor*)); SNEPPX_tape_destroy(tape); return -1.0f; }
127 for (size_t i = 0; i < nw; i++) {
128 wv[i] = SNEPPX_variable_create(params[i], 1);
129 }
130
131 /* Prepare input variable: for attention models, do embedding lookup */
132 SNEPPXVariable* inv = NULL;
133 if (trainer->model->attention && trainer->model->embed_weight && batch_input->ndim >= 2) {
134 /* Input is token indices [B, S] — do embedding lookup */
135 float* fdata = (float*)batch_input->data;
136 size_t n = batch_input->shape[0] * batch_input->shape[1];
137 SNEPPXTensor* idx_t = SNEPPX_tensor_create(batch_input->shape, batch_input->ndim, SNEPPX_INT32);
138 if (!idx_t) return -1.0f;
139 int* id = (int*)idx_t->data;
140 for (size_t i = 0; i < n; i++) id[i] = (int)fdata[i];
141 SNEPPXVariable* idx_v = SNEPPX_variable_create(idx_t, 0);
142 SNEPPX_tape_record(tape, idx_v);
143 inv = SNEPPX_embedding(tape, wv[nw - 2], idx_v);
144 /* Embedding returns 2D [B*S, D]; reshape to 3D [B, S, D] for attention */
145 if (inv && inv->data && inv->data->ndim == 2) {
146 size_t b = batch_input->shape[0], s = batch_input->shape[1];
147 size_t d = inv->data->shape[1];
148 size_t sh3[] = {b, s, d};
149 inv = SNEPPX_reshape(tape, inv, sh3, 3);
150 }
151 } else {
152 /* Input is raw float features */
153 size_t in_size = batch_input->size;
154 SNEPPXTensor* inv_t = SNEPPX_tensor_create(batch_input->shape, batch_input->ndim, SNEPPX_FLOAT32);
155 if (!inv_t) return -1.0f;
156 memcpy(inv_t->data, batch_input->data, in_size * sizeof(float));
157 inv = SNEPPX_variable_create(inv_t, 0);
158 }
159
160 /* Build the unified train graph */
161 SNEPPXVariable* outv = NULL;
162 int ret = SNEPPX_model_build_train_graph(trainer->model, tape, inv, wv, nw, &outv);
163
164 /* For decoder models (attention), unembed to vocab logits */
165 if (ret == 0 && outv && trainer->model->embed_weight && trainer->model->unembed_weight && nw >= 2) {
166 if (outv->data->ndim == 3) {
167 size_t outf_sh[] = {outv->data->shape[0] * outv->data->shape[1], outv->data->shape[2]};
168 SNEPPXVariable* outf = SNEPPX_reshape(tape, outv, outf_sh, 2);
169 outv = SNEPPX_matmul(tape, outf, wv[nw - 1]);
170 } else if (outv->data->ndim == 2) {
171 outv = SNEPPX_matmul(tape, outv, wv[nw - 1]);
172 } else if (outv->data->ndim == 1) {
173 size_t unsq_sh[] = {1, outv->data->shape[0]};
174 SNEPPXVariable* unsq = SNEPPX_reshape(tape, outv, unsq_sh, 2);
175 outv = SNEPPX_matmul(tape, unsq, wv[nw - 1]);
176 }
177 }
178
179 float loss_val = -1.0f;
180 if (ret == 0 && outv) {
181 size_t out_sz = outv->data->size;
182 size_t tgt_sz = batch_target->size;
183 size_t copy_sz = out_sz < tgt_sz ? out_sz : tgt_sz;
184 SNEPPXTensor* tgt_flat = SNEPPX_tensor_zeros(&out_sz, 1, SNEPPX_FLOAT32);
185 if (tgt_flat) {
186 memcpy((float*)tgt_flat->data, (float*)batch_target->data, copy_sz * sizeof(float));
187 }
188 SNEPPXVariable* tgv = SNEPPX_variable_create(tgt_flat, 0);
189 SNEPPXVariable* loss = SNEPPX_mse_loss(tape, outv, tgv);
190
191 if (loss) {
192 loss_val = SNEPPX_variable_item(loss);
193 SNEPPX_tape_backward(tape, loss);
194
195 SNEPPXTensor** grads = (SNEPPXTensor**)SNEPPX_malloc(nw * sizeof(SNEPPXTensor*), 64);
196 if (grads) {
197 for (size_t i = 0; i < nw; i++) {
198 grads[i] = wv[i]->grad;
199 }
200 if (trainer->config.use_cuda_optimizer) {
201 SNEPPX_trainer_cuda_optimizer_step(trainer->optimizer, params, grads, nw);
202 } else {
203 SNEPPX_optimizer_step(trainer->optimizer, params, grads, nw);
204 }
205 SNEPPX_free(grads, nw * sizeof(SNEPPXTensor*));
206 }
207 }
208
209 tgv->data = NULL; SNEPPX_variable_destroy(tgv);
210 }
211
212 size_t idx = trainer->step_count;
213 if (idx < trainer->loss_history->size) {
214 ((float*)trainer->loss_history->data)[idx] = loss_val;
215 }
216
217 trainer->step_count++;
218
219 for (size_t i = 0; i < nw; i++) { wv[i]->data = NULL; wv[i]->grad = NULL; SNEPPX_variable_destroy(wv[i]); }
220 /* inv was recorded on the tape (has gradient), so tape handles its destruction without explicit destroy */
221 SNEPPX_free(wv, nw * sizeof(SNEPPXVariable*));
222 SNEPPX_free(params, nw * sizeof(SNEPPXTensor*));
223 SNEPPX_tape_destroy(tape);
224
225 if (trainer->config.log_interval > 0 && trainer->step_count % trainer->config.log_interval == 0) {
226 printf("[Step %zu] loss = %.6f\n", trainer->step_count, (double)loss_val);
227 }
228
229 return loss_val;
230}

240float SNEPPX_trainer_evaluate(SNEPPXTrainer* trainer, const SNEPPXTensor* val_input, const SNEPPXTensor* val_target) {
241 if (!trainer || !val_input || !val_target) return -1.0f;
242
243 SNEPPXTensor* output = NULL;
244 int ret = SNEPPX_model_forward(trainer->model, val_input, &output);
245 if (ret != 0 || !output) return -1.0f;
246
247 float* od = (float*)output->data;
248 float* td = (float*)val_target->data;
249 size_t sz = output->size < val_target->size ? output->size : val_target->size;
250
251 float sum_sq = 0.0f;
252 for (size_t i = 0; i < sz; i++) {
253 float d = od[i] - td[i];
254 sum_sq += d * d;
255 }
256 float loss = sum_sq / (float)(sz > 0 ? sz : 1);
257
258 SNEPPX_tensor_destroy(output);
259 return loss;
260}

310int SNEPPX_trainer_load_checkpoint(SNEPPXTrainer* trainer, const char* path) {
311 if (!trainer || !path) return 1;
312 FILE* f = fopen(path, "rb");
313 if (!f) return 1;
314
315 unsigned int magic, version;
316 if (fread(&magic, sizeof(magic), 1, f) != 1 || magic != 0x41524958) {
317 fclose(f); return 1;
318 }
319 if (fread(&version, sizeof(version), 1, f) != 1 || version < 1 || version > 2) {
320 fclose(f); return 1;
321 }
322 if (fread(&trainer->step_count, sizeof(trainer->step_count), 1, f) != 1) {
323 fclose(f); return 1;
324 }
325
326 if (version >= 2) {
327 size_t nw = 0;
328 if (fread(&nw, sizeof(nw), 1, f) != 1) { fclose(f); return 1; }
329 if (nw > 0) {
330 SNEPPXTensor** params = (SNEPPXTensor**)SNEPPX_malloc(nw * sizeof(SNEPPXTensor*), 64);
331 if (params) {
332 size_t actual = SNEPPX_model_get_params(trainer->model, params, nw);
333 for (size_t i = 0; i < actual && i < nw; i++) {
334 if (!params[i]) continue;
335 unsigned int ndim;
336 size_t stored_shape[8];
337 size_t stored_size;
338 if (fread(&ndim, sizeof(ndim), 1, f) != 1) break;
339 if (ndim > 8) { for (size_t j = 0; j < ndim; j++) { size_t tmp; fread(&tmp, sizeof(tmp), 1, f); } break; }
340 if (fread(stored_shape, sizeof(size_t), ndim, f) != ndim) break;
341 if (fread(&stored_size, sizeof(stored_size), 1, f) != 1) break;
342 size_t copy_sz = stored_size < params[i]->size ? stored_size : params[i]->size;
343 if (fread(params[i]->data, sizeof(float), copy_sz, f) != copy_sz) break;
344 if (stored_size > copy_sz) { float tmp; for (size_t j = copy_sz; j < stored_size; j++) fread(&tmp, sizeof(float), 1, f); }
345 }
346 SNEPPX_free(params, nw * sizeof(SNEPPXTensor*));
347 }
348 }
349 }
350
351 fclose(f);
352 return 0;
353}
