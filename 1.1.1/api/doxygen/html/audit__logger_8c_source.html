25static uint32_t audit_crc32(const void* data, size_t len) {
26 const unsigned char* buf = (const unsigned char*)data;
27 uint32_t crc = 0xFFFFFFFF;
28 for (size_t i = 0; i < len; i++) {
29 crc ^= buf[i];
30 for (int j = 0; j < 8; j++)
31 crc = (crc >> 1) ^ (0xEDB88320 & -(crc & 1));
32 }
33 return ~crc;
34}

73int SNEPPX_audit_log(SNEPPXAuditLogger* audit, int event_type,
74 const char* description, uint64_t related_address) {
75 if (!audit || !audit->enabled || !description) return -1;
76 if (audit->entry_count >= SNEPPX_AUDIT_MAX_ENTRIES) return -1;
77 SNEPPXAuditEntry* e = &audit->entries[audit->entry_count++];
78 e->timestamp = (uint64_t)time(NULL);
79 e->event_type = event_type;
80 strncpy(e->description, description, SNEPPX_AUDIT_DESC_LEN - 1);
81 e->related_address = related_address;
82 e->crc = audit_crc32(e, offsetof(SNEPPXAuditEntry, crc));
83 audit->chain_crc = audit_crc32(&audit->chain_crc, sizeof(audit->chain_crc)) ^ e->crc;
84 return 0;
85}

114int SNEPPX_audit_verify_chain(SNEPPXAuditLogger* audit) {
115 if (!audit) return 0;
116 uint32_t expected_chain = 0;
117 for (int i = 0; i < audit->entry_count; i++) {
118 uint32_t entry_crc = audit_crc32(&audit->entries[i], offsetof(SNEPPXAuditEntry, crc));
119 if (entry_crc != audit->entries[i].crc) return 0;
120 expected_chain = audit_crc32(&expected_chain, sizeof(expected_chain)) ^ entry_crc;
121 }
122 return (expected_chain == audit->chain_crc) ? 1 : 0;
123}

195int SNEPPX_audit_sign_entry(SNEPPXAuditLogger* audit, int index, const uint8_t* key) {
196 if (!audit || !key) return -1;
197 if (index < 0 || index >= audit->entry_count) return -1;
198 SNEPPXAuditEntry* e = &audit->entries[index];
199 uint32_t sig = 0;
200 for (size_t i = 0; i < SNEPPX_AUDIT_HMAC_KEY_LEN; i++)
201 sig ^= ((uint32_t)key[i] << ((i % 4) * 8));
202 sig ^= e->crc;
203 e->crc = sig;
204 return 0;
205}
