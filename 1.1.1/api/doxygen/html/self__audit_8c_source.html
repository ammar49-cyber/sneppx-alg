50int SNEPPX_self_audit_add_check(SNEPPXSelfAudit* audit, const char* name,
51 SNEPPXAuditStatus status, const char* details) {
52 if (!audit || !name || audit->check_count >= SNEPPX_AUDIT_MAX_CHECKS) return -1;
53 SNEPPXAuditCheck* c = &audit->checks[audit->check_count];
54 strncpy(c->check_name, name, SNEPPX_CHECK_DESC_LEN - 1);
55 c->status = status;
56 if (details) strncpy(c->details, details, SNEPPX_CHECK_DESC_LEN - 1);
57 switch (status) {
58 case SNEPPX_AUDIT_PASS: audit->total_passed++; break;
59 case SNEPPX_AUDIT_FAIL: audit->total_failed++; break;
60 case SNEPPX_AUDIT_WARN: audit->total_warnings++; break;
61 default: break;
62 }
63 return audit->check_count++;
64}

114double SNEPPX_self_audit_score(SNEPPXSelfAudit* audit) {
115 if (!audit || audit->check_count == 0) return 0.0;
116 int total = audit->total_passed + audit->total_failed + audit->total_warnings;
117 if (total == 0) return 0.0;
118 double score = (double)audit->total_passed / (double)total * 100.0;
119 score -= (double)audit->total_failed / (double)total * 50.0;
120 score -= (double)audit->total_warnings / (double)total * 10.0;
121 return (score < 0) ? 0.0 : score;
122}

446static double audit_compute_category_score_internal(SNEPPXSelfAudit* audit, const char* category) {
447 if (!audit || !category) return 0.0;
