37int sneppx_ep_init(SNEPPX_ExpertParallel** ep,
38 const SNEPPX_DistributedConfig* config) {
39 if (!ep || !config) return -1;
40
41 SNEPPX_ExpertParallel* e = (SNEPPX_ExpertParallel*)calloc(1, sizeof(SNEPPX_ExpertParallel));
42 if (!e) return -1;
43
44 e->num_experts = config->num_experts;
45 e->num_local_experts = config->num_experts / config->expert_parallel_size;
46 e->ep_size = config->expert_parallel_size;
47 e->ep_rank = config->rank % config->expert_parallel_size;
48
49 // Build expert-to-rank mapping
50 e->expert_to_rank = (int*)calloc(config->num_experts, sizeof(int));
51 e->local_expert_ids = (int*)calloc(e->num_local_experts, sizeof(int));
52
53 for (int i = 0; i < config->num_experts; i++) {
54 e->expert_to_rank[i] = i % config->expert_parallel_size;
55 }
56
57 int local_idx = 0;
58 for (int i = e->ep_rank; i < config->num_experts; i += config->expert_parallel_size) {
59 if (local_idx < e->num_local_experts) {
60 e->local_expert_ids[local_idx++] = i;
61 }
62 }
63
64 *ep = e;
65 return 0;
66}
